      case 'eok': return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) || Math.abs(v) >= 100 ? 0 : 1)), '억'], rel: true });
      case 'krw': return F({ parts: (v) => krw(v, d.decimals), rel: true });
      case 'usd': return F({ parts: (v) => usd(v, d.decimals), rel: true, usd: true });
      case 'pct': return F({ parts: (v) => [num(v, dec(v, 1)), '%'], rel: false, dp: d.decimals });
      case 'days': return F({ parts: (v) => [num(v, dec(v, 1)), tx(root || d, '일')], rel: true });
      default: return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) ? 0 : 1)), d.unit || ''], rel: true });
    }
  }
  const show = (f, v) => withUnit(...f.parts(v));
  // 그릴 수 있는 숫자인지(빈 값 · 빈 글자 · NaN 제외)
  const fin = (v) => v != null && v !== '' && isFinite(v);
  // 비율의 차이(%p) 숫자: 소수 자릿수는 decimals(비우면 한 자리)를 따릅니다. 0이 아닌 차이가 0.0으로 보일 만큼 작으면
  // 자리를 늘려(최대 세 자리) 1.12%에서 1.16%로 오른 것을 0.0%p가 아니라 0.03%p로 적습니다.
  function pp(f, diff) {
    const a = Math.abs(diff);
    let k = f.dp != null ? f.dp : 1;
    while (k < 3 && a > 1e-9 && Math.round(a * 10 ** k) === 0) k++;
    return num(a, k);
  }
  // 증감: 방향은 화살표로, 좋고 나쁨(tone)은 better(up · down · neutral)로 정합니다. 비율은 %p로 적습니다.
  function delta(f, value, base, better) {
    const diff = value - base;
    const dir = Math.abs(diff) < 1e-9 ? 0 : Math.sign(diff);
    const want = better === 'down' ? -1 : better === 'neutral' ? 0 : 1;
    const tone = dir === 0 || want === 0 ? 'flat' : dir === want ? 'good' : 'bad';
    let text;
    if (!f.rel) text = tx(f, '%s%p', pp(f, diff));
    else if (base) text = `${num(Math.abs(diff / base) * 100, 1)}%`;
    else text = show(f, Math.abs(diff));
    return { dir, tone, text };
  }
  function arrow(dir) {
    const s = svgEl('svg', { class: 'vzb-arrow', viewBox: '0 0 12 12', 'aria-hidden': 'true' });
    s.append(svgEl('path', { d: dir > 0 ? 'M3.5 8.5l5-5M4.5 3.5h4v4' : 'M3.5 3.5l5 5M8.5 4.5v4h-4' }));
    return s;
  }
  // 대표 수치(단위는 작게)
  function figure(f, value) {
    const n = el('div', 'vzb-num');
    const [text, unit] = f.parts(value);
    n.append(text);
    if (unit) n.append(el('span', 'vzb-unit', unit));
    return n;
  }
  // 증감 배지: 화살표는 오르내림, 색은 좋고 나쁨(better)입니다.
  function badge(f, value, base, better) {
    const dl = delta(f, value, base, better);
    const b = el('span', `vzb-badge vzb-${dl.tone}`);
    if (dl.dir) b.append(arrow(dl.dir), el('span', 'vzb-sr', tx(f, dl.dir > 0 ? '증가' : '감소')));
    b.append(dl.text);
    return b;
  }
  // 머리: 제목 · 기간 부제(subtitle) · 대표 수치와 바로 뒤의 증감 배지. value가 null이면 제목만 둡니다.
  // 배지는 숫자 바로 뒤에 붙어 있고, 카드가 좁아 한 줄에 다 들어가지 않으면 비교 문구만 숫자 아래 줄로 내려갑니다.
  // 내려갔는지는 카드의 vzb-wrapped로 알려, 높이가 1단위인 카드가 아래 여백으로 그 줄을 받을 수 있게 합니다.
  function head(card, d, f, value, cmp, cmpText) {
    const hd = el('div', 'vzb-hd');
    hd.append(el('div', 'vzb-title', d.title || ''));
    if (d.subtitle) hd.append(el('div', 'vzb-sub', d.subtitle));
    card.append(hd);
    if (value == null) return;
    const fig = el('div', 'vzb-fig');
    const n = figure(f, value);
    fig.append(n);
    if (cmp && cmp.value != null) {
      const line = el('span', 'vzb-delta'), vs = el('span', 'vzb-vs', cmpText || tx(d, '%s 대비', cmp.label || tx(d, '비교 기간')));
      line.append(badge(f, value, cmp.value, d.better), vs);
      fig.append(line);
      onSize(fig, () => card.classList.toggle('vzb-wrapped', vs.offsetTop >= n.offsetTop + n.offsetHeight - 1));   // 숫자 칸 아래에서 시작하면 내려간 것
    }
    card.append(fig);
  }
  // 움직임 값(tokens.css의 --vzb-follow … --vzb-rest). 스프링은 '세기 잦아듦' 두 숫자이고, 시간은 초(s)나 밀리초(ms)입니다.
  // 역할: Follow(끄는 동안 손에 붙음) · Settle(놓이거나 열릴 때 자리를 잡음) · Pop(눌렸다 튀어 오름) · Move(값의 자리로 감, 넘치지 않음) ·
  // Fade(투명도와 색) · Press(눌리는 순간) · Lead(시작 전의 뜸) · Seq(차례의 전체 시간) · Rest(되풀이 전의 쉼). 읽지 못한 값은 아래 기본값을 씁니다.
  const MOTION = { follow: '1000 63', settle: '340 26', pop: '520 21', move: '200 28.3', fade: '0.14s', press: '0.08s', lead: '400ms', seq: '1200ms', rest: '2600ms' };
  // 스프링이 0에서 1로 갈 때의 자리(t는 초). 잦아듦이 2√세기에 닿으면 넘치지 않고 곧장 갑니다.
  function springAt(k, c) {
    const w = Math.sqrt(k), z = c / (2 * w);
    if (z < 0.999) { const wd = w * Math.sqrt(1 - z * z); return (t) => 1 - Math.exp(-z * w * t) * (Math.cos(wd * t) + ((z * w) / wd) * Math.sin(wd * t)); }
    return (t) => 1 - (1 + w * t) * Math.exp(-w * t);
  }
  // 스프링을 CSS 전환으로 옮긴 [시간, 곡선]: 목표의 0.5% 안에 든 때까지를 시간으로 잡고, 그 사이를 33점으로 잇습니다.
  function springCss(k, c) {
    const f = springAt(k, c), pts = [];
    let d = 0;
    for (let t = 0; t <= 3; t += 0.005) if (Math.abs(f(t) - 1) > 0.005) d = t;
    d = Math.round((d + 0.01) * 100) / 100;
    for (let i = 0; i <= 32; i++) pts.push(i === 0 ? 0 : i === 32 ? 1 : +f((d * i) / 32).toFixed(3));
    return [`${d}s`, `linear(${pts.join(', ')})`];
  }
  // motion(card): 그 카드에 적용되는 움직임 값. pair(이름)은 스프링의 [세기, 잦아듦], ms(이름)은 밀리초,
  // curve(이름)은 { at(지난 밀리초) → 0에서 1까지의 자리, dur: 다 가는 데 걸리는 밀리초 }, css(이름)은 CSS 전환의 [시간, 곡선]입니다.
  function motion(card) {
    const cs = getComputedStyle(card), raw = (n) => cs.getPropertyValue(`--vzb-${n}`).trim().replace(/\s+/g, ' ');
    const two = (v) => { const p = v.split(' ').map(Number); return p.length === 2 && p[0] > 0 && p[1] > 0 ? p : null; };
    const pair = (n) => two(raw(n)) || two(MOTION[n]);
    const time = (v) => { const x = parseFloat(v); return isFinite(x) && x >= 0 && /s$/.test(v) ? (/ms$/.test(v) ? x : x * 1000) : null; };
    const ms = (n) => { const v = time(raw(n)); return v == null ? time(MOTION[n]) : v; };
    const css = (n) => springCss(...pair(n));
    const curve = (n) => { const f = springAt(...pair(n)); return { at: (t) => (t <= 0 ? 0 : f(t / 1000)), dur: parseFloat(css(n)[0]) * 1000 }; };
    return { pair, ms, css, curve };
  }
  // 스프링 값을 :root나 카드에서 바꿔 적었으면, CSS 전환이 쓰는 시간과 곡선(--vzb-…-t · -e)을 그 값으로 다시 계산해 카드에 적습니다.
  function tune(card) {
    const m = motion(card);
    for (const n of ['settle', 'pop', 'move']) {
      if (m.pair(n).join(' ') === MOTION[n]) continue;
      const [t, e] = m.css(n);
      card.style.setProperty(`--vzb-${n}-t`, t);
      card.style.setProperty(`--vzb-${n}-e`, e);
    }
  }
  // 스프링으로 목표에 다가가는 값: s = { x: 지금 자리, v: 속도 }를 dt(밀리초)만큼 옮깁니다. 가던 중에 목표가 바뀌어도 속도를 이어 갑니다.
  function springTo(s, to, k, c, dt) {
    let n = Math.max(1, Math.ceil(dt / 4));
    const h = dt / 1000 / n;
    while (n--) { s.v += (-k * (s.x - to) - c * s.v) * h; s.x += s.v * h; }
  }
  // 장면 잇기: 데이터가 바뀌어 카드를 갈아 끼울 때, 앞 카드가 화면에 그리고 있던 장면(값과 자리)을 새 카드에 넘깁니다.
  // 카드는 keep(card, 장면을 돌려주는 함수)로 맡기고, 새 카드는 before(card)로 받아 그 장면에서 새 값으로 옮겨 갑니다.
  const SCENE = new WeakMap(), FROM = new WeakMap();
  const keep = (card, fn) => SCENE.set(card, fn);
  const before = (card) => { const s = FROM.get(card); FROM.delete(card); return s || null; };
  // 같은 종류의 카드를 모두 찾아 그립니다. 코드를 두 번 붙여 넣어도 카드는 한 번만 그립니다.
  // 데이터가 바뀌면 다시 그립니다: 페이지의 스크립트가 카드 안 JSON(<script type="application/json">)의 글자를 바꾸면 그 카드만
  // 새 요소로 갈아 끼워 새로 그립니다(필터 · 기간 컨트롤과 잇는 길). 갈아 끼운 카드에는 data-vzb-live가 붙어, 움직이는 카드는
  // 처음부터 다시 재생하지 않습니다. 앞 카드가 장면을 맡겨 두었으면(keep) 그 장면에서 새 값으로 옮겨 가고, 아니면 마지막 장면을 바로 그립니다.
  // 예전 요소가 페이지에서 빠지므로 그 카드의 움직임도 함께 끝납니다.
  // 대비 최소선: 페이지가 바꾼 색으로 글자가 바탕과 4.5:1, 포커스 링이 바탕과 3:1에 못 미치면 콘솔에 한 줄로 알립니다.
  // 그림은 페이지 값 그대로 둡니다(브랜드 색을 카드가 고치지 않습니다). 색은 1×1 캔버스에 칠해 읽으므로 color-mix · oklch도 읽힙니다.
  const CHECKS = [['ink', 'card', 4.5], ['ink-2', 'card', 4.5], ['muted', 'card', 4.5], ['good-ink', 'card', 4.5], ['bad-ink', 'card', 4.5],
    ['alert-ink', 'card', 4.5], ['ink-2', 'track', 4.5], ['card', 'ink', 4.5], ['tip-ink', 'tip', 4.5], ['focus', 'card', 3]];
  // 같은 알림은 페이지에서 한 번만: 카드마다 core.js가 따로 들어 있으므로, 공유 기호(Symbol.for)로 document에 한 벌만 둡니다.
  const WARNED = Symbol.for('vizney-blocks.contrast');
  function checkContrast(card) {
    let ctx;
    try { ctx = document.createElement('canvas').getContext('2d', { willReadFrequently: true }); } catch (e) { return; }
    if (!ctx) return;
    const probe = el('i');
    probe.style.cssText = 'position:absolute;visibility:hidden';
    card.append(probe);
    const read = (name) => {
      probe.style.color = name === 'focus' ? 'var(--vzb-focus, var(--vzb-primary))' : `var(--vzb-${name})`;
      return getComputedStyle(probe).color;
    };
    const lum = (bg, fg) => {
      ctx.clearRect(0, 0, 1, 1);
      ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, 1, 1);
      ctx.fillStyle = bg; ctx.fillRect(0, 0, 1, 1);
      if (fg) { ctx.fillStyle = fg; ctx.fillRect(0, 0, 1, 1); }
      const [r, g, b] = ctx.getImageData(0, 0, 1, 1).data;
      const c = (v) => { v /= 255; return v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4; };
      return 0.2126 * c(r) + 0.7152 * c(g) + 0.0722 * c(b);
    };
    const low = [];
    CHECKS.forEach(([fg, bg, need]) => {
      const B = read(bg), a = lum(B), b = lum(B, read(fg));
      const ratio = (Math.max(a, b) + 0.05) / (Math.min(a, b) + 0.05);
      if (ratio < need - 0.005) low.push(`${fg === 'focus' ? '--vzb-focus' : `--vzb-${fg}`} on --vzb-${bg} ${ratio.toFixed(2)}:1 (needs ${need}:1)`);
    });
    probe.remove();
    const msg = low.join('; ');
    const warned = document[WARNED] || (document[WARNED] = new Set());
    if (!low.length || warned.has(msg)) return;
    warned.add(msg);
    console.warn(`Vizney Blocks: low contrast with the page's --vzb-* colors: ${msg}`);
  }
  function mount(type, draw) {
    const watch = (card, src) => {
      if (!src || !window.MutationObserver) return;
      const mo = new MutationObserver(() => {
        mo.disconnect();
        if (!card.isConnected) return;
        const next = card.cloneNode(false);
        next.removeAttribute('data-vzb-ready');
        next.setAttribute('data-vzb-live', '');
        next.append(src);
        const scene = SCENE.get(card);
        if (scene) { try { FROM.set(next, scene()); } catch (e) { /* 장면을 못 넘기면 마지막 장면을 바로 그립니다 */ } }
        card.replaceWith(next);
        run();
      });
      mo.observe(src, { childList: true, characterData: true, subtree: true });
    };
    const run = () => {
      document.querySelectorAll(`.vzb-card[data-vzb="${type}"]:not([data-vzb-ready])`).forEach((card) => {
        card.setAttribute('data-vzb-ready', '');
        const src = card.querySelector('script[type="application/json"]');
        const raw = src ? src.textContent : '';
        watch(card, src);
        let d;
        try { d = JSON.parse(raw); } catch (e) {
          card.append(el('p', 'vzb-error', tx(/"lang"\s*:\s*"en"/.test(raw) ? { lang: 'en' } : null, '카드 데이터(JSON)를 읽지 못했습니다: %s', e.message)));
          return;
        }
        card.setAttribute('role', 'group');
        if (d.title) card.setAttribute('aria-label', d.title);
        // 어두운 바탕: JSON에 "theme": "dark"를 적으면 Vizney 검정 바탕에 노랑을 첫째 색으로 그립니다(tokens.css의 .vzb-dark).
        card.classList.toggle('vzb-dark', d.theme === 'dark');
        tune(card);
        checkContrast(card);
        try { draw(card, d, format(d)); } catch (e) {
          card.append(el('p', 'vzb-error', tx(d, '카드를 그리지 못했습니다: %s', e.message)));
        }
      });
    };
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
    else run();
  }
  // 서식 길이: 페이지가 :root에 적은 --vzb-<name>(0 · 4px · 0.5rem 등)을 px 숫자로 읽습니다. 비워 두었으면(initial) fallback을 돌려줍니다.
  // 스크립트가 그리는 SVG(막대 끝, 말풍선, 칸)가 CSS와 같은 모서리 변수를 읽게 하는 길입니다. 길이는 카드에 잠깐 넣은 요소의 폭으로 재서, rem · calc도 읽힙니다.
  function len(card, name, fallback) {
    if (!getComputedStyle(card).getPropertyValue(`--vzb-${name}`).trim()) return fallback;
    const probe = el('i');
    probe.style.cssText = `position:absolute;visibility:hidden;width:var(--vzb-${name})`;
    card.append(probe);
    const v = parseFloat(getComputedStyle(probe).width);
    probe.remove();
    return isFinite(v) && v >= 0 ? v : fallback;
  }
  // 크기가 바뀔 때마다 그래픽을 다시 그립니다. 차트는 카드에 남은 높이를 채웁니다.
  function onSize(node, fn) {
    let lw = 0, lh = 0;
    new ResizeObserver((entries) => {
      const w = Math.floor(entries[0].contentRect.width), h = Math.floor(entries[0].contentRect.height);
      if (w > 0 && h > 0 && (w !== lw || h !== lh)) { lw = w; lh = h; fn(w, h); }
    }).observe(node);
  }

  /* 툴팁과 조작: 마우스는 올리면 보이고 벗어나면 숨깁니다. 터치는 탭하면 보이고 다른 곳을 탭하면 닫힙니다.
     키보드는 그래프에 포커스한 뒤 화살표로 옮깁니다(가로 그래프는 좌우, 목록은 위아래). */
  let tip, tipDark = false;
  // value는 글자나 요소입니다. side가 참이면 점 위가 아니라 세로선 옆, 그래프 윗선에 맞춰 둡니다(선 · 막대 차트).
  function showTip(x, y, label, value, side) {
    if (!tip) {
      tip = el('div', 'vzb-tip');
      tip.setAttribute('aria-hidden', 'true');
      document.body.append(tip);
    }
    tip.classList.toggle('vzb-dark', tipDark);
    tip.replaceChildren(el('span', 'vzb-tip-label', label), typeof value === 'string' ? el('span', 'vzb-tip-value', value) : value);
    tip.classList.add('is-on');
    const w = tip.offsetWidth, h = tip.offsetHeight, W = window.innerWidth, H = window.innerHeight;
    if (side) {
      // side가 'left'이면 짚은 자리의 왼쪽을 먼저, 그 밖에는 오른쪽을 먼저 씁니다. 자리가 모자라면 반대쪽으로 넘기고, 높이는 짚은 자리에 가운데를 맞춥니다.
      // 짚은 표시의 위(머리의 대표 수치나 레이블)를 덮지 않아야 하는 차트가 씁니다.
      const right = x + 14 + w <= W - 8, left = x - w - 14 >= 8;
      tip.style.left = `${(side === 'left' ? left || !right : !right) ? Math.max(8, x - w - 14) : x + 14}px`;
      tip.style.top = `${Math.min(Math.max(8, y - h / 2), H - h - 8)}px`;
      return;
    }
    tip.style.left = `${Math.min(Math.max(8, x - w / 2), W - w - 8)}px`;
    tip.style.top = `${y - h - 12 < 8 ? y + 16 : y - h - 12}px`;
  }
  function hideTip() { if (tip) tip.classList.remove('is-on'); }
  // 툴팁 내용: { value, tone, note } 또는 { rows: [{ tone, value, name }], note }, { value, flow: [{ value, name }], note }. 값이 먼저, 이름은 뒤에 옅게 적습니다.
  // 증감은 차트와 같은 뜻의 색으로 적습니다: tone이 'good'이면 초록, 'bad'면 빨강. note의 줄도 { text, tone }으로 색을 줄 수 있습니다.
  function tipBody(spec, label) {
    const body = el('span', 'vzb-tip-body');
    let text = label;
    // value는 짚은 값(big이면 18px로 크게), rows는 그 값을 이루는 항목들(견본 · 값 · 이름)입니다. 둘을 함께 주면 값 아래에 항목을 적습니다.
    // 견본은 tone(색 이름)으로 칠하고, box면 네모로 그립니다. flow는 앞에서 뒤로 넘어가는 값들을 화살표(→)로 한 줄에 잇습니다(예: 앞 단계 → 뒤 단계).
    if (spec.value != null) {
      body.append(el('span', `vzb-tip-value${spec.big ? ' is-big' : ''}${spec.tone ? ` is-${spec.tone}` : ''}`, spec.value));
      text += ` ${spec.value}`;
    }
    if (spec.rows) {
      const grid = el('span', 'vzb-tip-rows');
      spec.rows.forEach((r) => {
        const row = el('span', 'vzb-tip-row'), key = el('i', `vzb-tip-key${r.tone ? ` vzb-${r.tone}` : ''}${r.box ? ' is-box' : ''}`);
        row.append(key, el('b', null, r.value), el('span', null, r.name));
        grid.append(row);
        text += `, ${r.name} ${r.value}`;
      });
      body.append(grid);
    }
    if (spec.flow) {
      const line = el('span', 'vzb-tip-flow');
      spec.flow.forEach((r, i) => {
        if (i) line.append(el('span', 'vzb-tip-arrow', '→'));
        const item = el('span', 'vzb-tip-flow-item');
        item.append(el('b', null, r.value), el('span', null, r.name));
        line.append(item);
      });
      body.append(line);
      text += `, ${spec.flow.map((r) => `${r.name} ${r.value}`).join(' → ')}`;
    }
    // note는 한 줄 글 또는 줄마다 나눈 배열입니다.
    (Array.isArray(spec.note) ? spec.note : [spec.note]).filter(Boolean).forEach((line) => {
      const o = typeof line === 'object' ? line : { text: line };
      body.append(el('span', `vzb-tip-note${o.tone ? ` is-${o.tone}` : ''}`, o.text));
      text += `, ${o.text}`;
    });
    return { node: body, text };
  }
  // 범례: [{ name, tone, box(면), dot(점), dim(옅게), tick(세로 눈금) }]. 선은 짧은 선, 막대와 면은 네모, 점은 동그라미로 표시합니다.
  function legendEl(items) {
    const ul = el('ul', 'vzb-legend');
    items.forEach((it) => {
      const li = el('li');
      li.append(el('i', `vzb-key vzb-${it.tone}${it.box ? ' is-box' : ''}${it.dot ? ' is-dot' : ''}${it.dim ? ' is-dim' : ''}${it.tick ? ' is-tick' : ''}`), it.name);
      ul.append(li);
    });
    return ul;
  }
  // 글자 폭(12px 기준, size를 주면 그 크기)과 SVG 글자
  let meter;
  function textW(s, weight, family, size) {
    meter = meter || document.createElement('canvas').getContext('2d');
    meter.font = `${weight} ${size || 12}px ${family}`;
    return meter.measureText(s).width;
  }
  function svgText(x, y, s, cls, anchor) {
    const t = svgEl('text', { x, y, class: cls, 'text-anchor': anchor || 'start' });
    t.textContent = s;
    return t;
  }
  // 차이를 부호와 함께 적습니다. 비율 지표는 %p입니다.
  function signed(f, diff) {
    const s = diff > 0 ? '+' : diff < 0 ? '-' : '';   // 줄어든 값은 하이픈(-)으로 적습니다. 복사해 표 계산기에 붙여도 음수로 읽힙니다
    return f.rel ? `${s}${show(f, Math.abs(diff))}` : `${s}${pp(f, diff)}%p`;
  }
  // pick(clientX, clientY) → 순번, point(순번) → { x, y, label, value, text, side }(화면 좌표), clear() → 강조 해제
  // n은 개수나 개수를 돌려주는 함수입니다. vertical이면 위아래 화살표로 옮깁니다. 돌려주는 { show(순번), at() }로 키를 더 붙입니다.
  // 아무것도 고르지 않은 채 화살표를 누르면 기간 차트는 마지막(최근)부터, vertical이나 first면 첫 순번부터 짚습니다.
  function interact(plot, n, pick, point, clear, vertical, first) {
    let idx = -1, touched = false;
    const count = () => (typeof n === 'function' ? n() : n);
    const live = el('span', 'vzb-sr');
    live.setAttribute('aria-live', 'polite');
    plot.after(live);
    const go = (i) => {
      if (i < 0 || i >= count()) return;
      idx = i;
      const p = point(i);
      tipDark = !!plot.closest('.vzb-dark');
      showTip(p.x, p.y, p.label, p.value, p.side);
      live.textContent = p.text || `${p.label} ${p.value}`;
    };
    const end = () => {
      if (idx < 0) return;
      idx = -1;
      clear();
      hideTip();
    };
    const [back, next] = vertical ? ['ArrowUp', 'ArrowDown'] : ['ArrowLeft', 'ArrowRight'];
    // pick이 -1이면(조각 밖) 강조를 풉니다.
    const at = (e) => { const i = pick(e.clientX, e.clientY); if (i < 0) end(); else go(i); };
    plot.addEventListener('pointermove', (e) => { if (e.pointerType !== 'touch' || touched) at(e); });
    plot.addEventListener('pointerdown', (e) => { if (e.pointerType === 'touch') { touched = true; at(e); } });
    plot.addEventListener('pointerleave', (e) => { if (e.pointerType !== 'touch') end(); });
    // 문서와 창에 건 듣기는 그래프가 페이지에서 빠지면(카드를 다시 그리면) 스스로 풉니다.
    const away = (e) => { if (!plot.isConnected) document.removeEventListener('pointerdown', away, true); else if (touched && !plot.contains(e.target)) { touched = false; end(); } };
    document.addEventListener('pointerdown', away, true);
    plot.addEventListener('keydown', (e) => {
      if (e.key !== back && e.key !== next) return;
      e.preventDefault();
      const last = count() - 1;
      go(idx < 0 ? (vertical || first ? 0 : last) : Math.max(0, Math.min(last, idx + (e.key === next ? 1 : -1))));
    });
    plot.addEventListener('blur', end);
    const scrolled = () => { if (!plot.isConnected) window.removeEventListener('scroll', scrolled); else end(); };
    window.addEventListener('scroll', scrolled, { passive: true });
    return { show: go, at: () => idx, end };
  }

  // Bar Overflow: 기간별(보통 월별) 값을 세로 막대로 세우고, 모든 기간에 같은 한도(limit, 예: 월 예산)를 가로선 하나로 긋습니다.

  // 같은 이름의 기간이 같은 막대입니다. 새로 생긴 기간은 바닥에서 자라고, 빠진 기간은 흐려지며 사라집니다.

  // 막대가 내려갔다가 다시 올라가 통통 튀어 보이기 때문입니다.
  // 옮겨 가는 동안 넘친 양과 아래 줄의 숫자는 화면의 막대를 그대로 세어 함께 바뀌고, 이번에 새로 한도를 넘은 막대만 다 옮겨 간 뒤 한 번 깜빡입니다.
  mount('bar-overflow', (card, d, f) => {

    // labels와 values는 같은 길이여야 합니다. 값이 null인 기간은 자리를 남기고 막대를 그리지 않습니다(기간을 빼면 시간 축이 어긋나기 때문입니다).
    const labels = Array.isArray(d.labels) ? d.labels.map(String) : [], raw = Array.isArray(d.values) ? d.values : [];

    items.forEach((it) => { it.over = it.v == null ? 0 : Math.max(0, it.v - lim); });

    const sumOf = (n, total) => (!n ? tx(d, '모두 %s 이내', inText)

    head(card, d, f, null);
    const foot = el('div', 'vzb-foot vzb-fill vzb-bb');

    // 한도선의 이름과 값은 선의 오른쪽 끝, 막대 칸 밖에 늘 같은 자리로 붙입니다(이름은 선 위, 값은 선 아래). 막대 사이에서 빈자리를 찾아 옮겨 다니지 않게

    const limLine = svgEl('line', { class: 'vzb-bb-line' }), limName = svgText(0, 0, label, 'vzb-bb-lim', 'start'), limVal = svgText(0, 0, limShow, 'vzb-bb-limv', 'start');

    // 세로 막대: round가 참이면 위쪽 끝만 둥글게, 아니면 각지게(한도선에 닿는 끝). 모서리는 --vzb-radius-mark(비우면 4px)이고 layout에서 읽습니다.

    // 처음의 뜸(Lead) · 차례의 전체 시간(Seq) · 되풀이 전의 쉼(Rest)과, 막대가 값으로 가는 움직임(Move: 넘치지 않음)은 움직임 값에서 읽습니다.
    const M = motion(card), MOVE = M.curve('move');

    let W = 0, H = 0, PW = 0, y0 = 0, s = 1, bw = 0, slot = 1, yl = 0, t = 0, FULL = 0, TOTAL = 0;
    // 옮겨 가기(데이터가 바뀌어 다시 그릴 때): 막대의 높이(sh, px)와 한도선의 높이(sly, px), 막대의 자리(sx)와 굵기(sw)를 스프링(Move)으로 새 자리에 보냅니다.

    let from = before(card), moving = false, flash = BLINK, sly = null, slv = null, sw = null, gone = [];
    function layout(w, h) {

      svg.setAttribute('height', h);

      y0 = h - NAMEH;
      s = Math.max(1, y0 - 4) / top;
      // 오른쪽 자리: 한도선 이름(12px)과 값(14px 굵게) 가운데 넓은 쪽에 선과의 틈 6px. 막대는 그 왼쪽(PW)에 둡니다.
      const gut = Math.ceil(Math.max(textW(label, 500, fam, 12), textW(limShow, 700, fam, 14))) + 8;

      slot = PW / N;
      bw = Math.max(4, Math.min(30, slot * 0.62));

      base.setAttribute('x2', fx(PW));
      base.setAttribute('y1', y0 + 0.5);
      base.setAttribute('y2', y0 + 0.5);

      limLine.setAttribute('x2', fx(PW + 2));

      const nameW = Math.max(...items.map((it) => textW(it.name, 400, fam, 12))) + 6;

      // 자리를 못 잡은 막대만 적지 않습니다(그 막대는 툴팁이 넘친 양을 적고, 합은 요약 줄에 남습니다).
      // 글자는 제 칸 밖으로 나가도 되지만 그림 밖으로는 나가지 않게 안쪽으로 당깁니다.
      const spot = (it, size) => {
        const lw = textW(it.ex.textContent, 700, fam, size) + 4, cx = slot * (it.k + 0.5);
        it.exW = lw;
        it.exX = Math.max(lw / 2, Math.min(PW - lw / 2, cx));
      };
      const clear = (list) => list.every((it, k) => !k || it.exX - it.exW / 2 >= list[k - 1].exX + list[k - 1].exW / 2 + 4);
      let exSize = 14;
      spill.forEach((it) => spot(it, 14));
      if (!clear(spill)) { exSize = 12; spill.forEach((it) => spot(it, 12)); }
      const shown = new Set(spill);
      if (!clear(spill)) {
        shown.clear();
        [...spill].sort((a, b) => b.over - a.over).forEach((it) => {
          const near = [...shown, it].sort((a, b) => a.k - b.k);
          if (clear(near)) shown.add(it);
        });
      }
      items.forEach((it) => {
        it.cx = slot * (it.k + 0.5);
        it.x = it.cx - bw / 2;
        it.yv = y0 - (it.v || 0) * s;
        it.nm.setAttribute('y', fx(y0 + 15));
        it.nm.setAttribute('opacity', (N - 1 - it.k) % every === 0 ? 1 : 0);
        it.exOn = shown.has(it);
        it.exDx = it.exOn ? it.exX - it.cx : 0;   // 넘친 양의 글자가 막대 가운데에서 비켜난 만큼
        it.ex.style.fontSize = exSize === 12 ? '12px' : '';
        it.start = T0 + it.k * STAG;
      });
      // 한도선의 이름(위)과 값(아래)은 선 끝에서 6px 띄워 선을 가운데로 위아래에 둡니다(높이는 render가 한도선과 함께 잡습니다).
      limName.setAttribute('x', fx(PW + 8));
      limVal.setAttribute('x', fx(PW + 8));
      FULL = T0 + (N - 1) * STAG + DUR;
      TOTAL = FULL + (spill.length ? WAIT + BLINK * TIMES : 240);
      if (from) { begin(from); from = null; kick(); }
      render();
    }
    // 앞 장면에서 시작합니다. 같은 이름의 막대는 그 자리와 높이에서, 새 막대는 제자리의 바닥에서 출발하고, 빠진 막대는 그 자리에 남겨 흐려지게 합니다.
    function begin(prev) {
      if (!prev || !prev.inView || !Array.isArray(prev.items)) return;
      const old = new Map(prev.items.map((p) => [p.key, p]));
      items.forEach((it) => {
        const p = it.v == null ? null : old.get(it.key);
        if (p) old.delete(it.key);
        it.sh = { x: p ? p.h : 0, v: p ? p.hv : 0 };
        it.sx = { x: p ? p.x : it.cx / PW, v: p ? p.xv : 0 };
        it.exA = p ? p.exA : 0;
        if (it.over > 0 && (!p || !p.over || p.fresh)) fresh.add(it);
      });
      const had = prev.lpx > 0;
      sly = { x: had ? prev.lpx : lim * s, v: had ? prev.lpxV || 0 : 0 };
      slv = { x: had ? prev.lim : lim, v: had ? prev.limV || 0 : 0 };
      sw = { x: prev.bw > 0 ? prev.bw : bw, v: 0 };
      gone = [...old.values()].map((p) => {
        const g = { h: p.h, x: p.x, a: p.a == null ? 1 : p.a, g: svgEl('g', { class: 'vzb-bb-col' }), fill: svgEl('path', { class: 'vzb-bb-fill' }), red: svgEl('path', { class: 'vzb-bb-over' }), nm: svgText(0, 0, p.name, 'vzb-bb-name', 'middle') };
        g.nm.setAttribute('opacity', p.nameOn ? 1 : 0);
        g.g.append(g.fill, g.red, g.nm);
        goneG.append(g.g);
        return g;
      });
      moving = true;
      flash = BLINK;
      t = 1e9;
    }
    function settle() {
      moving = false;
      gone.forEach((g) => g.g.remove());
      gone = [];
      flash = fresh.size ? 0 : BLINK;
    }
    // 스프링을 dt만큼 옮기고, 모두 제자리에 닿았으면 끝냅니다.
    function step(dt) {
      let busy = false;
      const go = (sp, to, span) => {
        springTo(sp, to, MK, MC, dt);
        if (Math.abs(sp.x - to) > span * 0.0006 || Math.abs(sp.v) > span * 0.01) busy = true; else { sp.x = to; sp.v = 0; }
      };
      const hh = Math.max(1, y0);
      go(sly, lim * s, hh);
      go(slv, lim, top);
      go(sw, bw, 30);
      items.forEach((it) => {
        if (it.v == null) return;
        go(it.sh, it.v * s, hh);
        go(it.sx, it.cx / PW, 1);
      });
      gone.forEach((g) => { g.a = Math.max(0, g.a - dt / FADE); if (g.a > 0) busy = true; });
      const fadeBy = dt / FADE;
      items.forEach((it) => {
        const to = it.exOn && it.sh && it.sh.x > sly.x ? 1 : 0;
        if (it.exA !== to) { it.exA = to > it.exA ? Math.min(1, it.exA + fadeBy) : Math.max(0, it.exA - fadeBy); busy = true; }
      });
      if (!busy) settle();
    }
    function render() {
      if (!W) return;
      const mv = moving;
      const ylC = mv ? y0 - sly.x : yl, bwC = mv ? sw.x : bw;
      const unit = mv && sly.x > 0.5 ? slv.x / sly.x : 0;   // 옮겨 가는 동안 1px이 나타내는 값(한도선의 높이가 한도의 값)
      limLine.setAttribute('y1', fx(ylC));
      limLine.setAttribute('y2', fx(ylC));
      limName.setAttribute('y', fx(ylC - 4));
      limVal.setAttribute('y', fx(ylC + 15));
      // 깜빡임: 다 찬 뒤 넘친 구간만 옅어졌다 돌아오기를 되풀이하고, 진한 상태로 끝납니다. 멈춰서 값을 읽는 동안은 진하게 둡니다.
      // 옮겨 간 뒤에는 이번에 새로 한도를 넘은 막대만 한 번 깜빡입니다.
      const b = (t - FULL - WAIT) / BLINK, wave = (x) => 1 - 0.84 * (0.5 - 0.5 * Math.cos(2 * Math.PI * x));
      const glow = b <= 0 || b >= TIMES || holds.size ? 1 : wave(b);
      const glow2 = mv || flash >= BLINK || holds.size ? 1 : wave(flash / BLINK);
      const alert = t >= FULL + WAIT;
      let cnt = 0, tot = 0;
      // 막대 하나(yv: 막대 끝의 자리): 한도에 닿은 막대는 한도선 아래에서 각지게 끊고, 넘친 만큼은 선 위에서 빨강으로 잇습니다.
      const bar = (o, cx, yv, hit) => {
        const x = cx - bwC / 2, cut = hit && yv <= ylC + GAP;
        o.fill.setAttribute('d', col(x, cut ? ylC + GAP : yv, bwC, y0, !cut));
        o.red.setAttribute('d', hit && yv < ylC - GAP ? col(x, yv, bwC, ylC - GAP, true) : '');
      };
      gone.forEach((g) => {
        bar(g, g.x * PW, y0 - g.h, g.h >= sly.x && sly.x > 0);
        g.nm.setAttribute('x', fx(g.x * PW));
        g.nm.setAttribute('y', fx(y0 + 15));
        g.g.setAttribute('opacity', fx(g.a));
      });
      items.forEach((it) => {
        const cx = mv ? it.sx.x * PW : it.cx;
        it.nm.setAttribute('x', fx(cx));
        if (it.v == null) { it.fill.setAttribute('d', ''); it.red.setAttribute('d', ''); it.ex.setAttribute('opacity', 0); return; }
        const h = mv ? Math.max(0, it.sh.x) : it.v * clamp(MOVE.at(t - it.start)) * s, yv = y0 - h;
        bar(it, cx, yv, mv ? h >= sly.x : it.v >= lim);
        const overNow = mv ? Math.max(0, (h - sly.x) * unit) : it.over;
        if (mv && overNow > 0) { cnt++; tot += overNow; }
        it.red.setAttribute('opacity', fx(fresh.has(it) && !mv ? Math.min(glow, glow2) : glow));
        const exText = signed(f, overNow);
        if (it.exT !== exText) { it.exT = exText; it.ex.textContent = exText; }
        it.ex.setAttribute('x', fx(cx + (it.exDx || 0)));
        it.ex.setAttribute('y', fx(yv - 5));
        it.ex.setAttribute('opacity', mv ? fx(it.exA) : it.exOn ? fx(clamp((t - it.start - DUR) / 200)) : 0);
        it.g.classList.toggle('is-over', mv ? overNow > 0 : it.over > 0 && alert);
      });
      if (mv) setSum(cnt, tot); else setSum(spill.length, sum);
      sumEl.style.opacity = mv ? 1 : fx(clamp((t - (spill.length ? FULL + WAIT : FULL)) / 240));
    }
    // 지금 화면의 장면: 데이터가 바뀌어 카드를 갈아 끼울 때 새 카드에 넘깁니다. 아직 그리기 전이면 받은 장면을 그대로 넘깁니다.
    keep(card, () => {
      if (!W) return from;
      const mv = moving;
      return {
        inView, lpx: mv ? sly.x : lim * s, lpxV: mv ? sly.v : 0, lim: mv ? slv.x : lim, limV: mv ? slv.v : 0, bw: mv ? sw.x : bw,
        items: items.filter((it) => it.v != null).map((it) => ({
          key: it.key, name: it.name, nameOn: it.nm.getAttribute('opacity') === '1',
          h: mv ? Math.max(0, it.sh.x) : it.v * clamp(MOVE.at(t - it.start)) * s, hv: mv ? it.sh.v : 0,
          x: mv ? it.sx.x : it.cx / PW, xv: mv ? it.sx.v : 0,
          exA: mv ? it.exA : it.exOn ? clamp((t - it.start - DUR) / 200) : 0,
          over: it.over > 0, fresh: fresh.has(it) && (mv || flash < BLINK),
        })).concat(gone.map((g) => ({ key: `\n${g.x}`, name: g.nm.textContent, nameOn: g.nm.getAttribute('opacity') === '1', h: g.h, hv: 0, x: g.x, xv: 0, exA: 0, over: false, fresh: false, a: g.a }))),
      };
    });

    // 재생: 화면에 보이고(inView) 마우스 · 포커스로 잠시 멈추지 않았을 때(holds)만 움직입니다. 카드가 페이지에서 빠지면 멈춥니다.
    const live = card.hasAttribute('data-vzb-live');   // 데이터가 바뀌어 다시 그린 카드: 처음부터 재생하지 않습니다
    const loop = !!card.closest('[data-vzb-loop]');
    const holds = new Set();
    let playing = false, seen = false, inView = false, rest = 0, last = 0, raf = 0;
    // 옮겨 가기와 그 뒤의 한 번 깜빡임은 데이터가 바뀐 데 대한 답이라, 마우스가 올라가 있어도 끝까지 갑니다.
    const on = () => card.isConnected && inView && (moving || flash < BLINK || (playing && !holds.size));
    function tick(stamp) {
      raf = 0;
      if (!on()) { last = 0; return; }
      const dt = last ? Math.min(100, stamp - last) : 16;
      last = stamp;
      if (moving) step(dt);
      else if (flash < BLINK) { flash += dt; if (flash >= BLINK) fresh.clear(); }
      else if (rest > 0) { rest -= dt; if (rest <= 0) t = 0; }
      else {
        t = Math.min(TOTAL, t + dt);
        if (t >= TOTAL) { if (loop) rest = M.ms('rest'); else playing = false; }
      }
      render();
      if (on()) raf = requestAnimationFrame(tick);
    }
    function kick() { if (!raf && on()) raf = requestAnimationFrame(tick); }
    function play() { if (moving) settle(); flash = BLINK; fresh.clear(); t = 0; rest = 0; last = 0; playing = true; render(); kick(); }
    btn.addEventListener('click', play);
    plot.addEventListener('pointerenter', (e) => { if (e.pointerType === 'mouse') { holds.add('hover'); render(); } });
    plot.addEventListener('pointerleave', () => { holds.delete('hover'); kick(); });
    plot.addEventListener('focus', () => { if (plot.matches(':focus-visible')) { holds.add('focus'); render(); } });   // 키보드로 들어왔을 때만 멈춥니다
    plot.addEventListener('blur', () => { holds.delete('focus'); kick(); });
    // 툴팁: 막대를 짚으면 값, 한도, 넘치거나 남은 양과 한도 대비 비율을 보여 주고 그 막대만 진하게 남깁니다.
    // 넘친 양의 글자는 숨기지 않고 그대로 두며(숫자가 사라지면 오류처럼 보이므로), 그 글자가 보이는 막대는 툴팁에 넘친 양을 되풀이하지 않습니다.
    interact(plot, N,
      (cx) => Math.max(0, Math.min(N - 1, Math.floor((cx - plot.getBoundingClientRect().left) / slot))),
      (k) => {
        const it = items[k], r = plot.getBoundingClientRect();
        svg.classList.add('is-hover');
        items.forEach((x) => x.g.classList.toggle('is-on', x === it));
        const note = [`${label} ${limShow}`];
        if (it.v != null) {
          const gap = show(f, Math.abs(it.v - lim)), share = lim > 0 ? num((it.v / lim) * 100, 0) : null;
          // 한도를 넘친 줄은 막대의 넘친 부분과 같은 빨강으로 적습니다.
          if (share == null) note.push({ text: tx(d, '%s 초과|한도', gap), tone: it.v > lim ? 'alert' : null });
          else if (it.v > lim && it.exOn) note.push({ text: tx(d, '%1의 %2', inText, `${share}%`), tone: 'alert' });
          else note.push(it.v > lim ? { text: tx(d, '%1 초과 · %2의 %3%', gap, inText, share), tone: 'alert' } : it.v < lim ? tx(d, '%1 남음 · %2의 %3%', gap, inText, share) : tx(d, '%1의 %2', inText, `${share}%`));
        }
        const tb = tipBody({ value: it.v == null ? tx(d, '데이터 없음') : show(f, it.v), note }, it.name);
        return { x: r.left + it.cx, y: r.top + Math.min(it.yv, yl) - 6, label: it.name, value: tb.node, text: tb.text };
      },
      () => { svg.classList.remove('is-hover'); items.forEach((x) => x.g.classList.remove('is-on')); },
      false, true);

    if (live) t = 1e9;   // 마지막 장면에서 시작합니다(앞 장면을 받았으면 layout이 거기서 옮겨 가기를 시작합니다)
    if (from) inView = !!from.inView;   // 앞 카드가 화면에 보이고 있었으면 바로 움직입니다
    onSize(plot, layout);
    if (document.fonts && document.fonts.ready) document.fonts.ready.then(() => { if (W) layout(W, H); });
    // 화면에 처음 보일 때 움직입니다. 화면 밖에 있는 동안은 멈춰 둡니다.
    const start = () => { if (!seen) { seen = true; if (!live) play(); } };
    if ('IntersectionObserver' in window) {
      const io = new IntersectionObserver((es) => {
        if (!card.isConnected) { io.disconnect(); return; }
        inView = es[es.length - 1].isIntersecting;
        if (!inView && (moving || flash < BLINK)) { if (moving) settle(); flash = BLINK; fresh.clear(); render(); }   // 화면 밖에서는 옮겨 가기를 끝낸 장면으로 둡니다
        if (inView) start();
        kick();
      }, { threshold: 0.3 });
      io.observe(card);
    } else { inView = true; start(); }
  });
})();
</script>
