<script>
(() => {
  'use strict';
  /* 공통 도구: 숫자 표기, 증감, 카드 머리, 그리기 시작 */
  const SVGNS = 'http://www.w3.org/2000/svg';
  function el(tag, cls, text) {
    const n = document.createElement(tag);
    if (cls) n.className = cls;
    if (text != null) n.textContent = text;
    return n;
  }
  function svgEl(tag, attrs) {
    const n = document.createElementNS(SVGNS, tag);
    for (const k in attrs) n.setAttribute(k, attrs[k]);
    return n;
  }
  const nfs = {};
  function num(v, d) {
    nfs[d] = nfs[d] || new Intl.NumberFormat('ko-KR', { minimumFractionDigits: d, maximumFractionDigits: d });
    return nfs[d].format(v);
  }
  function krw(v) {
    const a = Math.abs(v);
    if (a >= 1e12) return [num(v / 1e12, 1), '조'];
    if (a >= 1e8) return [num(v / 1e8, a >= 1e10 ? 0 : 1), '억'];
    if (a >= 1e4) return [num(v / 1e4, 0), '만'];
    return [num(v, 0), '원'];
  }
  // 달러 금액: 원래 값을 받아 K · M · B로 줄이고 소수 한 자리로 맞춥니다($745.6K). 접미사는 단위처럼 작게 씁니다.
  function usd(v, d) {
    const a = Math.abs(v), s = v < 0 ? '−' : '';
    const [div, u] = a >= 1e9 ? [1e9, 'B'] : a >= 1e6 ? [1e6, 'M'] : a >= 1e3 ? [1e3, 'K'] : [1, ''];
    return [`${s}$${num(a / div, d != null ? d : div > 1 ? 1 : 0)}`, u];
  }
  // format: eok(억 단위 값) · krw(원 단위 값) · usd(달러 값) · count(unit으로 단위) · pct(증감을 %p로) · days
  function format(d) {
    const dec = (v, fallback) => (d.decimals != null ? d.decimals : fallback);
    switch (d.format) {
      case 'eok': return { parts: (v) => [num(v, dec(v, Number.isInteger(v) || Math.abs(v) >= 100 ? 0 : 1)), '억'], rel: true };
      case 'krw': return { parts: krw, rel: true };
      case 'usd': return { parts: (v) => usd(v, d.decimals), rel: true, usd: true };
      case 'pct': return { parts: (v) => [num(v, dec(v, 1)), '%'], rel: false };
      case 'days': return { parts: (v) => [num(v, dec(v, 1)), '일'], rel: true };
      default: return { parts: (v) => [num(v, dec(v, Number.isInteger(v) ? 0 : 1)), d.unit || ''], rel: true };
    }
  }
  const show = (f, v) => f.parts(v).join('');
  // 그릴 수 있는 숫자인지(빈 값 · 빈 글자 · NaN 제외)
  const fin = (v) => v != null && v !== '' && isFinite(v);
  // 증감: 방향은 화살표로, 좋고 나쁨(tone)은 better(up · down · neutral)로 정합니다. 비율은 %p로 적습니다.
  function delta(f, value, base, better) {
    const diff = value - base;
    const dir = Math.abs(diff) < 1e-9 ? 0 : Math.sign(diff);
    const want = better === 'down' ? -1 : better === 'neutral' ? 0 : 1;
    const tone = dir === 0 || want === 0 ? 'flat' : dir === want ? 'good' : 'bad';
    let text;
    if (!f.rel) text = `${num(Math.abs(diff), 1)}%p`;
    else if (base) text = `${num(Math.abs(diff / base) * 100, 1)}%`;
    else text = show(f, Math.abs(diff));
    return { dir, tone, text };
  }
  function arrow(dir) {
    const s = svgEl('svg', { class: 'vzb-arrow', viewBox: '0 0 12 12', 'aria-hidden': 'true' });
    s.append(svgEl('path', { d: dir > 0 ? 'M3.5 8.5l5-5M4.5 3.5h4v4' : 'M3.5 3.5l5 5M8.5 4.5v4h-4' }));
    return s;
  }
  // 대표 수치(단위는 작게)
  function figure(f, value) {
    const n = el('div', 'vzb-num');
    const [text, unit] = f.parts(value);
    n.append(text);
    if (unit) n.append(el('span', 'vzb-unit', unit));
    return n;
  }
  // 증감 배지: 화살표는 오르내림, 색은 좋고 나쁨(better)입니다.
  function badge(f, value, base, better) {
    const dl = delta(f, value, base, better);
    const b = el('span', `vzb-badge vzb-${dl.tone}`);
    if (dl.dir) b.append(arrow(dl.dir), el('span', 'vzb-sr', dl.dir > 0 ? '증가' : '감소'));
    b.append(dl.text);
    return b;
  }
  // 머리: 제목 · 기간 부제(subtitle) · 대표 수치와 바로 뒤의 증감 배지. value가 null이면 제목만 둡니다.
  function head(card, d, f, value, cmp, cmpText) {
    const hd = el('div', 'vzb-hd');
    hd.append(el('div', 'vzb-title', d.title || ''));
    if (d.subtitle) hd.append(el('div', 'vzb-sub', d.subtitle));
    card.append(hd);
    if (value == null) return;
    const fig = el('div', 'vzb-fig');
    fig.append(figure(f, value));
    if (cmp && cmp.value != null) {
      const line = el('span', 'vzb-delta');
      line.append(badge(f, value, cmp.value, d.better), el('span', null, cmpText || `${cmp.label || '비교 기간'} 대비`));
      fig.append(line);
    }
    card.append(fig);
  }
  // 같은 종류의 카드를 모두 찾아 그립니다. 코드를 두 번 붙여 넣어도 카드는 한 번만 그립니다.
  function mount(type, draw) {
    const run = () => {
      document.querySelectorAll(`.vzb-card[data-vzb="${type}"]:not([data-vzb-ready])`).forEach((card) => {
        card.setAttribute('data-vzb-ready', '');
        const src = card.querySelector('script[type="application/json"]');
        let d;
        try { d = JSON.parse(src ? src.textContent : ''); } catch (e) {
          card.append(el('p', 'vzb-error', `카드 데이터(JSON)를 읽지 못했습니다: ${e.message}`));
          return;
        }
        card.setAttribute('role', 'group');
        if (d.title) card.setAttribute('aria-label', d.title);
        try { draw(card, d, format(d)); } catch (e) {
          card.append(el('p', 'vzb-error', `카드를 그리지 못했습니다: ${e.message}`));
        }
      });
    };
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
    else run();
  }
  // 크기가 바뀔 때마다 그래픽을 다시 그립니다. 차트는 카드에 남은 높이를 채웁니다.
  function onSize(node, fn) {
    let lw = 0, lh = 0;
    new ResizeObserver((entries) => {
      const w = Math.floor(entries[0].contentRect.width), h = Math.floor(entries[0].contentRect.height);
      if (w > 0 && h > 0 && (w !== lw || h !== lh)) { lw = w; lh = h; fn(w, h); }
    }).observe(node);
  }

  /* 툴팁과 조작: 마우스는 올리면 보이고 벗어나면 숨깁니다. 터치는 탭하면 보이고 다른 곳을 탭하면 닫힙니다.
     키보드는 그래프에 포커스한 뒤 화살표로 옮깁니다(가로 그래프는 좌우, 목록은 위아래). */
  let tip;
  // value는 글자나 요소입니다. side가 참이면 점 위가 아니라 세로선 옆, 그래프 윗선에 맞춰 둡니다(선 · 막대 차트).
  function showTip(x, y, label, value, side) {
    if (!tip) {
      tip = el('div', 'vzb-tip');
      tip.setAttribute('aria-hidden', 'true');
      document.body.append(tip);
    }
    tip.replaceChildren(el('span', 'vzb-tip-label', label), typeof value === 'string' ? el('span', 'vzb-tip-value', value) : value);
    tip.classList.add('is-on');
    const w = tip.offsetWidth, h = tip.offsetHeight, W = window.innerWidth, H = window.innerHeight;
    if (side) {
      // side가 'left'이면 기준선 왼쪽을 먼저, 그 밖에는 오른쪽을 먼저 씁니다. 자리가 모자라면 반대쪽으로 넘깁니다.
      const right = x + 14 + w <= W - 8, left = x - w - 14 >= 8;
      tip.style.left = `${(side === 'left' ? left || !right : !right) ? Math.max(8, x - w - 14) : x + 14}px`;
      tip.style.top = `${Math.min(Math.max(8, y), H - h - 8)}px`;
      return;
    }
    tip.style.left = `${Math.min(Math.max(8, x - w / 2), W - w - 8)}px`;
    tip.style.top = `${y - h - 12 < 8 ? y + 16 : y - h - 12}px`;
  }
  function hideTip() { if (tip) tip.classList.remove('is-on'); }
  // 툴팁 내용: { value, note } 또는 { rows: [{ tone, value, name }], note }. 값이 먼저, 이름은 뒤에 옅게 적습니다.
  function tipBody(spec, label) {
    const body = el('span', 'vzb-tip-body');
    let text = label;
    if (spec.rows) {
      const grid = el('span', 'vzb-tip-rows');
      spec.rows.forEach((r) => {
        const row = el('span', 'vzb-tip-row');
        row.append(el('i', `vzb-tip-key vzb-${r.tone}`), el('b', null, r.value), el('span', null, r.name));
        grid.append(row);
        text += `, ${r.name} ${r.value}`;
      });
      body.append(grid);
    } else {
      body.append(el('span', 'vzb-tip-value', spec.value));
      text += ` ${spec.value}`;
    }
    // note는 한 줄 글 또는 줄마다 나눈 배열입니다.
    (Array.isArray(spec.note) ? spec.note : [spec.note]).filter(Boolean).forEach((line) => {
      body.append(el('span', 'vzb-tip-note', line));
      text += `, ${line}`;
    });
    return { node: body, text };
  }
  // 범례: [{ name, tone, box(면), dot(점), dim(옅게), tick(세로 눈금) }]. 선은 짧은 선, 막대와 면은 네모, 점은 동그라미로 표시합니다.
  function legendEl(items) {
    const ul = el('ul', 'vzb-legend');
    items.forEach((it) => {
      const li = el('li');
      li.append(el('i', `vzb-key vzb-${it.tone}${it.box ? ' is-box' : ''}${it.dot ? ' is-dot' : ''}${it.dim ? ' is-dim' : ''}${it.tick ? ' is-tick' : ''}`), it.name);
      ul.append(li);
    });
    return ul;
  }
  // 글자 폭(12px 기준, size를 주면 그 크기)과 SVG 글자
  let meter;
  function textW(s, weight, family, size) {
    meter = meter || document.createElement('canvas').getContext('2d');
    meter.font = `${weight} ${size || 12}px ${family}`;
    return meter.measureText(s).width;
  }
  function svgText(x, y, s, cls, anchor) {
    const t = svgEl('text', { x, y, class: cls, 'text-anchor': anchor || 'start' });
    t.textContent = s;
    return t;
  }
  // 차이를 부호와 함께 적습니다. 비율 지표는 %p입니다.
  function signed(f, diff) {
    const s = diff > 0 ? '+' : diff < 0 ? '−' : '';
    return f.rel ? `${s}${show(f, Math.abs(diff))}` : `${s}${num(Math.abs(diff), 1)}%p`;
  }
  // pick(clientX, clientY) → 순번, point(순번) → { x, y, label, value, text, side }(화면 좌표), clear() → 강조 해제
  // n은 개수나 개수를 돌려주는 함수입니다. vertical이면 위아래 화살표로 옮깁니다. 돌려주는 { show(순번), at() }로 키를 더 붙입니다.
  // 아무것도 고르지 않은 채 화살표를 누르면 기간 차트는 마지막(최근)부터, vertical이나 first면 첫 순번부터 짚습니다.
  function interact(plot, n, pick, point, clear, vertical, first) {
    let idx = -1, touched = false;
    const count = () => (typeof n === 'function' ? n() : n);
    const live = el('span', 'vzb-sr');
    live.setAttribute('aria-live', 'polite');
    plot.after(live);
    const go = (i) => {
      if (i < 0 || i >= count()) return;
      idx = i;
      const p = point(i);
      showTip(p.x, p.y, p.label, p.value, p.side);
      live.textContent = p.text || `${p.label} ${p.value}`;
    };
    const end = () => {
      if (idx < 0) return;
      idx = -1;
      clear();
      hideTip();
    };
    const [back, next] = vertical ? ['ArrowUp', 'ArrowDown'] : ['ArrowLeft', 'ArrowRight'];
    // pick이 -1이면(조각 밖) 강조를 풉니다.
    const at = (e) => { const i = pick(e.clientX, e.clientY); if (i < 0) end(); else go(i); };
    plot.addEventListener('pointermove', (e) => { if (e.pointerType !== 'touch' || touched) at(e); });
    plot.addEventListener('pointerdown', (e) => { if (e.pointerType === 'touch') { touched = true; at(e); } });
    plot.addEventListener('pointerleave', (e) => { if (e.pointerType !== 'touch') end(); });
    document.addEventListener('pointerdown', (e) => { if (touched && !plot.contains(e.target)) { touched = false; end(); } }, true);
    plot.addEventListener('keydown', (e) => {
      if (e.key !== back && e.key !== next) return;
      e.preventDefault();
      const last = count() - 1;
      go(idx < 0 ? (vertical || first ? 0 : last) : Math.max(0, Math.min(last, idx + (e.key === next ? 1 : -1))));
    });
    plot.addEventListener('blur', end);
    window.addEventListener('scroll', end, { passive: true });
    return { show: go, at: () => idx, end };
  }

  /* 구성비 공통: 큰 세 항목만 주황 · 분홍빨강 · 노랑으로 칠하고 나머지는 회색 기타 하나로 묶습니다(KPI - Stacked Bar와 같은 규칙).
     비중 카드는 비중을 적고 금액은 툴팁에 둡니다. 조각이나 범례 줄에 마우스를 올리면 그 조각만 진하게 남깁니다. */

  // 큰 세 항목과 기타. sort가 false면 적은 순서대로 둡니다. members는 묶인 원래 항목입니다.
  function foldParts(items, sort) {
    const ok = (p) => p && p.value != null && +p.value > 0;
    let named = (items || []).filter((p) => ok(p) && !p.other);
    if (sort !== false) named = named.slice().sort((a, b) => b.value - a.value);
    const rest = named.slice(3).concat((items || []).filter((p) => ok(p) && p.other));
    const parts = named.slice(0, 3).map((p, i) => ({ name: p.name, value: +p.value, tone: `f${i + 1}`, members: [p] }));
    if (rest.length === 1) parts.push({ name: rest[0].name, value: +rest[0].value, tone: 'fx', members: rest });
    else if (rest.length) parts.push({ name: '기타', value: rest.reduce((a, p) => a + +p.value, 0), tone: 'fx', note: rest.map((p) => p.name).join('·'), members: rest });
    const total = parts.reduce((a, p) => a + p.value, 0);
    parts.forEach((p) => { p.share = total ? p.value / total : 0; });
    return { parts, total };
  }
  const pctOf = (x) => `${num(x * 100, 1)}%`;

  // 도넛 · 와플 카드 틀: 왼쪽에 그림(draw), 오른쪽에 색 · 이름 · 비중 범례.
  // cfg: items · sort · draw(fig, P) → 조각마다 강조할 요소 목록 · fit(fig, legend, w, h) → 카드에 남은 자리에 그림 크기 맞추기.
  // 돌려주는 show(cfg 일부)로 다른 항목을 다시 그립니다(탭 카드).
  function shareCard(card, d, f, cfg) {
    head(card, d, f, null);
    if (cfg.top) card.append(cfg.top);
    const foot = el('div', 'vzb-foot vzb-fill vzb-sh');
    const body = el('div', 'vzb-sh-body');
    body.tabIndex = 0;
    body.setAttribute('role', 'group');
    const fig = el('div', 'vzb-sh-fig');
    const leg = el('ul', 'vzb-sh-legend');
    body.append(fig, leg);
    foot.append(body);
    card.append(foot);
    let S, P, marks = [], rows = [];
    function use(next) {
      S = Object.assign({ f }, cfg, next);
      P = Object.assign(foldParts(S.items, S.sort), { f: S.f });
      fig.replaceChildren();
      marks = S.draw(fig, P);
      rows = P.parts.map((p, i) => {
        const li = el('li');
        li.dataset.i = i;
        li.append(el('i', `vzb-key is-box vzb-${p.tone}`), el('span', null, p.name), el('b', null, pctOf(p.share)));
        return li;
      });
      leg.replaceChildren(...rows);
      body.setAttribute('aria-label', `${S.name || d.title || ''}: ${P.parts.map((p) => `${p.name} ${pctOf(p.share)}`).join(', ')}, 합계 ${show(S.f, P.total)}. 좌우 화살표로 항목을 옮깁니다.`);
    }
    use({});
    let size = null;
    if (cfg.fit) onSize(foot, (w, h) => { size = [w, h]; cfg.fit(fig, leg, w, h); });
    const dim = (i) => P.parts.forEach((_, j) => {
      const off = i >= 0 && j !== i;
      (marks[j] || []).forEach((m) => m.classList.toggle('is-dim', off));
      if (rows[j]) rows[j].classList.toggle('is-dim', off);
    });
    const api = interact(body, () => P.parts.length,
      (cx, cy) => {
        const t = document.elementFromPoint(cx, cy), hit = t && t.closest('[data-i]');
        return hit && body.contains(hit) ? +hit.dataset.i : -1;
      },
      (i) => {
        dim(i);
        const p = P.parts[i], name = p.note ? `${p.name} (${p.note})` : p.name;
        const t = tipBody({ value: show(S.f, p.value), note: `전체의 ${pctOf(p.share)}` }, name);
        const r = (marks[i] && marks[i][0] ? marks[i][0] : rows[i]).getBoundingClientRect();
        return { x: r.left + r.width / 2, y: r.top, label: name, value: t.node, text: t.text };
      },
      () => dim(-1));
    return { show(next) { api.end(); use(next); if (cfg.fit && size) cfg.fit(fig, leg, ...size); } };
  }

  // 도넛: 지름 140px, 굵기 16px, 조각 사이 2px. 가운데에 합계를 적습니다.
  function ringPath(c, r1, r0, s, e) {
    if (e - s >= Math.PI * 2 - 1e-6) return ringPath(c, r1, r0, s, s + Math.PI) + ringPath(c, r1, r0, s + Math.PI, s + Math.PI * 2 - 1e-4);
    const R = (v) => +v.toFixed(2), big = e - s > Math.PI ? 1 : 0;
    const pt = (r, a) => `${R(c + r * Math.cos(a))},${R(c + r * Math.sin(a))}`;
    return `M${pt(r1, s)}A${r1},${r1} 0 ${big} 1 ${pt(r1, e)}L${pt(r0, e)}A${r0},${r0} 0 ${big} 0 ${pt(r0, s)}Z`;
  }
  function drawDonut(fig, P) {
    const size = 140, r = 70, w = 16, svg = svgEl('svg', { viewBox: `0 0 ${size} ${size}`, class: 'vzb-sh-donut', 'aria-hidden': 'true' });
    const marks = [];
    let a = -Math.PI / 2;
    P.parts.forEach((p, i) => {
      const ang = p.share * Math.PI * 2, pad = P.parts.length > 1 ? Math.min(ang * 0.4, 2 / (r - w / 2)) : 0;
      const path = svgEl('path', { d: ringPath(size / 2, r, r - w, a + pad / 2, a + ang - pad / 2), class: `vzb-sh-slice vzb-${p.tone}` });
      path.dataset.i = i;
      svg.append(path);
      marks.push([path]);
      a += ang;
    });
    const center = el('div', 'vzb-sh-center');
    center.append(figure(P.f, P.total), el('span', 'vzb-sh-cap', '합계'));
    fig.append(svg, center);
    return marks;
  }
  // 와플: 10×10칸. 칸 수는 가장 큰 나머지부터 올려 100칸에 맞추고, 큰 항목부터 왼쪽 위에서 채웁니다. 칸 크기는 fitWaffle이 정합니다.
  function drawWaffle(fig, P) {
    const raw = P.parts.map((p) => p.share * 100), cnt = raw.map(Math.floor);
    let left = P.total ? 100 - cnt.reduce((a, b) => a + b, 0) : 0;
    raw.map((v, i) => [v - cnt[i], i]).sort((x, y) => y[0] - x[0]).forEach(([, i]) => { if (left > 0) { cnt[i] += 1; left -= 1; } });
    const grid = el('div', 'vzb-sh-waffle');
    const marks = P.parts.map(() => []);
    P.parts.forEach((p, i) => {
      for (let k = 0; k < cnt[i]; k++) {
        const cell = el('i', `vzb-${p.tone}`);
        cell.dataset.i = i;
        grid.append(cell);
        marks[i].push(cell);
      }
    });
    fig.append(grid, el('p', 'vzb-sh-cap', `한 칸 1% · 합계 ${show(P.f, P.total)}`));
    return marks;
  }
  // 와플 칸 크기: 범례가 제 폭(카드 폭의 절반까지)을 쓰고 남은 폭과, 카드에 남은 높이 가운데 작은 쪽에 맞춰 10–24px로 정합니다.
  function fitWaffle(fig, leg, w, h) {
    const grid = fig.querySelector('.vzb-sh-waffle'), cap = fig.querySelector('.vzb-sh-cap');
    if (!grid) return;
    const was = leg.style.width;
    leg.style.width = 'max-content';
    const lw = Math.min(leg.getBoundingClientRect().width, 240, w / 2);
    leg.style.width = was;
    const byW = (w - lw - 32 - 18) / 10, byH = (h - (cap ? cap.offsetHeight + 10 : 0) - 18) / 10;
    grid.style.setProperty('--vzb-cell', `${Math.max(10, Math.min(24, Math.floor(Math.min(byW, byH))))}px`);
  }

  /* 탭 카드: 머리 아래를 똑같은 폭의 칸으로 나눠 지표 이름과 대표 수치를 두고, 고른 탭만 그립니다.
     탭마다 format이 달라도 되고, 대표 수치는 value가 없으면 마지막 값입니다. 값이 없는 탭(목록)은 이름만 적습니다.
     draw는 lineCard · barCard · basicList, make(탭)은 그 탭의 그리기 설정입니다. 탭은 좌우 화살표로도 옮깁니다. */
  function tabsCard(card, d, draw, make) {
    const tabs = (d.tabs || []).slice(0, 3);
    if (!tabs.length) throw new Error('tabs를 확인하세요.');
    const n = (d.labels || []).length;
    const fmts = tabs.map((t) => format(t));
    const row = el('div', 'vzb-tabs');
    row.setAttribute('role', 'tablist');
    row.setAttribute('aria-label', d.title || '지표');
    let cur = 0, chart = null;
    const spec = (i) => Object.assign({ f: fmts[i], fmt: tabs[i], name: tabs[i].name }, make(tabs[i]));
    if (!tabs.some((t) => Array.isArray(t.values))) row.classList.add('is-plain');
    const btns = tabs.map((t, i) => {
      const b = el('button', 'vzb-tab');
      b.type = 'button';
      b.setAttribute('role', 'tab');
      const vals = t.values || [];
      let k = Math.min(vals.length, n) - 1;
      while (k >= 0 && vals[k] == null) k--;
      const v = t.value != null ? t.value : k >= 0 ? vals[k] : null;
      b.append(el('span', null, t.name));
      if (v != null) {
        const [text, unit] = fmts[i].parts(v);
        const fig = el('b', null, text);
        if (unit) fig.append(el('small', null, unit));
        b.append(fig);
      }
      b.addEventListener('click', () => pick(i));
      row.append(b);
      return b;
    });
    row.addEventListener('keydown', (e) => {
      if (e.key !== 'ArrowLeft' && e.key !== 'ArrowRight') return;
      e.preventDefault();
      pick((cur + (e.key === 'ArrowRight' ? 1 : tabs.length - 1)) % tabs.length);
      btns[cur].focus();
    });
    const mark = () => btns.forEach((b, k) => { b.setAttribute('aria-selected', String(k === cur)); b.tabIndex = k === cur ? 0 : -1; });
    function pick(i) {
      if (i === cur) return;
      cur = i;
      mark();
      chart.show(spec(i));
    }
    mark();
    chart = draw(card, d, fmts[0], Object.assign({ top: row }, spec(0)));
  }

  // Donut - Tabs: 머리 아래 세 칸 탭으로 기준(채널 · 품목 · 지사 등)을 바꿔 보는 도넛입니다. 탭마다 items를 따로 적습니다.
  mount('donut-tabs', (card, d) => {
    tabsCard(card, d, shareCard, (t) => ({ items: t.items, sort: t.sort, draw: drawDonut }));
  });
})();
</script>
