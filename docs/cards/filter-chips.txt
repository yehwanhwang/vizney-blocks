  function krw(v, d) {
    const a = Math.abs(v), k = (x) => (d != null ? d : x);
    if (a >= 1e12) return [num(v / 1e12, k(1)), '조'];
    if (a >= 1e8) return [num(v / 1e8, k(a >= 1e10 ? 0 : 1)), '억'];
    if (a >= 1e5) return [num(v / 1e4, k(0)), '만'];
    return [num(v, 0), '원'];
  }
  // 달러 금액: 원래 값을 받아 K · M · B로 줄이고 소수 한 자리로 맞춥니다($745.6K). 접미사는 단위처럼 작게 씁니다.
  function usd(v, d) {
    const a = Math.abs(v), s = v < 0 ? '−' : '';
    const [div, u] = a >= 1e9 ? [1e9, 'B'] : a >= 1e6 ? [1e6, 'M'] : a >= 1e3 ? [1e3, 'K'] : [1, ''];
    return [`${s}$${num(a / div, d != null ? d : div > 1 ? 1 : 0)}`, u];
  }
  // format: eok(억 단위 값) · krw(원 단위 값) · usd(달러 값) · count(unit으로 단위) · pct(증감을 %p로) · days
  // 축이나 탭처럼 카드 안의 다른 객체로 만들 때는 두 번째 인자로 카드 데이터를 주어 lang을 잇습니다.
  function format(d, root) {
    const dec = (v, fallback) => (d.decimals != null ? d.decimals : fallback);
    const F = (o) => Object.assign(o, { lang: (root || d).lang });
    switch (d.format) {
      case 'eok': return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) || Math.abs(v) >= 100 ? 0 : 1)), '억'], rel: true });
      case 'krw': return F({ parts: (v) => krw(v, d.decimals), rel: true });
      case 'usd': return F({ parts: (v) => usd(v, d.decimals), rel: true, usd: true });
      case 'pct': return F({ parts: (v) => [num(v, dec(v, 1)), '%'], rel: false, dp: d.decimals });
      case 'days': return F({ parts: (v) => [num(v, dec(v, 1)), tx(root || d, '일')], rel: true });
      default: return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) ? 0 : 1)), d.unit || ''], rel: true });
    }
  }
  const show = (f, v) => withUnit(...f.parts(v));
  // 그릴 수 있는 숫자인지(빈 값 · 빈 글자 · NaN 제외)
  const fin = (v) => v != null && v !== '' && isFinite(v);
  // 비율의 차이(%p) 숫자: 소수 자릿수는 decimals(비우면 한 자리)를 따릅니다. 0이 아닌 차이가 0.0으로 보일 만큼 작으면
  // 자리를 늘려(최대 세 자리) 1.12%에서 1.16%로 오른 것을 0.0%p가 아니라 0.03%p로 적습니다.
  function pp(f, diff) {
    const a = Math.abs(diff);
    let k = f.dp != null ? f.dp : 1;
    while (k < 3 && a > 1e-9 && Math.round(a * 10 ** k) === 0) k++;
    return num(a, k);
  }
  // 증감: 방향은 화살표로, 좋고 나쁨(tone)은 better(up · down · neutral)로 정합니다. 비율은 %p로 적습니다.
  function delta(f, value, base, better) {
    const diff = value - base;
    const dir = Math.abs(diff) < 1e-9 ? 0 : Math.sign(diff);
    const want = better === 'down' ? -1 : better === 'neutral' ? 0 : 1;
    const tone = dir === 0 || want === 0 ? 'flat' : dir === want ? 'good' : 'bad';
    let text;
    if (!f.rel) text = tx(f, '%s%p', pp(f, diff));
    else if (base) text = `${num(Math.abs(diff / base) * 100, 1)}%`;
    else text = show(f, Math.abs(diff));
    return { dir, tone, text };
  }
  function arrow(dir) {
    const s = svgEl('svg', { class: 'vzb-arrow', viewBox: '0 0 12 12', 'aria-hidden': 'true' });
    s.append(svgEl('path', { d: dir > 0 ? 'M3.5 8.5l5-5M4.5 3.5h4v4' : 'M3.5 3.5l5 5M8.5 4.5v4h-4' }));
    return s;
  }
  // 대표 수치(단위는 작게)
  function figure(f, value) {
    const n = el('div', 'vzb-num');
    const [text, unit] = f.parts(value);
    n.append(text);
    if (unit) n.append(el('span', 'vzb-unit', unit));
    return n;
  }
  // 증감 배지: 화살표는 오르내림, 색은 좋고 나쁨(better)입니다.
  function badge(f, value, base, better) {
    const dl = delta(f, value, base, better);
    const b = el('span', `vzb-badge vzb-${dl.tone}`);
    if (dl.dir) b.append(arrow(dl.dir), el('span', 'vzb-sr', tx(f, dl.dir > 0 ? '증가' : '감소')));
    b.append(dl.text);
    return b;
  }
  // 머리: 제목 · 기간 부제(subtitle) · 대표 수치와 바로 뒤의 증감 배지. value가 null이면 제목만 둡니다.
  // 배지는 숫자 바로 뒤에 붙어 있고, 카드가 좁아 한 줄에 다 들어가지 않으면 비교 문구만 숫자 아래 줄로 내려갑니다.
  // 내려갔는지는 카드의 vzb-wrapped로 알려, 높이가 1단위인 카드가 아래 여백으로 그 줄을 받을 수 있게 합니다.
  function head(card, d, f, value, cmp, cmpText) {
    const hd = el('div', 'vzb-hd');
    hd.append(el('div', 'vzb-title', d.title || ''));
    if (d.subtitle) hd.append(el('div', 'vzb-sub', d.subtitle));
    card.append(hd);
    if (value == null) return;
    const fig = el('div', 'vzb-fig');
    const n = figure(f, value);
    fig.append(n);
    if (cmp && cmp.value != null) {
      const line = el('span', 'vzb-delta'), vs = el('span', 'vzb-vs', cmpText || tx(d, '%s 대비', cmp.label || tx(d, '비교 기간')));
      line.append(badge(f, value, cmp.value, d.better), vs);
      fig.append(line);
      onSize(fig, () => card.classList.toggle('vzb-wrapped', vs.offsetTop >= n.offsetTop + n.offsetHeight - 1));   // 숫자 칸 아래에서 시작하면 내려간 것
    }
    card.append(fig);
  }
  // 같은 종류의 카드를 모두 찾아 그립니다. 코드를 두 번 붙여 넣어도 카드는 한 번만 그립니다.
  // 데이터가 바뀌면 다시 그립니다: 페이지의 스크립트가 카드 안 JSON(<script type="application/json">)의 글자를 바꾸면 그 카드만
  // 새 요소로 갈아 끼워 새로 그립니다(필터 · 기간 컨트롤과 잇는 길). 갈아 끼운 카드에는 data-vzb-live가 붙어, 움직이는 카드는
  // 처음부터 다시 재생하지 않고 마지막 장면을 바로 그립니다. 예전 요소가 페이지에서 빠지므로 그 카드의 움직임도 함께 끝납니다.
  function mount(type, draw) {
    const watch = (card, src) => {
      if (!src || !window.MutationObserver) return;
      const mo = new MutationObserver(() => {
        mo.disconnect();
        if (!card.isConnected) return;
        const next = card.cloneNode(false);
        next.removeAttribute('data-vzb-ready');
        next.setAttribute('data-vzb-live', '');
        next.append(src);
        card.replaceWith(next);
        run();
      });
      mo.observe(src, { childList: true, characterData: true, subtree: true });
    };
    const run = () => {
      document.querySelectorAll(`.vzb-card[data-vzb="${type}"]:not([data-vzb-ready])`).forEach((card) => {
        card.setAttribute('data-vzb-ready', '');
        const src = card.querySelector('script[type="application/json"]');
        const raw = src ? src.textContent : '';
        watch(card, src);
        let d;
        try { d = JSON.parse(raw); } catch (e) {
          card.append(el('p', 'vzb-error', tx(/"lang"\s*:\s*"en"/.test(raw) ? { lang: 'en' } : null, '카드 데이터(JSON)를 읽지 못했습니다: %s', e.message)));
          return;
        }
        card.setAttribute('role', 'group');
        if (d.title) card.setAttribute('aria-label', d.title);
        // 어두운 바탕: JSON에 "theme": "dark"를 적으면 Vizney 검정 바탕에 노랑을 첫째 색으로 그립니다(tokens.css의 .vzb-dark).
        card.classList.toggle('vzb-dark', d.theme === 'dark');
        try { draw(card, d, format(d)); } catch (e) {
          card.append(el('p', 'vzb-error', tx(d, '카드를 그리지 못했습니다: %s', e.message)));
        }
      });
    };
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
    else run();
  }
  // 크기가 바뀔 때마다 그래픽을 다시 그립니다. 차트는 카드에 남은 높이를 채웁니다.
  function onSize(node, fn) {
    let lw = 0, lh = 0;
    new ResizeObserver((entries) => {
      const w = Math.floor(entries[0].contentRect.width), h = Math.floor(entries[0].contentRect.height);
      if (w > 0 && h > 0 && (w !== lw || h !== lh)) { lw = w; lh = h; fn(w, h); }
    }).observe(node);
  }

  /* 굴러가는 숫자(컨트롤 공통): 숫자 한 자리마다 0–9를 세로로 쌓은 칸을 두고 새 값의 자리로 굴립니다(구독 수가 오르는 단추처럼).
     roll(box, text): box 안의 글자를 text로 바꿉니다. 자리 수와 쉼표 자리가 그대로면 칸이 제자리에서 구르고, 바뀌면 칸을 새로 만들어
     0에서 굴러 올라옵니다. 움직임은 roll.css의 transition이 그리고, box는 화면 낭독기에 읽히지 않으므로 값은 부르는 쪽이 따로 알립니다. */
  function roll(box, text) {
    const chars = [...String(text)], shape = chars.map((c) => (/\d/.test(c) ? '0' : c)).join('');
    box.setAttribute('aria-hidden', 'true');
    if (box.dataset.shape !== shape) {
      box.dataset.shape = shape;
      box.replaceChildren(...chars.map((c) => {
        if (!/\d/.test(c)) return el('span', 'vzb-rl-ch', c);
        const col = el('span', 'vzb-rl-col'), strip = el('span', 'vzb-rl-strip');
        for (let k = 0; k < 10; k++) strip.append(el('span', null, String(k)));
        col.append(strip);
        return col;
      }));
      box.getBoundingClientRect();   // 새 칸은 0에 선 채로 그려진 뒤 굴러갑니다
    }
    [...box.children].forEach((node, i) => { if (node.classList.contains('vzb-rl-col')) node.firstChild.style.transform = `translateY(${-10 * +chars[i]}%)`; });
  }

  // Filter Chips: 정해 둔 항목(options) 가운데 여러 개를 칩으로 고르는 컨트롤입니다(채널 · 지역 · 상품군 필터). 카드 머리 없이 칩만 그립니다.
  // 고른 칩은 글자색으로 채우고 체크가 그려집니다. 맨 앞의 전체 칩(all, 비우면 '전체', 빈 글자면 없음)은 고른 수를 굴러가는 숫자로 보여 주고,
  // 누르면 모두 고르거나 이미 모두 골랐으면 모두 풉니다. value는 처음 고른 항목이고 비우면 모두입니다.
  // 값이 바뀌면 카드에서 vzb-change 이벤트(bubbles, detail: { name, value: 고른 항목을 options 순서대로 })를 보내고, data-vzb-value에도 적습니다.
  // 칩은 누르면 살짝 눌렸다 튀어 오르고, 고르면 체크가 그려지며 칩이 넓어집니다(filter-chips.css).
  // 칩이 두 줄을 넘으면 한 줄로 두고 가로로 밀어 봅니다. 카드를 감싼 요소에 data-vzb-loop가 있으면(갤러리) 손을 대기 전까지 고르기를 차례로 보여 줍니다.
  mount('filter-chips', (card, d, f) => {
    const opts = [...new Set((Array.isArray(d.options) ? d.options : []).filter((o) => o != null && o !== '').map(String))];
    if (!opts.length) throw new Error(tx(d, 'options를 확인하세요.'));
    const sel = new Set(Array.isArray(d.value) ? d.value.map(String).filter((v) => opts.includes(v)) : opts);
    const foot = el('div', 'vzb-ch'), list = el('div', 'vzb-ch-list'), say = el('p', 'vzb-sr');
    list.setAttribute('role', 'group');
    list.setAttribute('aria-label', d.label || d.title || tx(d, '필터'));   // 제목을 그리지 않으므로 화면 낭독기용 이름은 label에서 읽습니다
    list.setAttribute('data-vzb-input', '');
    say.setAttribute('aria-live', 'polite');
    const tick = () => {
      const s = svgEl('svg', { class: 'vzb-ch-ck', viewBox: '0 0 12 12', 'aria-hidden': 'true' });
      s.append(svgEl('path', { d: 'M2.2 6.4l2.5 2.5 5.1-5.6' }));
      return s;
    };
    const chip = (label) => {
      const b = el('button', 'vzb-ch-chip');
      b.type = 'button';
      b.append(tick(), el('span', 'vzb-ch-tx', label));
      return b;
    };
    const allLabel = d.all == null ? tx(d, '전체') : String(d.all);
    let all = null, count = null;
    if (allLabel) {
      all = chip(allLabel);
      all.classList.add('is-all');
      count = el('span', 'vzb-rl vzb-ch-count');
      all.append(count);
      list.append(all);
    }
    const chips = opts.map((o) => { const b = chip(o); list.append(b); return b; });
    foot.append(list, say);
    card.append(foot);

    function paint() {
      chips.forEach((b, i) => b.setAttribute('aria-pressed', String(sel.has(opts[i]))));
      if (all) {
        all.setAttribute('aria-pressed', String(sel.size === opts.length));
        all.setAttribute('aria-label', tx(d, '%1, %2개 중 %3개 고름', allLabel, opts.length, sel.size));
        roll(count, String(sel.size));
      }
    }
    function emit(quiet) {
      const value = opts.filter((o) => sel.has(o));
      card.setAttribute('data-vzb-value', JSON.stringify(value));
      if (quiet) return;
      say.textContent = tx(d, '%1개 중 %2개 고름', opts.length, sel.size);
      card.dispatchEvent(new CustomEvent('vzb-change', { bubbles: true, detail: { name: d.name || '', value } }));
    }
    // 두 줄을 넘으면 한 줄로 두고 가로로 밉니다. 모든 칩에 체크가 그려졌을 때(가장 넓을 때)로 재므로, 고를 때마다 줄 모양이 바뀌지 않습니다.
    // 가로로 밀 때는 오른쪽 끝을 흐리게 해 칩이 더 있음을 알리고, 끝까지 밀면 흐림을 거둡니다.
    const fit = () => {
      list.classList.remove('is-scroll');
      list.classList.add('is-measure');
      const first = list.firstElementChild, over = !!first && list.scrollHeight > first.offsetHeight * 2 + 12;
      list.classList.remove('is-measure');
      list.classList.toggle('is-scroll', over);
      ends();
    };
    const ends = () => list.classList.toggle('is-end', list.scrollLeft + list.clientWidth >= list.scrollWidth - 2);
    list.addEventListener('scroll', ends, { passive: true });
    function toggle(i) {   // i: 항목 번호, -1은 전체 칩
      if (i < 0) { if (sel.size === opts.length) sel.clear(); else opts.forEach((o) => sel.add(o)); }
      else if (sel.has(opts[i])) sel.delete(opts[i]);
      else sel.add(opts[i]);
      paint();
      emit();
    }
    let idle = 0;
    list.addEventListener('click', (e) => {
      const b = e.target.closest('.vzb-ch-chip');
      if (!b) return;
      idle = performance.now() + 6000;
      toggle(b === all ? -1 : chips.indexOf(b));
    });
    paint();
    emit(true);
    onSize(list, fit);

    // 갤러리(data-vzb-loop): 화면에 보이는 동안, 손을 대기 전과 손을 뗀 지 6초 뒤부터 칩을 차례로 눌러 보입니다.
    if (card.closest('[data-vzb-loop]')) {
      const plan = [3, 4, -1, 5, 4];   // 한 바퀴 돌면 처음 고른 넷으로 돌아옵니다
      let k = 0, seen = true;
      if ('IntersectionObserver' in window) new IntersectionObserver((es) => { seen = es[es.length - 1].isIntersecting; }).observe(card);
      const timer = setInterval(() => {
        if (!card.isConnected) { clearInterval(timer); return; }
        if (!seen || performance.now() < idle) return;
        const i = plan[k++ % plan.length], b = i < 0 ? all : chips[i % chips.length];
        if (!b) return;
        b.classList.add('is-press');
        setTimeout(() => b.classList.remove('is-press'), 120);
        toggle(i < 0 ? -1 : i % chips.length);
      }, 1300);
    }
  });
})();
</script>
