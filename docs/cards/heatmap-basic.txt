  'use strict';
  /* 공통 도구: 숫자 표기, 증감, 카드 머리, 그리기 시작 */
  const SVGNS = 'http://www.w3.org/2000/svg';
  function el(tag, cls, text) {
    const n = document.createElement(tag);
    if (cls) n.className = cls;
    if (text != null) n.textContent = text;
    return n;
  }
  function svgEl(tag, attrs) {
    const n = document.createElementNS(SVGNS, tag);
    for (const k in attrs) n.setAttribute(k, attrs[k]);
    return n;
  }
  const nfs = {};
  function num(v, d) {
    nfs[d] = nfs[d] || new Intl.NumberFormat('ko-KR', { minimumFractionDigits: d, maximumFractionDigits: d });
    return nfs[d].format(v);
  }
  function krw(v) {
    const a = Math.abs(v);
    if (a >= 1e12) return [num(v / 1e12, 1), '조'];
    if (a >= 1e8) return [num(v / 1e8, a >= 1e10 ? 0 : 1), '억'];
    if (a >= 1e4) return [num(v / 1e4, 0), '만'];
    return [num(v, 0), '원'];
  }
  // format: eok(억 단위 값) · krw(원 단위 값) · count(unit으로 단위) · pct(증감을 %p로) · days
  function format(d) {
    const dec = (v, fallback) => (d.decimals != null ? d.decimals : fallback);
    switch (d.format) {
      case 'eok': return { parts: (v) => [num(v, dec(v, Number.isInteger(v) || Math.abs(v) >= 100 ? 0 : 1)), '억'], rel: true };
      case 'krw': return { parts: krw, rel: true };
      case 'pct': return { parts: (v) => [num(v, dec(v, 1)), '%'], rel: false };
      case 'days': return { parts: (v) => [num(v, dec(v, 1)), '일'], rel: true };
      default: return { parts: (v) => [num(v, dec(v, Number.isInteger(v) ? 0 : 1)), d.unit || ''], rel: true };
    }
  }
  const show = (f, v) => f.parts(v).join('');
  // 그릴 수 있는 숫자인지(빈 값 · 빈 글자 · NaN 제외)

  function delta(f, value, base, better) {
    const diff = value - base;
    const dir = Math.abs(diff) < 1e-9 ? 0 : Math.sign(diff);
    const want = better === 'down' ? -1 : better === 'neutral' ? 0 : 1;
    const tone = dir === 0 || want === 0 ? 'flat' : dir === want ? 'good' : 'bad';

    else if (base) text = `${num(Math.abs(diff / base) * 100, 1)}%`;
    else text = show(f, Math.abs(diff));
    return { dir, tone, text };
  }
  function arrow(dir) {
    const s = svgEl('svg', { class: 'vzb-arrow', viewBox: '0 0 12 12', 'aria-hidden': 'true' });
    s.append(svgEl('path', { d: dir > 0 ? 'M3.5 8.5l5-5M4.5 3.5h4v4' : 'M3.5 3.5l5 5M8.5 4.5v4h-4' }));
    return s;
  }
  // 대표 수치(단위는 작게)

  // 증감 배지: 화살표는 오르내림, 색은 좋고 나쁨(better)입니다.
  function badge(f, value, base, better) {
    const dl = delta(f, value, base, better);

    if (dl.dir) b.append(arrow(dl.dir), el('span', 'vzb-sr', dl.dir > 0 ? '증가' : '감소'));

  function mount(type, draw) {
    const run = () => {
      document.querySelectorAll(`.vzb-card[data-vzb="${type}"]:not([data-vzb-ready])`).forEach((card) => {
        card.setAttribute('data-vzb-ready', '');

          card.append(el('p', 'vzb-error', `카드를 그리지 못했습니다: ${e.message}`));

  // 크기가 바뀔 때마다 그래픽을 다시 그립니다. 차트는 카드에 남은 높이를 채웁니다.
  function onSize(node, fn) {
    let lw = 0, lh = 0;
    new ResizeObserver((entries) => {
      const w = Math.floor(entries[0].contentRect.width), h = Math.floor(entries[0].contentRect.height);
      if (w > 0 && h > 0 && (w !== lw || h !== lh)) { lw = w; lh = h; fn(w, h); }
    }).observe(node);
  }

  /* 툴팁과 조작: 마우스는 올리면 보이고 벗어나면 숨깁니다. 터치는 탭하면 보이고 다른 곳을 탭하면 닫힙니다.
     키보드는 그래프에 포커스한 뒤 화살표로 옮깁니다(가로 그래프는 좌우, 목록은 위아래). */
  let tip;

  function showTip(x, y, label, value, side) {
    if (!tip) {
      tip = el('div', 'vzb-tip');
      tip.setAttribute('aria-hidden', 'true');
      document.body.append(tip);
    }
    tip.replaceChildren(el('span', 'vzb-tip-label', label), typeof value === 'string' ? el('span', 'vzb-tip-value', value) : value);
    tip.classList.add('is-on');
    const w = tip.offsetWidth, h = tip.offsetHeight, W = window.innerWidth, H = window.innerHeight;
    if (side) {
      // side가 'left'이면 기준선 왼쪽을 먼저, 그 밖에는 오른쪽을 먼저 씁니다. 자리가 모자라면 반대쪽으로 넘깁니다.
      const right = x + 14 + w <= W - 8, left = x - w - 14 >= 8;
      tip.style.left = `${(side === 'left' ? left || !right : !right) ? Math.max(8, x - w - 14) : x + 14}px`;
      tip.style.top = `${Math.min(Math.max(8, y), H - h - 8)}px`;
      return;
    }
    tip.style.left = `${Math.min(Math.max(8, x - w / 2), W - w - 8)}px`;
    tip.style.top = `${y - h - 12 < 8 ? y + 16 : y - h - 12}px`;

  // 툴팁 내용: { value, note } 또는 { rows: [{ tone, value, name }], note }. 값이 먼저, 이름은 뒤에 옅게 적습니다.
  function tipBody(spec, label) {

  // 차이를 부호와 함께 적습니다. 비율 지표는 %p입니다.
  function signed(f, diff) {
    const s = diff > 0 ? '+' : diff < 0 ? '−' : '';
    return f.rel ? `${s}${show(f, Math.abs(diff))}` : `${s}${num(Math.abs(diff), 1)}%p`;

  // pick(clientX, clientY) → 순번, point(순번) → { x, y, label, value, text, side }(화면 좌표), clear() → 강조 해제
  // n은 개수나 개수를 돌려주는 함수입니다. vertical이면 위아래 화살표로 옮깁니다. 돌려주는 { show(순번), at() }로 키를 더 붙입니다.
  // 아무것도 고르지 않은 채 화살표를 누르면 기간 차트는 마지막(최근)부터, vertical이나 first면 첫 순번부터 짚습니다.
  function interact(plot, n, pick, point, clear, vertical, first) {
    let idx = -1, touched = false;
    const count = () => (typeof n === 'function' ? n() : n);
    const live = el('span', 'vzb-sr');
    live.setAttribute('aria-live', 'polite');
    plot.after(live);
    const go = (i) => {
      if (i < 0 || i >= count()) return;
      idx = i;
      const p = point(i);
      showTip(p.x, p.y, p.label, p.value, p.side);
      live.textContent = p.text || `${p.label} ${p.value}`;
    };
    const end = () => {
      if (idx < 0) return;
      idx = -1;
      clear();
      hideTip();
    };
    const [back, next] = vertical ? ['ArrowUp', 'ArrowDown'] : ['ArrowLeft', 'ArrowRight'];
    // pick이 -1이면(조각 밖) 강조를 풉니다.
    const at = (e) => { const i = pick(e.clientX, e.clientY); if (i < 0) end(); else go(i); };
    plot.addEventListener('pointermove', (e) => { if (e.pointerType !== 'touch' || touched) at(e); });
    plot.addEventListener('pointerdown', (e) => { if (e.pointerType === 'touch') { touched = true; at(e); } });
    plot.addEventListener('pointerleave', (e) => { if (e.pointerType !== 'touch') end(); });
    document.addEventListener('pointerdown', (e) => { if (touched && !plot.contains(e.target)) { touched = false; end(); } }, true);
    plot.addEventListener('keydown', (e) => {
      if (e.key !== back && e.key !== next) return;
      e.preventDefault();
      const last = count() - 1;
      go(idx < 0 ? (vertical || first ? 0 : last) : Math.max(0, Math.min(last, idx + (e.key === next ? 1 : -1))));
    });
    plot.addEventListener('blur', end);
    window.addEventListener('scroll', end, { passive: true });
    return { show: go, at: () => idx, end };
  }

  /* 표 카드 공통: 머리글은 가운데, 이름은 왼쪽, 숫자는 오른쪽 정렬과 같은 폭 숫자로 적고, 단위는 열 제목에 한 번만 적습니다.
     줄에 마우스를 올리면 그 줄을 옅게 칠합니다. sortable이면 열 제목을 눌러 정렬하고, 합계 줄(total)은 정렬과 상관없이 맨 아래에 둡니다. */

  // 칸 숫자: 단위 없이 적습니다(단위는 열 제목에). pct 열은 %를 붙입니다.
  function cellNum(v, c) {
    if (v == null || !isFinite(v)) return '–';
    const dec = c.decimals != null ? c.decimals : Number.isInteger(+v) ? 0 : 1;
    return `${num(+v, dec).replace(/^-/, '−')}${c.pct ? '%' : ''}`;
  }
  // 증감 배지: 기준(base) 대비 증감률. KPI 카드의 배지와 같습니다.
  const relF = { parts: (v) => [num(v, 1), ''], rel: true };
  const changeBadge = (value, base, better) => (value == null || !base ? el('span', 'vzb-tb-none', '–') : badge(relF, +value, +base, better));
  // 합계 줄의 증감: 두 값이 모두 있는 줄만 더한 뒤 합계끼리 계산합니다.
  const ok = (v) => v != null && v !== '' && isFinite(v);
  function changeTotal(rows, of, base, better) {
    let a = 0, b = 0;
    rows.forEach((r) => { if (ok(r[of]) && ok(r[base])) { a += +r[of]; b += +r[base]; } });
    return changeBadge(a, b, better);
  }
  // 스파크라인: 72×20px 곡선과 마지막 점. 줄마다 자기 범위에 맞춰 그리되, 범위가 값의 10%보다 좁으면 10%로 넓혀
  // 거의 변하지 않은 줄이 크게 출렁여 보이지 않게 합니다(그 줄은 가운데 높이에서 거의 평평하게 그려집니다).
  function sparkEl(values) {
    const w = 72, h = 20, pad = 3, vals = (values || []).map((v) => (v == null ? null : +v));
    const got = vals.filter((v) => v != null), n = vals.length;
    let lo = Math.min(...got), hi = Math.max(...got);
    const span = Math.max(hi - lo, Math.max(Math.abs(lo), Math.abs(hi)) * 0.1);
    lo -= (span - (hi - lo)) / 2;
    hi = lo + span;
    const svg = svgEl('svg', { class: 'vzb-tb-spark', viewBox: `0 0 ${w} ${h}`, width: w, height: h, 'aria-hidden': 'true' });
    if (!got.length) return svg;
    const pts = [];
    vals.forEach((v, i) => { if (v != null) pts.push({ x: n < 2 ? w / 2 : pad + (i * (w - pad * 2)) / (n - 1), y: h - pad - ((v - lo) / (hi - lo || 1)) * (h - pad * 2) }); });
    const last = pts[pts.length - 1];
    svg.append(svgEl('path', { d: monotone(pts), class: 'vzb-tb-spark-line' }), svgEl('circle', { cx: last.x, cy: last.y, r: 2.5, class: 'vzb-tb-spark-dot' }));
    return svg;
  }

  // cfg: columns [{ key, label, unit, pct, decimals, align('txt' | 'num'), value(row), cell(row, ctx) → 요소, total(rows) → 값이나 요소, sort(false면 정렬 안 함), head(클래스) }]
  //   rows · total(합계 줄 이름) · sortable · sortKey · sortDir(1 오름 · -1 내림) · top(머리 아래 요소) · prepare(rows) → ctx · rowClass(row)
  function tableCard(card, d, f, cfg) {
    head(card, d, f, null);
    if (cfg.top) card.append(cfg.top);
    const foot = el('div', 'vzb-foot vzb-tb');
    const table = el('table', 'vzb-tb-table');
    const thead = el('thead'), tbody = el('tbody'), tfoot = el('tfoot');
    table.append(el('caption', 'vzb-sr', `${d.title || ''}${d.subtitle ? `, ${d.subtitle}` : ''}${cfg.sortable ? '. 열 제목을 누르면 정렬합니다.' : ''}`), thead, tbody, tfoot);
    foot.append(table);
    card.append(foot);
    const cols = cfg.columns, rows = cfg.rows || [];
    const ctx = Object.assign({ f, rows }, cfg.prepare ? cfg.prepare(rows) : {});
    const val = (r, c) => (c.value ? c.value(r) : r[c.key]);
    const align = (c, j) => c.align || (j ? 'num' : 'txt');
    let sortKey = cfg.sortKey, dir = cfg.sortDir || -1;
    const hr = el('tr');
    const ths = cols.map((c, j) => {
      const th = el('th', `vzb-tb-${align(c, j)}${c.head ? ` ${c.head}` : ''}`);
      th.scope = 'col';
      const label = el('span', null, c.label || '');
      if (c.unit) label.append(el('small', null, `(${c.unit})`));
      if (cfg.sortable && c.sort !== false) {
        const btn = el('button', 'vzb-tb-sort');
        btn.type = 'button';
        btn.append(label, el('i', 'vzb-tb-arrow'));
        btn.addEventListener('click', () => {
          if (sortKey === c.key) dir = -dir;
          else { sortKey = c.key; dir = align(c, j) === 'txt' ? 1 : -1; }
          render();
        });
        th.append(btn);
      } else th.append(label);
      hr.append(th);
      return th;
    });
    thead.append(hr);
    const cellOf = (r, c, j) => {
      const td = el(j ? 'td' : 'th', `vzb-tb-${align(c, j)}`);
      if (!j) td.scope = 'row';
      const v = val(r, c), out = c.cell ? c.cell(r, ctx) : typeof v === 'string' ? v : cellNum(v, c);
      if (!j && typeof out === 'string') {   // 긴 이름은 말줄임표로 줄이고 전체 이름은 title에 둡니다.
        const name = el('span', 'vzb-tb-name', out);
        name.title = out;
        td.append(name);
      } else if (typeof out === 'string') td.textContent = out; else if (out) td.append(out);
      return td;
    };
    function render() {
      const list = rows.slice();
      const c = sortKey && cols.find((x) => x.key === sortKey);
      if (c) {
        list.sort((a, b) => {
          const x = val(a, c), y = val(b, c), nx = x == null || x === '' || (typeof x === 'number' && !isFinite(x)), ny = y == null || y === '' || (typeof y === 'number' && !isFinite(y));
          if (nx || ny) return nx === ny ? 0 : nx ? 1 : -1;   // 빈 값은 정렬 방향과 상관없이 맨 아래에 둡니다
          return (typeof x === 'string' || typeof y === 'string' ? String(x).localeCompare(String(y), 'ko', { numeric: true }) : x - y) * dir;
        });
      }
      if (!list.length) {
        const td = el('td', 'vzb-tb-empty', '표시할 데이터가 없습니다.');
        td.colSpan = cols.length;
        const tr = el('tr');
        tr.append(td);
        tbody.replaceChildren(tr);
        return;
      }
      tbody.replaceChildren(...list.map((r) => {
        const tr = el('tr', cfg.rowClass ? cfg.rowClass(r) : null);
        cols.forEach((c, j) => tr.append(cellOf(r, c, j)));
        return tr;
      }));
      ths.forEach((th, j) => th.setAttribute('aria-sort', cfg.sortable && cols[j].key === sortKey ? (dir > 0 ? 'ascending' : 'descending') : 'none'));
    }
    render();
    if (cfg.total && rows.length) {
      const tr = el('tr');
      cols.forEach((c, j) => {
        const td = el(j ? 'td' : 'th', `vzb-tb-${align(c, j)}`);
        if (!j) { td.scope = 'row'; td.textContent = cfg.total; }
        else if (c.total) {
          const out = c.total(rows, ctx);
          if (out == null) td.textContent = '';
          else if (typeof out === 'number') td.textContent = cellNum(out, c);
          else if (typeof out === 'string') td.textContent = out;
          else td.append(out);
        }
        tr.append(td);
      });
      tfoot.append(tr);
    }
    return { table, tbody, ctx };
  }
  const sumOf = (key) => (rows) => rows.reduce((a, r) => a + (+r[key] || 0), 0);
  // 값 열의 단위: unit을 적지 않으면 format의 단위(억 · 건 등)를 씁니다.
  const unitOf = (d, f) => d.unit || f.parts(0)[1] || '';

  // Heatmap - Basic: 행 × 열 칸을 값의 크기에 따라 주황 한 가지의 농도로 칠합니다. 칸에는 숫자를 적지 않아 색의 패턴이 먼저 보이고,
  // 값은 표 위 범례의 양 끝 값과 칸의 툴팁으로 읽습니다(화면 낭독기는 칸마다 숫자를 읽습니다). "scale": "row"면 행마다 따로 칠합니다.
  // 처음에는 적은 순서대로 두고, 열 제목을 누르면 그 열로 정렬합니다. 행 이름 열을 누르면 다시 처음 순서(숫자 순)가 됩니다.
  mount('heatmap-basic', (card, d, f) => {
    const heads = d.columns || [], rows = d.rows || [], byRow = d.scale === 'row';
    const range = (vals) => {
      const got = (vals || []).filter(ok).map(Number);
      return got.length ? [Math.min(...got), Math.max(...got)] : null;
    };
    const all = range(rows.flatMap((r) => r.values || []));
    const mix = (v, [lo, hi]) => 8 + (hi > lo ? (v - lo) / (hi - lo) : 0.5) * 72;   // 주황을 8–80% 섞습니다. 값이 모두 같으면 가운데 농도입니다
    const ramp = all && el('div', 'vzb-tb-ramp');
    if (ramp) ramp.append(el('span', null, byRow ? '행마다 작음' : show(f, all[0])), el('i'), el('span', null, byRow ? '큼' : show(f, all[1])));
    const { table } = tableCard(card, d, f, {
      rows,
      top: ramp,
      sortable: true,
      columns: [{ key: 'name', label: d.nameLabel || '' }].concat(heads.map((label, j) => ({
        key: `c${j}`, label, value: (r) => (r.values || [])[j],
        cell: (r) => {
          const v = (r.values || [])[j], s = el('span', `vzb-tb-heat${ok(v) ? '' : ' is-none'}`);
          s.append(el('span', 'vzb-sr', cellNum(v, {})));
          if (ok(v)) s.style.setProperty('--vzb-heat', `${mix(+v, byRow ? range(r.values) : all).toFixed(0)}%`);
          s.dataset.label = `${r.name} · ${label}`;
          s.dataset.value = ok(v) ? show(f, +v) : '–';
          return s;
        },
      }))),
    });
    table.classList.add('is-heat');
    // 칸에 마우스를 올리거나 탭하면 그 칸의 값을 툴팁으로 보여줍니다.
    table.addEventListener('pointerover', (e) => {
      const s = e.target.closest && e.target.closest('.vzb-tb-heat');
      if (!s) { hideTip(); return; }
      const b = s.getBoundingClientRect();
      showTip(b.left + b.width / 2, b.top, s.dataset.label, s.dataset.value);
    });
    table.addEventListener('pointerleave', hideTip);
  });
})();
</script>
