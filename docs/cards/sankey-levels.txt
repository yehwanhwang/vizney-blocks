(() => {
  'use strict';
  /* 공통 도구: 숫자 표기, 증감, 카드 머리, 그리기 시작 */
  const SVGNS = 'http://www.w3.org/2000/svg';
  function el(tag, cls, text) {
    const n = document.createElement(tag);
    if (cls) n.className = cls;
    if (text != null) n.textContent = text;
    return n;
  }
  function svgEl(tag, attrs) {
    const n = document.createElementNS(SVGNS, tag);
    for (const k in attrs) n.setAttribute(k, attrs[k]);
    return n;
  }
  const nfs = {};
  function num(v, d) {
    nfs[d] = nfs[d] || new Intl.NumberFormat('ko-KR', { minimumFractionDigits: d, maximumFractionDigits: d });
    return nfs[d].format(v);
  }
  function krw(v) {
    const a = Math.abs(v);
    if (a >= 1e12) return [num(v / 1e12, 1), '조'];
    if (a >= 1e8) return [num(v / 1e8, a >= 1e10 ? 0 : 1), '억'];
    if (a >= 1e4) return [num(v / 1e4, 0), '만'];
    return [num(v, 0), '원'];
  }
  // format: eok(억 단위 값) · krw(원 단위 값) · count(unit으로 단위) · pct(증감을 %p로) · days
  function format(d) {
    const dec = (v, fallback) => (d.decimals != null ? d.decimals : fallback);
    switch (d.format) {
      case 'eok': return { parts: (v) => [num(v, dec(v, Number.isInteger(v) || Math.abs(v) >= 100 ? 0 : 1)), '억'], rel: true };
      case 'krw': return { parts: krw, rel: true };
      case 'pct': return { parts: (v) => [num(v, dec(v, 1)), '%'], rel: false };
      case 'days': return { parts: (v) => [num(v, dec(v, 1)), '일'], rel: true };
      default: return { parts: (v) => [num(v, dec(v, Number.isInteger(v) ? 0 : 1)), d.unit || ''], rel: true };
    }
  }
  const show = (f, v) => f.parts(v).join('');
  // 그릴 수 있는 숫자인지(빈 값 · 빈 글자 · NaN 제외)
  const fin = (v) => v != null && v !== '' && isFinite(v);
  // 증감: 방향은 화살표로, 좋고 나쁨(tone)은 better(up · down · neutral)로 정합니다. 비율은 %p로 적습니다.
  function delta(f, value, base, better) {
    const diff = value - base;
    const dir = Math.abs(diff) < 1e-9 ? 0 : Math.sign(diff);
    const want = better === 'down' ? -1 : better === 'neutral' ? 0 : 1;
    const tone = dir === 0 || want === 0 ? 'flat' : dir === want ? 'good' : 'bad';
    let text;
    if (!f.rel) text = `${num(Math.abs(diff), 1)}%p`;
    else if (base) text = `${num(Math.abs(diff / base) * 100, 1)}%`;
    else text = show(f, Math.abs(diff));
    return { dir, tone, text };
  }
  function arrow(dir) {
    const s = svgEl('svg', { class: 'vzb-arrow', viewBox: '0 0 12 12', 'aria-hidden': 'true' });
    s.append(svgEl('path', { d: dir > 0 ? 'M3.5 8.5l5-5M4.5 3.5h4v4' : 'M3.5 3.5l5 5M8.5 4.5v4h-4' }));
    return s;
  }
  // 대표 수치(단위는 작게)
  function figure(f, value) {
    const n = el('div', 'vzb-num');
    const [text, unit] = f.parts(value);
    n.append(text);
    if (unit) n.append(el('span', 'vzb-unit', unit));
    return n;
  }
  // 증감 배지: 화살표는 오르내림, 색은 좋고 나쁨(better)입니다.
  function badge(f, value, base, better) {
    const dl = delta(f, value, base, better);
    const b = el('span', `vzb-badge vzb-${dl.tone}`);
    if (dl.dir) b.append(arrow(dl.dir), el('span', 'vzb-sr', dl.dir > 0 ? '증가' : '감소'));
    b.append(dl.text);
    return b;
  }
  // 머리: 제목 · 기간 부제(subtitle) · 대표 수치와 바로 뒤의 증감 배지. value가 null이면 제목만 둡니다.
  function head(card, d, f, value, cmp, cmpText) {

    card.append(hd);

      line.append(badge(f, value, cmp.value, d.better), el('span', null, cmpText || `${cmp.label || '비교 기간'} 대비`));
      fig.append(line);
    }
    card.append(fig);
  }

  function mount(type, draw) {
    const run = () => {
      document.querySelectorAll(`.vzb-card[data-vzb="${type}"]:not([data-vzb-ready])`).forEach((card) => {
        card.setAttribute('data-vzb-ready', '');
        const src = card.querySelector('script[type="application/json"]');
        let d;
        try { d = JSON.parse(src ? src.textContent : ''); } catch (e) {
          card.append(el('p', 'vzb-error', `카드 데이터(JSON)를 읽지 못했습니다: ${e.message}`));
          return;
        }
        card.setAttribute('role', 'group');
        if (d.title) card.setAttribute('aria-label', d.title);
        try { draw(card, d, format(d)); } catch (e) {
          card.append(el('p', 'vzb-error', `카드를 그리지 못했습니다: ${e.message}`));
        }
      });
    };
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
    else run();
  }

  function onSize(node, fn) {
    let lw = 0, lh = 0;
    new ResizeObserver((entries) => {
      const w = Math.floor(entries[0].contentRect.width), h = Math.floor(entries[0].contentRect.height);
      if (w > 0 && h > 0 && (w !== lw || h !== lh)) { lw = w; lh = h; fn(w, h); }
    }).observe(node);
  }

  /* 툴팁과 조작: 마우스는 올리면 보이고 벗어나면 숨깁니다. 터치는 탭하면 보이고 다른 곳을 탭하면 닫힙니다.
     키보드는 그래프에 포커스한 뒤 화살표로 옮깁니다(가로 그래프는 좌우, 목록은 위아래). */

  function showTip(x, y, label, value, side) {

      tip.setAttribute('aria-hidden', 'true');
      document.body.append(tip);
    }

    tip.classList.add('is-on');
    const w = tip.offsetWidth, h = tip.offsetHeight, W = window.innerWidth, H = window.innerHeight;
    if (side) {
      // side가 'left'이면 기준선 왼쪽을 먼저, 그 밖에는 오른쪽을 먼저 씁니다. 자리가 모자라면 반대쪽으로 넘깁니다.
      const right = x + 14 + w <= W - 8, left = x - w - 14 >= 8;

      return;
    }
    tip.style.left = `${Math.min(Math.max(8, x - w / 2), W - w - 8)}px`;
    tip.style.top = `${y - h - 12 < 8 ? y + 16 : y - h - 12}px`;
  }
  function hideTip() { if (tip) tip.classList.remove('is-on'); }
  // 툴팁 내용: { value, note } 또는 { rows: [{ tone, value, name }], note }. 값이 먼저, 이름은 뒤에 옅게 적습니다.
  function tipBody(spec, label) {

      });
      body.append(grid);
    } else {
      body.append(el('span', 'vzb-tip-value', spec.value));
      text += ` ${spec.value}`;
    }
    // note는 한 줄 글 또는 줄마다 나눈 배열입니다.
    (Array.isArray(spec.note) ? spec.note : [spec.note]).filter(Boolean).forEach((line) => {
      body.append(el('span', 'vzb-tip-note', line));
      text += `, ${line}`;
    });
    return { node: body, text };
  }
  // 범례: [{ name, tone, box(면), dot(점), dim(옅게), tick(세로 눈금) }]. 선은 짧은 선, 막대와 면은 네모, 점은 동그라미로 표시합니다.
  function legendEl(items) {
    const ul = el('ul', 'vzb-legend');
    items.forEach((it) => {
      const li = el('li');
      li.append(el('i', `vzb-key vzb-${it.tone}${it.box ? ' is-box' : ''}${it.dot ? ' is-dot' : ''}${it.dim ? ' is-dim' : ''}${it.tick ? ' is-tick' : ''}`), it.name);
      ul.append(li);
    });
    return ul;
  }
  // 글자 폭(12px 기준, size를 주면 그 크기)과 SVG 글자
  let meter;
  function textW(s, weight, family, size) {
    meter = meter || document.createElement('canvas').getContext('2d');
    meter.font = `${weight} ${size || 12}px ${family}`;
    return meter.measureText(s).width;
  }
  function svgText(x, y, s, cls, anchor) {
    const t = svgEl('text', { x, y, class: cls, 'text-anchor': anchor || 'start' });
    t.textContent = s;
    return t;
  }
  // 차이를 부호와 함께 적습니다. 비율 지표는 %p입니다.
  function signed(f, diff) {
    const s = diff > 0 ? '+' : diff < 0 ? '−' : '';
    return f.rel ? `${s}${show(f, Math.abs(diff))}` : `${s}${num(Math.abs(diff), 1)}%p`;
  }
  // pick(clientX, clientY) → 순번, point(순번) → { x, y, label, value, text, side }(화면 좌표), clear() → 강조 해제
  // n은 개수나 개수를 돌려주는 함수입니다. vertical이면 위아래 화살표로 옮깁니다. 돌려주는 { show(순번), at() }로 키를 더 붙입니다.
  // 아무것도 고르지 않은 채 화살표를 누르면 기간 차트는 마지막(최근)부터, vertical이나 first면 첫 순번부터 짚습니다.
  function interact(plot, n, pick, point, clear, vertical, first) {
    let idx = -1, touched = false;
    const count = () => (typeof n === 'function' ? n() : n);
    const live = el('span', 'vzb-sr');
    live.setAttribute('aria-live', 'polite');
    plot.after(live);
    const go = (i) => {
      if (i < 0 || i >= count()) return;
      idx = i;
      const p = point(i);
      showTip(p.x, p.y, p.label, p.value, p.side);
      live.textContent = p.text || `${p.label} ${p.value}`;
    };
    const end = () => {
      if (idx < 0) return;
      idx = -1;
      clear();
      hideTip();
    };
    const [back, next] = vertical ? ['ArrowUp', 'ArrowDown'] : ['ArrowLeft', 'ArrowRight'];
    // pick이 -1이면(조각 밖) 강조를 풉니다.
    const at = (e) => { const i = pick(e.clientX, e.clientY); if (i < 0) end(); else go(i); };
    plot.addEventListener('pointermove', (e) => { if (e.pointerType !== 'touch' || touched) at(e); });
    plot.addEventListener('pointerdown', (e) => { if (e.pointerType === 'touch') { touched = true; at(e); } });
    plot.addEventListener('pointerleave', (e) => { if (e.pointerType !== 'touch') end(); });
    document.addEventListener('pointerdown', (e) => { if (touched && !plot.contains(e.target)) { touched = false; end(); } }, true);
    plot.addEventListener('keydown', (e) => {
      if (e.key !== back && e.key !== next) return;
      e.preventDefault();
      const last = count() - 1;
      go(idx < 0 ? (vertical || first ? 0 : last) : Math.max(0, Math.min(last, idx + (e.key === next ? 1 : -1))));
    });
    plot.addEventListener('blur', end);
    window.addEventListener('scroll', end, { passive: true });
    return { show: go, at: () => idx, end };
  }

  /* 생키 공통: links [{ from, to, value }]만으로 노드와 단을 만듭니다. 들어오는 흐름이 없는 노드에서 시작해 흐름마다 한 단씩 오른쪽으로 놓고,
     나가는 흐름이 없는 노드는 마지막 단에 모읍니다. 노드 순서는 links에 처음 나온 순서이고, 노드 높이는 들어온 양과 나간 양 중 큰 쪽입니다.
     모든 단이 같은 비율(1px당 값)을 써서 띠 두께를 단끼리 견줄 수 있고, 단마다 세로 가운데에 모읍니다. 노드 사이는 높이의 12%(16–32px)만큼 띄웁니다.
     이름과 값은 첫 단은 노드 왼쪽, 나머지 단은 노드 오른쪽에 두 줄로 적고, 겹치면 위아래로 밀어 냅니다.
     띠나 노드(이름 포함)에 마우스를 올리거나, 그래프에 포커스한 뒤 좌우 화살표를 누르면(노드 다음 흐름 순서) 나머지를 옅게 하고 툴팁을 띄웁니다.
     cfg: paint(nodes, links, cols) 노드와 띠의 tone(색 이름)을 정합니다. 띠는 정하지 않으면 출발 노드의 색입니다. */
  function sankeyCard(card, d, f, cfg) {
    const nodes = [], byId = new Map();
    const node = (id) => {
      const key = String(id);
      let n = byId.get(key);
      if (!n) { n = { id: key, name: key, ins: [], outs: [] }; byId.set(key, n); nodes.push(n); }
      return n;
    };
    const links = (d.links || []).filter((l) => l && l.from != null && l.to != null && String(l.from) !== String(l.to) && fin(l.value) && +l.value > 0)
      .map((l) => ({ s: node(l.from), t: node(l.to), value: +l.value }));
    links.forEach((l) => { l.s.outs.push(l); l.t.ins.push(l); });
    const total = (list) => list.reduce((a, x) => a + x.value, 0);
    nodes.forEach((n) => { n.value = Math.max(total(n.ins), total(n.outs)); n.depth = 0; });
    // 단: 출발 노드에서 가장 먼 거리. 흐름이 되돌아오는 고리가 있어도 노드 수만큼만 돌고 멈춥니다.
    for (let r = 0; r < nodes.length; r++) {
      let moved = false;
      links.forEach((l) => { if (l.t.depth < l.s.depth + 1) { l.t.depth = l.s.depth + 1; moved = true; } });
      if (!moved) break;
    }
    const deepest = nodes.length ? Math.max(...nodes.map((n) => n.depth)) : 0;
    const cols = Array.from({ length: deepest + 1 }, (_, c) => nodes.filter((n) => (n.outs.length ? n.depth : deepest) === c)).filter((c) => c.length);
    cols.forEach((c, i) => c.forEach((n) => { n.col = i; }));
    const L = cols.length - 1;
    if (cfg.paint) cfg.paint(nodes, links, cols);
    nodes.forEach((n) => { n.tone = n.tone || 'f1'; });
    links.forEach((l) => { l.tone = l.tone || l.s.tone; });
    // 짚는 순서: 노드(단 순서, 위에서 아래로), 이어서 흐름(출발 노드, 도착 노드 순서)
    const order = [];
    cols.forEach((c) => c.forEach((n) => { n.k = order.length; order.push({ node: n }); }));
    const at = (n) => n.col * 1e4 + cols[n.col].indexOf(n);
    links.slice().sort((a, b) => at(a.s) - at(b.s) || at(a.t) - at(b.t)).forEach((l) => { l.k = order.length; order.push({ link: l }); });

    head(card, d, f, null);
    const foot = el('div', 'vzb-foot vzb-fill vzb-sk');
    const plot = el('div', 'vzb-plot');
    plot.tabIndex = 0;
    plot.setAttribute('role', 'img');
    plot.setAttribute('aria-label', `${d.title || ''} 생키 차트, 노드 ${nodes.length}개와 흐름 ${links.length}개. 좌우 화살표로 노드와 흐름을 차례로 짚습니다. 모든 흐름은 이어지는 표에 있습니다.`);
    const svg = svgEl('svg', { class: 'vzb-sk-svg', 'aria-hidden': 'true' });
    plot.append(svg);
    foot.append(plot);
    card.append(foot);
    // 화면 낭독기용 흐름 표
    const sr = el('div', 'vzb-sr'), table = el('table'), hr = el('tr');
    hr.append(el('th', null, '출발'), el('th', null, '도착'), el('th', null, '값'));
    table.append(el('caption', null, d.title || ''), hr);
    links.forEach((l) => { const tr = el('tr'); tr.append(el('td', null, l.s.name), el('td', null, l.t.name), el('td', null, show(f, l.value))); table.append(tr); });
    sr.append(table);
    card.append(sr);

    const fam = getComputedStyle(card).fontFamily;
    const NW = 8, LH = 32;   // 노드 폭, 이름표(이름 · 값 두 줄) 높이
    const fx = (v) => v.toFixed(1);
    const pct = (a, b) => `${num(b ? (a / b) * 100 : 0, 1)}%`;
    let lastW = 0, lastH = 0;
    function draw(w, h) {
      lastW = w; lastH = h;
      svg.setAttribute('width', w);
      svg.setAttribute('height', h);
      svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
      svg.replaceChildren();
      if (!links.length) return;
      // 이름표 폭: 이름(12px)과 값(13px 굵게, 단위 11px) 중 넓은 쪽
      nodes.forEach((n) => {
        const [t, u] = f.parts(n.value);
        n.num = t; n.unit = u;
        n.lw = Math.max(textW(n.name, 400, fam), textW(t, 700, fam, 13) + (u ? textW(u, 400, fam, 11) + 2 : 0));
      });
      const leftW = Math.ceil(Math.max(...cols[0].map((n) => n.lw))) + 10;
      const rightW = L ? Math.ceil(Math.max(...cols[L].map((n) => n.lw))) + 10 : 0;
      const xa = leftW, xb = Math.max(xa, w - rightW - NW);
      const gap = Math.round(Math.max(16, Math.min(32, h * 0.12)));
      const ky = Math.min(...cols.map((c) => (h - (c.length - 1) * gap) / Math.max(1e-9, total(c))));
      cols.forEach((c, ci) => {
        const tall = c.reduce((a, n) => a + Math.max(1, n.value * ky), 0) + (c.length - 1) * gap;
        let y = (h - tall) / 2;
        c.forEach((n) => { n.x = L ? xa + (ci / L) * (xb - xa) : xa; n.y0 = y; n.h = Math.max(1, n.value * ky); n.y1 = y + n.h; y = n.y1 + gap; });
      });
      // 노드 안에서 띠는 맞은편 노드가 위에 있는 순서로 쌓아 엇갈림을 줄입니다.
      nodes.forEach((n) => {
        n.outs.sort((a, b) => a.t.y0 - b.t.y0);
        n.ins.sort((a, b) => a.s.y0 - b.s.y0);
        let y = n.y0;
        n.outs.forEach((l) => { l.w = l.value * ky; l.sy = y; y += l.w; });
        y = n.y0;
        n.ins.forEach((l) => { l.ty = y; y += l.value * ky; });
      });
      const ribbon = (l) => {
        const x0 = l.s.x + NW, x1 = l.t.x, xm = (x0 + x1) / 2, a = l.sy, b = l.ty, lw = l.w;
        return `M${fx(x0)},${fx(a)}C${fx(xm)},${fx(a)} ${fx(xm)},${fx(b)} ${fx(x1)},${fx(b)}V${fx(b + lw)}C${fx(xm)},${fx(b + lw)} ${fx(xm)},${fx(a + lw)} ${fx(x0)},${fx(a + lw)}Z`;
      };
      const linkG = svgEl('g', {}), nodeG = svgEl('g', {}), labG = svgEl('g', {});
      // 굵은 띠를 먼저 그려 가는 띠가 위에 오게 합니다(가는 띠도 짚을 수 있게).
      links.slice().sort((a, b) => b.value - a.value).forEach((l) => {
        l.el = svgEl('path', { d: ribbon(l), class: `vzb-sk-link vzb-${l.tone}`, 'data-k': l.k });
        linkG.append(l.el);
      });
      nodes.forEach((n) => {
        n.el = svgEl('rect', { x: fx(n.x), y: fx(n.y0), width: NW, height: fx(n.h), rx: 2, class: `vzb-sk-node vzb-${n.tone}`, 'data-k': n.k });
        nodeG.append(n.el);
      });
      // 이름표: 노드 가운데에 맞추고, 같은 단에서 겹치면 위에서 아래로, 바닥을 넘으면 아래에서 위로 밀어 냅니다.
      cols.forEach((c) => {
        c.forEach((n, i) => { n.ly = Math.max((n.y0 + n.y1) / 2, LH / 2, i ? c[i - 1].ly + LH : -Infinity); });
        for (let i = c.length - 1; i >= 0; i--) c[i].ly = Math.min(c[i].ly, h - LH / 2, i < c.length - 1 ? c[i + 1].ly - LH : Infinity);
      });
      nodes.forEach((n) => {
        const left = n.col === 0 && L > 0, x = left ? n.x - 8 : n.x + NW + 8, anchor = left ? 'end' : 'start';
        const g = svgEl('g', { class: `vzb-sk-lab${n.col > 0 && n.col < L ? ' is-over' : ''}`, 'data-k': n.k });
        const v = svgText(x, n.ly + 13, n.num, 'vzb-sk-val', anchor);
        if (n.unit) { const u = svgEl('tspan', { class: 'vzb-sk-unit', dx: 2 }); u.textContent = n.unit; v.append(u); }
        g.append(svgText(x, n.ly - 3, n.name, 'vzb-sk-name', anchor), v);
        labG.append(g);
        n.lab = g;
      });
      svg.append(linkG, nodeG, labG);
    }
    onSize(plot, draw);
    if (document.fonts && document.fonts.ready) document.fonts.ready.then(() => { if (lastW) draw(lastW, lastH); });
    interact(plot, order.length,
      (cx, cy) => {
        const e = document.elementFromPoint(cx, cy), hit = e && e.closest ? e.closest('[data-k]') : null;
        return hit && svg.contains(hit) ? +hit.getAttribute('data-k') : -1;
      },
      (k) => {
        const it = order[k], onN = new Set(), onL = new Set();
        if (it.node) { onN.add(it.node); it.node.ins.concat(it.node.outs).forEach((l) => { onL.add(l); onN.add(l.s); onN.add(l.t); }); }
        else { onL.add(it.link); onN.add(it.link.s); onN.add(it.link.t); }
        svg.classList.add('is-hover');
        nodes.forEach((n) => { if (n.el) { n.el.classList.toggle('is-on', onN.has(n)); n.lab.classList.toggle('is-on', onN.has(n)); } });
        links.forEach((l) => { if (l.el) l.el.classList.toggle('is-on', onL.has(l)); });
        const r = plot.getBoundingClientRect();
        if (it.link) {
          const l = it.link, notes = [];
          if (l.value < l.s.value) notes.push(`${l.s.name}의 ${pct(l.value, l.s.value)}`);
          if (l.value < l.t.value) notes.push(`${l.t.name}의 ${pct(l.value, l.t.value)}`);
          const label = `${l.s.name} → ${l.t.name}`, t = tipBody({ value: show(f, l.value), note: notes }, label);
          return { x: r.left + (l.s.x + NW + l.t.x) / 2, y: r.top + (l.sy + l.ty) / 2 - 4, label, value: t.node, text: t.text };
        }
        // 노드: 마지막 단은 어디서 왔는지, 나머지 단은 어디로 가는지를 큰 순서로 다섯 줄까지 적습니다.
        const n = it.node, end = L > 0 && n.col === L;
        const list = (end ? n.ins : n.outs).slice().sort((a, b) => b.value - a.value);
        const notes = [], colSum = total(cols[n.col]);
        if (list.length > 5) notes.push(`외 ${list.length - 5}곳 ${show(f, total(list.slice(5)))}`);
        if (list.length > 1) notes.push(`합계 ${show(f, n.value)}`);
        if (end && n.ins.length === 1) { if (n.value < n.ins[0].s.value) notes.push(`${n.ins[0].s.name}의 ${pct(n.value, n.ins[0].s.value)}`); }
        else if (n.value < colSum) notes.push(`전체의 ${pct(n.value, colSum)}`);
        const spec = list.length > 1
          ? { rows: list.slice(0, 5).map((l) => ({ tone: l.tone, value: show(f, l.value), name: end ? l.s.name : l.t.name })), note: notes }
          : { value: show(f, n.value), note: notes };
        const t = tipBody(spec, n.name);
        return { x: r.left + (end ? n.x : n.x + NW), y: r.top + Math.max(0, n.y0), label: n.name, value: t.node, text: t.text, side: end ? 'left' : true };
      },
      () => { svg.classList.remove('is-hover'); svg.querySelectorAll('.is-on').forEach((x) => x.classList.remove('is-on')); },
      false, true);
  }

  // Sankey - Levels: 세 단 이상으로 이어지는 흐름에서 마지막 단의 한 노드(focus)를 짚습니다. focus와 focus로 가는 띠는 파랑,
  // 마지막 단의 나머지 노드로 가는 띠는 회색, 앞 단의 띠는 주황으로 칠해, 들어온 양 가운데 얼마가 focus까지 남는지 보이게 합니다.
  // 앞 단과 focus의 색을 달리해 앞 단의 특정 노드가 focus로 바로 흘러가는 것처럼 읽히지 않게 합니다. focus를 비우면 모두 주황입니다.
  mount('sankey-levels', (card, d, f) => {
    sankeyCard(card, d, f, {
      paint(nodes, links, cols) {
        const end = cols[cols.length - 1], focus = d.focus == null ? null : String(d.focus);
        nodes.forEach((n) => { n.tone = focus == null || !end.includes(n) ? 'f1' : n.id === focus ? 't3' : 'f0'; });
        links.forEach((l) => { l.tone = l.t.tone; });
      },
    });
  });
})();
