  'use strict';
  /* 공통 도구: 숫자 표기, 증감, 카드 머리, 그리기 시작 */
  const SVGNS = 'http://www.w3.org/2000/svg';
  function el(tag, cls, text) {
    const n = document.createElement(tag);
    if (cls) n.className = cls;
    if (text != null) n.textContent = text;
    return n;
  }
  function svgEl(tag, attrs) {
    const n = document.createElementNS(SVGNS, tag);
    for (const k in attrs) n.setAttribute(k, attrs[k]);
    return n;
  }
  const nfs = {};
  function num(v, d) {
    nfs[d] = nfs[d] || new Intl.NumberFormat('ko-KR', { minimumFractionDigits: d, maximumFractionDigits: d });
    return nfs[d].format(v);
  }
  function krw(v) {
    const a = Math.abs(v);
    if (a >= 1e12) return [num(v / 1e12, 1), '조'];
    if (a >= 1e8) return [num(v / 1e8, a >= 1e10 ? 0 : 1), '억'];
    if (a >= 1e4) return [num(v / 1e4, 0), '만'];
    return [num(v, 0), '원'];
  }
  // 달러 금액: 원래 값을 받아 K · M · B로 줄이고 소수 한 자리로 맞춥니다($745.6K). 접미사는 단위처럼 작게 씁니다.
  function usd(v, d) {
    const a = Math.abs(v), s = v < 0 ? '−' : '';
    const [div, u] = a >= 1e9 ? [1e9, 'B'] : a >= 1e6 ? [1e6, 'M'] : a >= 1e3 ? [1e3, 'K'] : [1, ''];
    return [`${s}$${num(a / div, d != null ? d : div > 1 ? 1 : 0)}`, u];
  }
  // format: eok(억 단위 값) · krw(원 단위 값) · usd(달러 값) · count(unit으로 단위) · pct(증감을 %p로) · days
  function format(d) {
    const dec = (v, fallback) => (d.decimals != null ? d.decimals : fallback);
    switch (d.format) {
      case 'eok': return { parts: (v) => [num(v, dec(v, Number.isInteger(v) || Math.abs(v) >= 100 ? 0 : 1)), '억'], rel: true };
      case 'krw': return { parts: krw, rel: true };
      case 'usd': return { parts: (v) => usd(v, d.decimals), rel: true, usd: true };
      case 'pct': return { parts: (v) => [num(v, dec(v, 1)), '%'], rel: false };
      case 'days': return { parts: (v) => [num(v, dec(v, 1)), '일'], rel: true };
      default: return { parts: (v) => [num(v, dec(v, Number.isInteger(v) ? 0 : 1)), d.unit || ''], rel: true };

  const show = (f, v) => f.parts(v).join('');

  function delta(f, value, base, better) {

    const dir = Math.abs(diff) < 1e-9 ? 0 : Math.sign(diff);
    const want = better === 'down' ? -1 : better === 'neutral' ? 0 : 1;
    const tone = dir === 0 || want === 0 ? 'flat' : dir === want ? 'good' : 'bad';
    let text;
    if (!f.rel) text = `${num(Math.abs(diff), 1)}%p`;
    else if (base) text = `${num(Math.abs(diff / base) * 100, 1)}%`;
    else text = show(f, Math.abs(diff));
    return { dir, tone, text };
  }
  function arrow(dir) {
    const s = svgEl('svg', { class: 'vzb-arrow', viewBox: '0 0 12 12', 'aria-hidden': 'true' });
    s.append(svgEl('path', { d: dir > 0 ? 'M3.5 8.5l5-5M4.5 3.5h4v4' : 'M3.5 3.5l5 5M8.5 4.5v4h-4' }));
    return s;
  }
  // 대표 수치(단위는 작게)
  function figure(f, value) {
    const n = el('div', 'vzb-num');
    const [text, unit] = f.parts(value);
    n.append(text);
    if (unit) n.append(el('span', 'vzb-unit', unit));
    return n;
  }
  // 증감 배지: 화살표는 오르내림, 색은 좋고 나쁨(better)입니다.

    b.append(dl.text);
    return b;
  }
  // 머리: 제목 · 기간 부제(subtitle) · 대표 수치와 바로 뒤의 증감 배지. value가 null이면 제목만 둡니다.
  function head(card, d, f, value, cmp, cmpText) {

    hd.append(el('div', 'vzb-title', d.title || ''));
    if (d.subtitle) hd.append(el('div', 'vzb-sub', d.subtitle));

      fig.append(line);
    }

    const run = () => {
      document.querySelectorAll(`.vzb-card[data-vzb="${type}"]:not([data-vzb-ready])`).forEach((card) => {
        card.setAttribute('data-vzb-ready', '');
        const src = card.querySelector('script[type="application/json"]');
        let d;
        try { d = JSON.parse(src ? src.textContent : ''); } catch (e) {
          card.append(el('p', 'vzb-error', `카드 데이터(JSON)를 읽지 못했습니다: ${e.message}`));
          return;
        }
        card.setAttribute('role', 'group');
        if (d.title) card.setAttribute('aria-label', d.title);
        try { draw(card, d, format(d)); } catch (e) {
          card.append(el('p', 'vzb-error', `카드를 그리지 못했습니다: ${e.message}`));
        }
      });
    };
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
    else run();
  }
  // 크기가 바뀔 때마다 그래픽을 다시 그립니다. 차트는 카드에 남은 높이를 채웁니다.
  function onSize(node, fn) {
    let lw = 0, lh = 0;
    new ResizeObserver((entries) => {
      const w = Math.floor(entries[0].contentRect.width), h = Math.floor(entries[0].contentRect.height);
      if (w > 0 && h > 0 && (w !== lw || h !== lh)) { lw = w; lh = h; fn(w, h); }
    }).observe(node);
  }

  /* 툴팁과 조작: 마우스는 올리면 보이고 벗어나면 숨깁니다. 터치는 탭하면 보이고 다른 곳을 탭하면 닫힙니다.
     키보드는 그래프에 포커스한 뒤 화살표로 옮깁니다(가로 그래프는 좌우, 목록은 위아래). */
  let tip;
  // value는 글자나 요소입니다. side가 참이면 점 위가 아니라 세로선 옆, 그래프 윗선에 맞춰 둡니다(선 · 막대 차트).
  function showTip(x, y, label, value, side) {
    if (!tip) {
      tip = el('div', 'vzb-tip');
      tip.setAttribute('aria-hidden', 'true');
      document.body.append(tip);
    }
    tip.replaceChildren(el('span', 'vzb-tip-label', label), typeof value === 'string' ? el('span', 'vzb-tip-value', value) : value);
    tip.classList.add('is-on');
    const w = tip.offsetWidth, h = tip.offsetHeight, W = window.innerWidth, H = window.innerHeight;
    if (side) {
      // side가 'left'이면 기준선 왼쪽을 먼저, 그 밖에는 오른쪽을 먼저 씁니다. 자리가 모자라면 반대쪽으로 넘깁니다.
      const right = x + 14 + w <= W - 8, left = x - w - 14 >= 8;
      tip.style.left = `${(side === 'left' ? left || !right : !right) ? Math.max(8, x - w - 14) : x + 14}px`;
      tip.style.top = `${Math.min(Math.max(8, y), H - h - 8)}px`;
      return;
    }
    tip.style.left = `${Math.min(Math.max(8, x - w / 2), W - w - 8)}px`;
    tip.style.top = `${y - h - 12 < 8 ? y + 16 : y - h - 12}px`;
  }
  function hideTip() { if (tip) tip.classList.remove('is-on'); }
  // 툴팁 내용: { value, note } 또는 { rows: [{ tone, value, name }], note }. 값이 먼저, 이름은 뒤에 옅게 적습니다.
  function tipBody(spec, label) {
    const body = el('span', 'vzb-tip-body');
    let text = label;
    if (spec.rows) {
      const grid = el('span', 'vzb-tip-rows');
      spec.rows.forEach((r) => {
        const row = el('span', 'vzb-tip-row');
        row.append(el('i', `vzb-tip-key vzb-${r.tone}`), el('b', null, r.value), el('span', null, r.name));
        grid.append(row);
        text += `, ${r.name} ${r.value}`;
      });
      body.append(grid);
    } else {
      body.append(el('span', 'vzb-tip-value', spec.value));
      text += ` ${spec.value}`;
    }
    // note는 한 줄 글 또는 줄마다 나눈 배열입니다.
    (Array.isArray(spec.note) ? spec.note : [spec.note]).filter(Boolean).forEach((line) => {
      body.append(el('span', 'vzb-tip-note', line));
      text += `, ${line}`;
    });
    return { node: body, text };
  }
  // 범례: [{ name, tone, box(면), dot(점), dim(옅게), tick(세로 눈금) }]. 선은 짧은 선, 막대와 면은 네모, 점은 동그라미로 표시합니다.
  function legendEl(items) {
    const ul = el('ul', 'vzb-legend');
    items.forEach((it) => {
      const li = el('li');
      li.append(el('i', `vzb-key vzb-${it.tone}${it.box ? ' is-box' : ''}${it.dot ? ' is-dot' : ''}${it.dim ? ' is-dim' : ''}${it.tick ? ' is-tick' : ''}`), it.name);
      ul.append(li);
    });
    return ul;
  }
  // 글자 폭(12px 기준, size를 주면 그 크기)과 SVG 글자
  let meter;
  function textW(s, weight, family, size) {
    meter = meter || document.createElement('canvas').getContext('2d');
    meter.font = `${weight} ${size || 12}px ${family}`;
    return meter.measureText(s).width;
  }
  function svgText(x, y, s, cls, anchor) {
    const t = svgEl('text', { x, y, class: cls, 'text-anchor': anchor || 'start' });
    t.textContent = s;
    return t;
  }
  // 차이를 부호와 함께 적습니다. 비율 지표는 %p입니다.
  function signed(f, diff) {
    const s = diff > 0 ? '+' : diff < 0 ? '−' : '';
    return f.rel ? `${s}${show(f, Math.abs(diff))}` : `${s}${num(Math.abs(diff), 1)}%p`;
  }
  // pick(clientX, clientY) → 순번, point(순번) → { x, y, label, value, text, side }(화면 좌표), clear() → 강조 해제
  // n은 개수나 개수를 돌려주는 함수입니다. vertical이면 위아래 화살표로 옮깁니다. 돌려주는 { show(순번), at() }로 키를 더 붙입니다.
  // 아무것도 고르지 않은 채 화살표를 누르면 기간 차트는 마지막(최근)부터, vertical이나 first면 첫 순번부터 짚습니다.
  function interact(plot, n, pick, point, clear, vertical, first) {
    let idx = -1, touched = false;
    const count = () => (typeof n === 'function' ? n() : n);
    const live = el('span', 'vzb-sr');
    live.setAttribute('aria-live', 'polite');
    plot.after(live);
    const go = (i) => {
      if (i < 0 || i >= count()) return;
      idx = i;
      const p = point(i);
      showTip(p.x, p.y, p.label, p.value, p.side);
      live.textContent = p.text || `${p.label} ${p.value}`;
    };
    const end = () => {
      if (idx < 0) return;
      idx = -1;
      clear();
      hideTip();
    };
    const [back, next] = vertical ? ['ArrowUp', 'ArrowDown'] : ['ArrowLeft', 'ArrowRight'];
    // pick이 -1이면(조각 밖) 강조를 풉니다.
    const at = (e) => { const i = pick(e.clientX, e.clientY); if (i < 0) end(); else go(i); };
    plot.addEventListener('pointermove', (e) => { if (e.pointerType !== 'touch' || touched) at(e); });
    plot.addEventListener('pointerdown', (e) => { if (e.pointerType === 'touch') { touched = true; at(e); } });
    plot.addEventListener('pointerleave', (e) => { if (e.pointerType !== 'touch') end(); });
    document.addEventListener('pointerdown', (e) => { if (touched && !plot.contains(e.target)) { touched = false; end(); } }, true);
    plot.addEventListener('keydown', (e) => {
      if (e.key !== back && e.key !== next) return;
      e.preventDefault();
      const last = count() - 1;
      go(idx < 0 ? (vertical || first ? 0 : last) : Math.max(0, Math.min(last, idx + (e.key === next ? 1 : -1))));
    });
    plot.addEventListener('blur', end);
    window.addEventListener('scroll', end, { passive: true });
    return { show: go, at: () => idx, end };
  }

  /* 축 차트 공통(라인 · 막대): 머리와 범례, 세로축 눈금 3–4개(단위는 맨 위 눈금에만)와 옅은 실선 격자, 다섯 개 안팎으로 솎아 낸 기간 이름,
     오른쪽 끝 이름, 툴팁 내용, 화면 낭독기용 값 표. 선과 막대 모양은 각자의 그리기(paint)가 맡습니다. */

  // 세로축 눈금: 1 · 2 · 2.5 · 5 단위의 끝수 없는 값으로 범위를 감쌉니다(눈금이 3–4개뿐이라 2.5 단위를 더해 범위가 덜 헐거워지게 합니다).
  // 높이에 맞는 눈금 수(3개부터 most개까지) 가운데 값이 높이를 가장 많이 쓰는 간격을 고릅니다.
  function yTicks(min, max, most) {
    if (min === max) { const pad = Math.abs(min) * 0.1 || 1; min -= pad; max += pad; }
    let best = null;
    const e0 = Math.floor(Math.log10((max - min) / most)) - 1;
    for (let e = e0; e <= e0 + 3; e++) {
      for (const m of [1, 2, 2.5, 5]) {
        const step = m * Math.pow(10, e);
        const lo = Math.floor(min / step + 1e-9) * step, hi = Math.ceil(max / step - 1e-9) * step;
        const n = Math.round((hi - lo) / step) + 1;
        if (n < 3 || n > most) continue;
        const score = (max - min) / (hi - lo) - 0.04 * n;
        if (!best || score > best.score) best = { lo, hi, step, n, score };
      }
    }
    return best || { lo: min, hi: max, step: max - min, n: 2 };
  }
  // 눈금 간격에 필요한 소수 자릿수(2.5 단위는 한 자리 더)
  function stepDec(step) {
    for (let k = 0; k < 6; k++) { const v = step * Math.pow(10, k); if (Math.abs(v - Math.round(v)) < 1e-6) return k; }
    return 6;
  }
  // 눈금 글자: 단위를 붙이되 짧게 적습니다. 원 단위 값은 가장 큰 눈금에 맞춘 한 단위(조 · 억 · 만)로 통일하고, 음수는 빼기 기호(−)로 적습니다.
  function tickText(d, dec, step, top) {
    const t = tickRaw(d, dec, step, top);
    return (v) => t(v).replace(/^-/, '−');
  }
  function tickRaw(d, dec, step, top) {
    switch (d.format) {
      case 'eok': return (v) => `${num(v, dec)}억`;
      case 'krw': {
        const [div, u] = top >= 1e12 ? [1e12, '조'] : top >= 1e8 ? [1e8, '억'] : top >= 1e4 ? [1e4, '만'] : [1, '원'];
        const dd = Math.max(0, -Math.floor(Math.log10(step / div) + 1e-9));
        return (v) => `${num(v / div, dd)}${u}`;
      }
      case 'usd': {
        // 달러는 가장 큰 눈금에 맞춘 한 단위(B · M · K)로 통일합니다($150K).
        const [div, u] = top >= 1e9 ? [1e9, 'B'] : top >= 1e6 ? [1e6, 'M'] : top >= 1e3 ? [1e3, 'K'] : [1, ''];
        const dd = Math.max(0, -Math.floor(Math.log10(step / div) + 1e-9));
        return (v) => `${v < 0 ? '-' : ''}$${num(Math.abs(v) / div, dd)}${u}`;
      }
      case 'pct': return (v) => `${num(v, dec)}%`;
      case 'days': return (v) => `${num(v, dec)}일`;
      default: {
        // 개수는 unit을 붙이고, 10만이 넘으면 만 단위로 줄여 적습니다(150,000건 → 15만건).
        const u = d.unit || '';
        if (top < 1e5) return (v) => `${num(v, dec)}${u}`;
        const dd = stepDec(step / 1e4);
        return (v) => `${num(v / 1e4, dd)}만${u}`;
      }
    }
  }
  function lastIdx(vals, n) {
    for (let i = Math.min(vals.length, n) - 1; i >= 0; i--) if (vals[i] != null) return i;
    return -1;
  }

  // cfg: series [{ name, values, tone }] — tone은 색 이름(t1 · t2 · t3 선, f1 · f2 · f3 막대, t0 · f0 비교 회색 등)입니다.
  //   paint { gutter(S, c), draw(svg, c, S) → 선 끝 글자의 높이 목록, mark(i, c, S, back, front) } · band(기간마다 칸)
  //   legend(범례, legendEl 참고) · top(머리 아래 요소) · extra(세로축에 넣을 값) · zero(0을 축에 넣기)
  //   stack(쌓은 합으로 범위) · headroom(값 범위 위아래 여유, 범위에 대한 비율) · padTop · side [{ value, lines }](오른쪽 끝 이름) · under, over(svg, c) 덧그림
  //   tip(순번) → { value, note } 또는 { rows, note } · sort(툴팁을 값이 큰 순서로) · f, fmt(값과 눈금 표기)
  // 돌려주는 show(cfg 일부)로 같은 카드에 다른 지표를 다시 그립니다(탭 카드).
  function axisCard(card, d, f, cfg) {
    const labels = d.labels || [], n = labels.length;
    head(card, d, f, null);
    if (cfg.top) card.append(cfg.top);
    if (cfg.legend) card.append(legendEl(cfg.legend));
    const foot = el('div', 'vzb-foot vzb-fill vzb-ax');
    const plot = el('div', 'vzb-plot');
    plot.tabIndex = 0;
    plot.setAttribute('role', 'img');
    const svg = svgEl('svg', { class: 'vzb-ax-svg', 'aria-hidden': 'true' });
    plot.append(svg);
    foot.append(plot);
    card.append(foot);
    const sr = el('div', 'vzb-sr');
    card.append(sr);
    const fam = getComputedStyle(card).fontFamily;
    const c = { labels, n, fam };
    let S, back, front, lastW = 0, lastH = 0;

    // 그릴 지표를 정합니다. 값 범위, 마지막 기간, 화면 낭독기용 표를 다시 만듭니다.
    function use(next) {
      S = Object.assign({ f, fmt: d, curve: d.curve }, cfg, next);
      const vals = [];
      S.series.forEach((s) => s.values.slice(0, n).forEach((v) => { if (v != null && isFinite(v)) vals.push(+v); }));
      if (S.stack) {
        for (let i = 0; i < n; i++) {
          let pos = 0, neg = 0;
          S.series.forEach((s) => { const v = s.values[i]; if (v > 0) pos += +v; else if (v < 0) neg += +v; });
          vals.push(pos, neg);
        }
      }
      (S.extra || []).forEach((v) => { if (v != null && isFinite(v)) vals.push(+v); });
      if (S.zero) vals.push(0);
      if (!n || !vals.length) throw new Error('labels와 values(또는 series)를 확인하세요.');
      S.vmin = Math.min(...vals);
      S.vmax = Math.max(...vals);
      // 선 차트는 값 범위 위아래에 여유를 두어 선이 그래프 가운데에 머물게 합니다. 0 이상인 값은 0 아래로 내려가지 않습니다.
      if (S.headroom && S.vmax > S.vmin) {
        const pad = (S.vmax - S.vmin) * S.headroom, lo = S.vmin - pad;
        S.vmin = S.vmin >= 0 && lo < 0 ? 0 : lo;
        S.vmax += pad;
      }
      const first = S.series[0] ? lastIdx(S.series[0].values, n) : -1;   // 첫 선(현재)의 마지막 기간
      S.latest = first >= 0 ? first : Math.max(-1, ...S.series.map((s) => lastIdx(s.values, n)));
      plot.setAttribute('aria-label', `${S.name || d.title || ''} ${S.kind || '차트'}, ${labels[0] || ''}–${labels[n - 1] || ''}. 좌우 화살표로 기간을 옮깁니다. 모든 값은 이어지는 표에 있습니다.`);
      const table = el('table');
      const hr = el('tr');
      hr.append(el('th', null, '기간'));
      S.series.forEach((s) => hr.append(el('th', null, s.name)));
      table.append(el('caption', null, S.name || d.title || ''), hr);
      labels.forEach((l, i) => {
        const tr = el('tr');
        tr.append(el('th', null, l));
        S.series.forEach((s) => tr.append(el('td', null, s.values[i] == null ? '' : show(S.f, s.values[i]))));
        table.append(tr);
      });
      sr.replaceChildren(table);
    }
    use({});

    function draw(w, h) {
      lastW = w; lastH = h;
      const top = S.padTop || 8;
      const sides = (S.side || []).filter((m) => isFinite(m.value));
      const ends = S.paint.gutter(S, c);
      const endW = Math.max(0, ...ends.map((s) => textW(s, 500, fam)), ...sides.map((m) => Math.max(...m.lines.map((l) => textW(l, 500, fam)))));
      const labW = Math.max(...labels.map((l) => textW(l, 400, fam)));
      // 눈금과 가로 배치: 바닥(base)에 따라 눈금 수와 눈금 글자 폭이 바뀌므로 함께 계산합니다.
      const layout = (base) => {
        const t = yTicks(S.vmin, S.vmax, Math.max(3, Math.min(4, Math.floor((base - top) / 28) + 1)));
        const dec = stepDec(t.step), ticks = [];
        for (let k = 0; k < t.n; k++) ticks.push(+(t.lo + k * t.step).toFixed(dec));
        const tt = tickText(S.fmt, dec, t.step, Math.max(...ticks.map(Math.abs)));
        // 단위(억 · % · 만 · $ 등)는 맨 위 눈금에만 붙이고, 나머지 눈금은 숫자만 적습니다.
        const tickLabel = ticks.map((v, k) => (k === ticks.length - 1 ? tt(v) : tt(v).replace(/[^\d.,−]+$/, '').replace(/^(−?)\$/, '$1')));
        const gut = Math.ceil(Math.max(...tickLabel.map((s) => textW(s, 400, fam)))) + 10;
        const x0 = gut + 4, right = w - Math.max(endW ? endW + 14 : 0, S.band ? 4 : labW / 2 + 1, 8);
        const step = S.band ? (right - x0) / Math.max(1, n) : (right - x0) / Math.max(1, n - 1);
        return { base, t, ticks, tickLabel, gut, x0, right, step };
      };
      // 항목 축(categorical)의 이름이 한 줄에 다 들어가지 않으면 솎아 내지 않고 두 줄로 엇갈려 적습니다(바닥을 14px 올립니다).
      let L = layout(h - 24);
      const stagger = !!S.categorical && n > 1 && labW + 6 > L.step;
      if (stagger) L = layout(h - 38);
      const { base, t, ticks, tickLabel, gut, x0, right } = L, bottom = base;
      c.step = L.step;
      c.xs = labels.map((_, i) => (S.band ? x0 + c.step * (i + 0.5) : n < 2 ? (x0 + right) / 2 : x0 + i * c.step));
      c.y = (v) => bottom - ((v - t.lo) / (t.hi - t.lo)) * (bottom - top);
      Object.assign(c, { w, h, top, bottom, left: x0, right });
      svg.setAttribute('width', w);
      svg.setAttribute('height', h);
      svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
      svg.replaceChildren();
      ticks.forEach((v, k) => {
        const y = Math.round(c.y(v)) + 0.5;
        svg.append(svgEl('line', { x1: gut, x2: right, y1: y, y2: y, class: v === 0 ? 'vzb-ax-zero' : 'vzb-ax-grid' }), svgText(gut - 8, y + 4, tickLabel[k], 'vzb-ax-label', 'end'));
      });
      // 기간 이름은 겹치지 않고 다섯 개 안팎이 되도록 솎아 내되, 마지막 값이 있는 기간은 항상 남깁니다. 항목 축(categorical)은 모두 적고 굵게 하지 않습니다.
      const every = stagger ? 1 : Math.max(1, Math.ceil((labW + (S.categorical ? 6 : 16)) / Math.max(1, c.step)), S.categorical ? 1 : Math.ceil((n - 1) / 4));
      const anchor = S.categorical ? 0 : S.latest >= 0 ? S.latest : n - 1;
      labels.forEach((l, i) => {
        if ((((anchor - i) % every) + every) % every) return;
        svg.append(svgText(c.xs[i], base + 18 + (stagger && i % 2 ? 14 : 0), l, !S.categorical && i === anchor ? 'vzb-ax-label is-last' : 'vzb-ax-label', 'middle'));
      });
      if (S.under) S.under(svg, c);
      back = svgEl('g', {});
      svg.append(back);
      const taken = S.paint.draw(svg, c, S) || [];
      // 오른쪽 끝 이름은 선 높이를 가운데로 두고, 선 끝 값과 겹치면 그 값의 반대쪽으로 비켜 적습니다.
      sides.forEach((m) => {
        const y = c.y(m.value), lh = 14;
        let first = y + 4 - ((m.lines.length - 1) * lh) / 2;
        const span = () => [first - 11, first + (m.lines.length - 1) * lh + 3];
        taken.forEach((ty) => {
          const [a, b] = span();
          if (ty + 8 <= a || ty - 8 >= b) return;
          first += ty < y ? ty + 8 - a : ty - 8 - b;
        });
        m.lines.forEach((l, j) => svg.append(svgText(right + 10, first + j * lh, l, j ? 'vzb-ax-end vzb-ax-mute' : 'vzb-ax-end')));
      });
      if (S.over) S.over(svg, c);
      front = svgEl('g', {});
      svg.append(front);
    }
    onSize(plot, draw);
    if (document.fonts && document.fonts.ready) document.fonts.ready.then(() => { if (lastW) draw(lastW, lastH); });

    function defaultTip(i) {
      const series = S.series;
      if (series.length === 1) return { value: series[0].values[i] == null ? '–' : show(S.f, series[0].values[i]) };
      const rows = series.filter((s) => s.values[i] != null).map((s) => ({ tone: s.tone, name: s.name, value: show(S.f, s.values[i]), v: s.values[i] }));
      if (S.sort) rows.sort((a, b) => b.v - a.v);
      return { rows };
    }
    function point(i) {
      back.replaceChildren();
      front.replaceChildren();
      S.paint.mark(i, c, S, back, front);
      const t = tipBody(S.tip ? S.tip(i) : defaultTip(i), labels[i]);
      const r = plot.getBoundingClientRect();
      return { x: r.left + c.xs[i], y: r.top + c.top, label: labels[i], value: t.node, text: t.text, side: true };
    }
    interact(plot, n,
      (cx) => {
        const x = cx - plot.getBoundingClientRect().left;
        let best = 0;
        c.xs.forEach((v, i) => { if (Math.abs(v - x) < Math.abs(c.xs[best] - x)) best = i; });
        return best;
      },
      point,
      () => { if (back) back.replaceChildren(); if (front) front.replaceChildren(); });
    return {
      show(next) {
        use(next);
        if (back) back.replaceChildren();
        if (front) front.replaceChildren();
        hideTip();
        if (lastW) draw(lastW, lastH);
      },
    };
  }

  /* 스캐터 공통: 두 측정값을 가로 · 세로축에 두고 항목마다 점을 찍습니다. 축마다 끝수 없는 눈금 3–5개와 옅은 실선 격자를 두고,
     축 이름과 단위는 축에 붙여 세로축은 그래프 왼쪽 위(↑), 가로축은 눈금 아래 오른쪽 끝(→)에 적습니다. 점은 지름 9px에 흰 테두리 2px입니다.
     마우스는 28px 안에서 가장 가까운 점을, 키보드는 좌우 화살표로 가로값 순서(path면 적은 순서)를 따라 옮깁니다.
     cfg: toneOf(p) · rankGroups(groups) 색을 줄 묶음 순서 · label(p, 순번) 이름을 적을 점 · ref { x, y, xText } 기준선 · quadrants [왼쪽 위, 오른쪽 위, 왼쪽 아래, 오른쪽 아래]
          path(적은 순서로 잇기) · radius(p, 순번) · fade(p, 순번) 점의 옅기 · yText(p) 툴팁 값 · note(p) 툴팁 덧붙임 · legend */
  function scatterCard(card, d, f, cfg) {
    const ax = d.x || {}, ay = d.y || {}, fx = format(ax), fy = format(ay);
    const pts = (d.points || []).filter((p) => p && fin(p.x) && fin(p.y)).map((p) => Object.assign({}, p, { x: +p.x, y: +p.y }));
    // 달러(usd) 축은 가장 큰 값으로 K · M · B 하나를 골라 눈금은 나눈 값으로, 축 이름에는 $K 같은 단위를 적습니다.
    const scaleOf = (fa, vals) => { const top = Math.max(0, ...vals.map(Math.abs)); return !fa.usd ? [1, ''] : top >= 1e9 ? [1e9, '$B'] : top >= 1e6 ? [1e6, '$M'] : top >= 1e3 ? [1e3, '$K'] : [1, '$']; };
    const sx = scaleOf(fx, pts.map((p) => p.x)), sy = scaleOf(fy, pts.map((p) => p.y));
    const unitOfAxis = (a, fa) => (fa.usd ? (a === ax ? sx : sy)[1] : fa.parts(0)[1]) || '';
    const nameOf = (a, fa, fallback) => `${a.label || fallback}${unitOfAxis(a, fa) ? `(${unitOfAxis(a, fa)})` : ''}`;
    // 묶음(group)이 있으면 점 수가 많은 세 묶음만 주황 · 분홍빨강 · 파랑으로, 나머지는 회색으로 칠합니다(같은 수면 먼저 나온 묶음이 앞).
    const groups = [];
    pts.forEach((p) => { if (p.group == null) return; const g = groups.find((x) => x.name === p.group); if (g) g.count += 1; else groups.push({ name: p.group, count: 1 }); });
    const ranked = cfg.rankGroups ? cfg.rankGroups(groups) : groups.slice().sort((a, b) => b.count - a.count);
    ranked.forEach((g, i) => { g.tone = ['f1', 'f2', 't3'][i] || 'f0'; });
    const groupOf = (p) => groups.find((x) => x.name === p.group);
    const toneOf = (p) => (cfg.toneOf ? cfg.toneOf(p) : (groupOf(p) || {}).tone || 'f1');
    const rest = ranked.slice(3);
    const legend = cfg.legend || (ranked.length > 1 ? ranked.slice(0, 3).map((g) => ({ name: g.name, tone: g.tone, dot: true })).concat(rest.length ? [{ name: rest.length === 1 ? rest[0].name : '기타', tone: 'f0', dot: true }] : []) : null);
    head(card, d, f, null);
    if (legend) card.append(legendEl(legend));
    const foot = el('div', 'vzb-foot vzb-fill vzb-ax vzb-sc');
    const plot = el('div', 'vzb-plot');
    plot.tabIndex = 0;
    plot.setAttribute('role', 'img');
    plot.setAttribute('aria-label', `${d.title || ''} 산점도, 점 ${pts.length}개. 좌우 화살표로 점을 옮깁니다. 모든 값은 이어지는 표에 있습니다.`);
    const svg = svgEl('svg', { class: 'vzb-ax-svg vzb-sc-svg', 'aria-hidden': 'true' });
    plot.append(svg);
    foot.append(plot);
    card.append(foot);
    // 화면 낭독기용 값 표
    const sr = el('div', 'vzb-sr'), table = el('table'), hr = el('tr');
    hr.append(el('th', null, '이름'), el('th', null, nameOf(ax, fx, '가로값')), el('th', null, nameOf(ay, fy, '세로값')));
    table.append(el('caption', null, d.title || ''), hr);
    pts.forEach((p) => { const tr = el('tr'); tr.append(el('th', null, p.name || ''), el('td', null, show(fx, p.x)), el('td', null, show(fy, p.y))); table.append(tr); });
    sr.append(table);
    card.append(sr);
    const order = pts.map((_, i) => i);
    if (!cfg.path) order.sort((a, b) => pts[a].x - pts[b].x || pts[a].y - pts[b].y);
    const fam = getComputedStyle(card).fontFamily;
    let X = null, Y = null, dots = [], front = null, lastW = 0, lastH = 0;
    // 값 범위에 8% 여유를 두고, 0 이상인 값은 0 아래로 내려가지 않게 합니다.
    const span = (vals) => {
      const lo0 = Math.min(...vals), hi0 = Math.max(...vals), pad = (hi0 - lo0 || Math.abs(hi0) || 1) * 0.08;
      return [lo0 >= 0 && lo0 - pad < 0 ? 0 : lo0 - pad, hi0 + pad];
    };
    // 축 범위는 여유를 둔 값 범위 그대로 쓰고, 눈금은 그 안의 끝수 없는 값(1 · 2 · 2.5 · 5 단위)에만 긋습니다.
    // 눈금은 most개를 넘지 않는 한 많이 두되, 2.5 단위는 다른 간격으로 셋 이상 둘 수 없을 때만 씁니다.
    const ticksOf = (lo, hi, most) => {
      let best = null;
      const e0 = Math.floor(Math.log10((hi - lo) / most)) - 1;
      for (let e = e0; e <= e0 + 3; e++) {
        for (const m of [1, 2, 2.5, 5]) {
          const step = m * Math.pow(10, e), a = Math.ceil(lo / step - 1e-9), b = Math.floor(hi / step + 1e-9), cnt = b - a + 1;
          if (cnt < 2 || cnt > most) continue;
          const score = cnt - (m === 2.5 ? 2.5 : 0);
          if (!best || score > best.score) best = { step, a, cnt, score };
        }
      }
      if (!best) return { list: [lo, hi], dec: 1 };
      const dec = stepDec(best.step);
      return { list: Array.from({ length: best.cnt }, (_, k) => +((best.a + k) * best.step).toFixed(dec)), dec };
    };
    const tickTxt = (v, dec, sc) => num(v / sc[0], Math.max(0, dec - Math.round(Math.log10(sc[0])))).replace(/^-/, '−');
    function draw(w, h) {
      lastW = w; lastH = h;
      svg.setAttribute('width', w);
      svg.setAttribute('height', h);
      svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
      svg.replaceChildren();
      dots = [];
      if (!pts.length) return;
      const ref = cfg.ref || {};
      const [xl, xh] = span(pts.map((p) => p.x).concat(fin(ref.x) ? [ref.x] : []));
      const [yl, yh] = span(pts.map((p) => p.y).concat(fin(ref.y) ? [ref.y] : []));
      // 위에는 세로축 이름, 아래에는 가로 눈금과 가로축 이름이 한 줄씩 들어갑니다.
      const top = 26, bottom = h - 40;
      const yt = ticksOf(yl, yh, Math.max(3, Math.min(4, Math.floor((bottom - top) / 40) + 1)));
      const gut = Math.ceil(Math.max(...yt.list.map((v) => textW(tickTxt(v, yt.dec, sy), 400, fam)))) + 10;
      const right = w - 8;
      const xt = ticksOf(xl, xh, Math.max(3, Math.min(5, Math.floor((right - gut) / 64) + 1)));
      svg.append(svgText(0, 12, `↑ ${nameOf(ay, fy, '세로값')}`, 'vzb-sc-name'), svgText(right, h - 3, `${nameOf(ax, fx, '가로값')} →`, 'vzb-sc-name', 'end'));
      X = (v) => gut + ((v - xl) / (xh - xl || 1)) * (right - gut);
      Y = (v) => bottom - ((v - yl) / (yh - yl || 1)) * (bottom - top);
      // 그래프 바깥 테두리: 아래와 왼쪽은 축선, 위와 오른쪽은 격자와 같은 옅은 선
      svg.append(svgEl('rect', { x: gut + 0.5, y: top + 0.5, width: Math.max(0, right - gut - 1), height: Math.max(0, bottom - top - 1), class: 'vzb-sc-frame' }));
      yt.list.forEach((v) => {
        const y = Math.round(Y(v)) + 0.5;
        svg.append(svgEl('line', { x1: gut, x2: right, y1: y, y2: y, class: v === 0 ? 'vzb-ax-zero' : 'vzb-ax-grid' }), svgText(gut - 8, y + 4, tickTxt(v, yt.dec, sy), 'vzb-ax-label', 'end'));
      });
      xt.list.forEach((v) => {
        const x = Math.round(X(v)) + 0.5;
        svg.append(svgEl('line', { x1: x, x2: x, y1: top, y2: bottom, class: v === 0 ? 'vzb-ax-zero' : 'vzb-ax-grid' }), svgText(x, bottom + 18, tickTxt(v, xt.dec, sx), 'vzb-ax-label', 'middle'));
      });
      // 이미 적은 글자의 자리(기준선 이름 · 구역 이름 · 점 이름). 점 이름은 이 자리와 겹치지 않는 곳을 찾습니다.
      const taken = [];
      const boxAt = (tx, ty, txt, anchor) => { const w = textW(txt, 500, fam), l = anchor === 'end' ? tx - w : anchor === 'middle' ? tx - w / 2 : tx; return [l - 2, ty - 11, l + w + 2, ty + 3]; };
      // 기준선(평균 등)과 사분면 이름
      if (fin(ref.x)) {
        const x = Math.round(X(ref.x)) + 0.5;
        svg.append(svgEl('line', { x1: x, x2: x, y1: top, y2: bottom, class: 'vzb-sc-ref' }));
        if (ref.xText) { svg.append(svgText(x + 6, top + 12, ref.xText, 'vzb-ax-note')); taken.push(boxAt(x + 6, top + 12, ref.xText, 'start')); }
      }
      if (fin(ref.y)) {
        const y = Math.round(Y(ref.y)) + 0.5;
        svg.append(svgEl('line', { x1: gut, x2: right, y1: y, y2: y, class: 'vzb-sc-ref' }));
      }
      if (cfg.quadrants) {
        const q = cfg.quadrants, at = [[gut + 8, top + 30, 'start'], [right - 8, top + 30, 'end'], [gut + 8, bottom - 10, 'start'], [right - 8, bottom - 10, 'end']];
        q.forEach((name, k) => { if (name) { svg.append(svgText(at[k][0], at[k][1], name, 'vzb-sc-quad', at[k][2])); taken.push(boxAt(at[k][0], at[k][1], name, at[k][2])); } });
      }
      // 이어 그리기: 묶음이 있으면 구간마다 도착점의 묶음 색으로, 없으면 주황 한 가지로 잇습니다.
      if (cfg.path && groups.length) {
        pts.slice(1).forEach((p, k) => svg.append(svgEl('line', { x1: X(pts[k].x), y1: Y(pts[k].y), x2: X(p.x), y2: Y(p.y), class: `vzb-sc-seg vzb-${toneOf(p)}` })));
      } else if (cfg.path) svg.append(svgEl('path', { d: pts.map((p, i) => `${i ? 'L' : 'M'}${X(p.x).toFixed(1)},${Y(p.y).toFixed(1)}`).join(''), class: 'vzb-sc-path' }));
      // 회색 묶음을 먼저 찍어 색 있는 점이 위에 오게 합니다.
      const drawOrder = pts.map((_, i) => i).sort((a, b) => (toneOf(pts[a]) === 'f0' ? 0 : 1) - (toneOf(pts[b]) === 'f0' ? 0 : 1));
      dots = new Array(pts.length);
      drawOrder.forEach((i) => {
        const p = pts[i];
        const dot = svgEl('circle', { cx: X(p.x).toFixed(1), cy: Y(p.y).toFixed(1), r: cfg.radius ? cfg.radius(p, i) : 4.5, class: `vzb-sc-dot vzb-${toneOf(p)}` });
        if (cfg.fade) dot.style.fillOpacity = cfg.fade(p, i);
        svg.append(dot);
        dots[i] = dot;
      });
      // 이름은 고른 점에만 적습니다. 점 오른쪽 · 왼쪽 · 위 · 아래 순서로, 그래프 안이면서 구역 이름이나 먼저 적은 이름과 겹치지 않는 자리에 둡니다.
      const hit = (b) => taken.some((o) => b[0] < o[2] && b[2] > o[0] && b[1] < o[3] && b[3] > o[1]);
      const inside = (b) => b[0] >= gut && b[2] <= right && b[1] >= top - 6 && b[3] <= bottom + 6;
      pts.forEach((p, i) => {
        if (!(cfg.label ? cfg.label(p, i) : p.label)) return;
        const x = X(p.x), y = Y(p.y), txt = String(p.name || '');
        const spots = [[x + 9, y + 4, 'start'], [x - 9, y + 4, 'end'], [x, y - 10, 'middle'], [x, y + 18, 'middle']];
        const pick = spots.find((sp) => { const b = boxAt(sp[0], sp[1], txt, sp[2]); return inside(b) && !hit(b); }) || spots.find((sp) => inside(boxAt(sp[0], sp[1], txt, sp[2]))) || spots[0];
        taken.push(boxAt(pick[0], pick[1], txt, pick[2]));
        svg.append(svgText(pick[0], pick[1], txt, 'vzb-ax-note', pick[2]));
      });
      front = svgEl('g', {});
      svg.append(front);
    }
    onSize(plot, draw);
    if (document.fonts && document.fonts.ready) document.fonts.ready.then(() => { if (lastW) draw(lastW, lastH); });
    interact(plot, () => order.length,
      (cx, cy) => {
        if (!X) return -1;
        const r = plot.getBoundingClientRect(), x = cx - r.left, y = cy - r.top;
        let best = -1, bd = 28 * 28;
        order.forEach((i, k) => { const dx = X(pts[i].x) - x, dy = Y(pts[i].y) - y, dd = dx * dx + dy * dy; if (dd < bd) { bd = dd; best = k; } });
        return best;
      },
      (k) => {
        const i = order[k], p = pts[i];
        svg.classList.add('is-hover');
        dots.forEach((c, j) => c.classList.toggle('is-on', j === i));
        front.replaceChildren(svgEl('circle', { cx: X(p.x), cy: Y(p.y), r: 7.5, class: `vzb-sc-ring vzb-${toneOf(p)}` }));
        const r = plot.getBoundingClientRect();
        const note = [p.group != null ? String(p.group) : '', `${ax.label || '가로값'} ${show(fx, p.x)}`, cfg.note ? cfg.note(p) : ''].filter(Boolean).join(' · ');
        const t = tipBody({ value: cfg.yText ? cfg.yText(p) : show(fy, p.y), note }, p.name || '');
        return { x: r.left + X(p.x), y: r.top + Y(p.y) - 8, label: p.name || '', value: t.node, text: t.text };
      },
      () => { svg.classList.remove('is-hover'); dots.forEach((c) => c.classList.remove('is-on')); if (front) front.replaceChildren(); });
  }

  // Scatter - Connected: 기간마다 찍은 점을 시간 순서대로 이어, 두 측정값이 함께 어느 쪽으로 움직였는지 봅니다.
  // 묶음(group, 예: 연도)이 있으면 최근 묶음부터 주황 · 분홍빨강 · 파랑으로 칠하고, 없으면 오래된 점일수록 옅게 찍습니다.
  // 마지막 점은 크게 찍고, 처음과 마지막 점에만 기간 이름을 적습니다. 키보드는 좌우 화살표로 시간 순서를 따라갑니다.
  mount('scatter-connected', (card, d, f) => {
    const list = (d.points || []).filter((p) => p && fin(p.x) && fin(p.y)), n = list.length;
    const grouped = list.some((p) => p.group != null);
    const lastAt = (name) => { for (let i = n - 1; i >= 0; i--) if (list[i].group === name) return i; return -1; };
    scatterCard(card, d, f, {
      path: true,
      rankGroups: (gs) => gs.slice().sort((a, b) => lastAt(b.name) - lastAt(a.name)),
      label: (p, i) => i === 0 || i === n - 1,
      radius: (p, i) => (i === n - 1 ? 5.5 : 4),
      fade: grouped ? null : (p, i) => (n > 1 ? 0.3 + (0.7 * i) / (n - 1) : 1),
    });
  });
})();
</script>
