    '%1. %2 %3 → %4 %5, %6. 좌우 화살표로 항목을 옮깁니다. 모든 값은 이어지는 표에 있습니다.': '%1. %2 %3 → %4 %5, %6. Use the left and right arrows to move between items. Every value is in the table that follows.',
    '%1. %2의 비중은 %3로, %4 대비 %5입니다. 좌우 화살표로 항목을 옮깁니다. 모든 값은 이어지는 표에 있습니다.': '%1. %2 holds %3, %5 against %4. Use the left and right arrows to move between items. Every value is in the table that follows.',
    '%1. %2의 비중은 %3입니다. 좌우 화살표로 항목을 옮깁니다. 모든 값은 이어지는 표에 있습니다.': '%1. %2 holds %3. Use the left and right arrows to move between items. Every value is in the table that follows.',
    '카드 데이터(JSON)를 읽지 못했습니다: %s': 'Could not read the card data (JSON): %s', '카드를 그리지 못했습니다: %s': 'Could not draw the card: %s',
  };
  // tx(d 또는 f, 원문, 값…): d.lang(format()이 f에 옮겨 둡니다)이 en이면 영어 문구를, 아니면 원문을 돌려줍니다.
  function tx(x, s, ...a) {
    let i = 0;
    const out = x && x.lang === 'en' && EN[s] !== undefined ? EN[s] : s.split('|')[0];
    return out.replace(/%(\d|s)/g, (m, k) => String(a[k === 's' ? i++ : k - 1]));
  }
  // 숫자 뒤의 단위: 영어 단위(days · orders)는 한 칸 띄우고, 기호와 한 글자 단위(% · K · 억)는 붙입니다.
  const withUnit = (n, u) => (/^[A-Za-z]{2}/.test(u) ? `${n} ${u}` : n + u);
  function el(tag, cls, text) {
    const n = document.createElement(tag);
    if (cls) n.className = cls;
    if (text != null) n.textContent = text;
    return n;
  }
  function svgEl(tag, attrs) {
    const n = document.createElementNS(SVGNS, tag);
    for (const k in attrs) n.setAttribute(k, attrs[k]);
    return n;
  }
  const nfs = {};
  function num(v, d) {
    nfs[d] = nfs[d] || new Intl.NumberFormat('ko-KR', { minimumFractionDigits: d, maximumFractionDigits: d });
    return nfs[d].format(v);
  }
  // 원 금액: 조 · 억 · 만으로 줄여 적습니다. 10만 원보다 작으면 87,324원처럼 원 단위 그대로 적어 객단가 · CAC가 "9만"으로
  // 뭉개지지 않게 합니다. decimals를 주면 줄인 숫자의 소수 자릿수를 그 값으로 맞춥니다(1억 8,600만 원은 2면 1.86억).
  function krw(v, d) {
    const a = Math.abs(v), k = (x) => (d != null ? d : x);
    if (a >= 1e12) return [num(v / 1e12, k(1)), '조'];
    if (a >= 1e8) return [num(v / 1e8, k(a >= 1e10 ? 0 : 1)), '억'];
    if (a >= 1e5) return [num(v / 1e4, k(0)), '만'];
    return [num(v, 0), '원'];
  }
  // 달러 금액: 원래 값을 받아 K · M · B로 줄이고 소수 한 자리로 맞춥니다($745.6K). 접미사는 단위처럼 작게 씁니다.
  function usd(v, d) {
    const a = Math.abs(v), s = v < 0 ? '−' : '';
    const [div, u] = a >= 1e9 ? [1e9, 'B'] : a >= 1e6 ? [1e6, 'M'] : a >= 1e3 ? [1e3, 'K'] : [1, ''];
    return [`${s}$${num(a / div, d != null ? d : div > 1 ? 1 : 0)}`, u];
  }
  // format: eok(억 단위 값) · krw(원 단위 값) · usd(달러 값) · count(unit으로 단위) · pct(증감을 %p로) · days
  // 축이나 탭처럼 카드 안의 다른 객체로 만들 때는 두 번째 인자로 카드 데이터를 주어 lang을 잇습니다.
  function format(d, root) {
    const dec = (v, fallback) => (d.decimals != null ? d.decimals : fallback);
    const F = (o) => Object.assign(o, { lang: (root || d).lang });
    switch (d.format) {
      case 'eok': return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) || Math.abs(v) >= 100 ? 0 : 1)), '억'], rel: true });
      case 'krw': return F({ parts: (v) => krw(v, d.decimals), rel: true });
      case 'usd': return F({ parts: (v) => usd(v, d.decimals), rel: true, usd: true });
      case 'pct': return F({ parts: (v) => [num(v, dec(v, 1)), '%'], rel: false, dp: d.decimals });
      case 'days': return F({ parts: (v) => [num(v, dec(v, 1)), tx(root || d, '일')], rel: true });
      default: return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) ? 0 : 1)), d.unit || ''], rel: true });
    }
  }
  const show = (f, v) => withUnit(...f.parts(v));
  // 그릴 수 있는 숫자인지(빈 값 · 빈 글자 · NaN 제외)
  const fin = (v) => v != null && v !== '' && isFinite(v);
  // 비율의 차이(%p) 숫자: 소수 자릿수는 decimals(비우면 한 자리)를 따릅니다. 0이 아닌 차이가 0.0으로 보일 만큼 작으면
  // 자리를 늘려(최대 세 자리) 1.12%에서 1.16%로 오른 것을 0.0%p가 아니라 0.03%p로 적습니다.
  function pp(f, diff) {
    const a = Math.abs(diff);
    let k = f.dp != null ? f.dp : 1;
    while (k < 3 && a > 1e-9 && Math.round(a * 10 ** k) === 0) k++;
    return num(a, k);
  }
  // 증감: 방향은 화살표로, 좋고 나쁨(tone)은 better(up · down · neutral)로 정합니다. 비율은 %p로 적습니다.
  function delta(f, value, base, better) {
    const diff = value - base;
    const dir = Math.abs(diff) < 1e-9 ? 0 : Math.sign(diff);
    const want = better === 'down' ? -1 : better === 'neutral' ? 0 : 1;
    const tone = dir === 0 || want === 0 ? 'flat' : dir === want ? 'good' : 'bad';
    let text;
    if (!f.rel) text = tx(f, '%s%p', pp(f, diff));
    else if (base) text = `${num(Math.abs(diff / base) * 100, 1)}%`;
    else text = show(f, Math.abs(diff));
    return { dir, tone, text };
  }
  function arrow(dir) {
    const s = svgEl('svg', { class: 'vzb-arrow', viewBox: '0 0 12 12', 'aria-hidden': 'true' });
    s.append(svgEl('path', { d: dir > 0 ? 'M3.5 8.5l5-5M4.5 3.5h4v4' : 'M3.5 3.5l5 5M8.5 4.5v4h-4' }));
    return s;
  }
  // 대표 수치(단위는 작게)
  function figure(f, value) {
    const n = el('div', 'vzb-num');
    const [text, unit] = f.parts(value);
    n.append(text);
    if (unit) n.append(el('span', 'vzb-unit', unit));
    return n;
  }
  // 증감 배지: 화살표는 오르내림, 색은 좋고 나쁨(better)입니다.
  function badge(f, value, base, better) {
    const dl = delta(f, value, base, better);
    const b = el('span', `vzb-badge vzb-${dl.tone}`);
    if (dl.dir) b.append(arrow(dl.dir), el('span', 'vzb-sr', tx(f, dl.dir > 0 ? '증가' : '감소')));
    b.append(dl.text);
    return b;
  }
  // 머리: 제목 · 기간 부제(subtitle) · 대표 수치와 바로 뒤의 증감 배지. value가 null이면 제목만 둡니다.
  // 배지는 숫자 바로 뒤에 붙어 있고, 카드가 좁아 한 줄에 다 들어가지 않으면 비교 문구만 숫자 아래 줄로 내려갑니다.
  // 내려갔는지는 카드의 vzb-wrapped로 알려, 높이가 1단위인 카드가 아래 여백으로 그 줄을 받을 수 있게 합니다.
  function head(card, d, f, value, cmp, cmpText) {
    const hd = el('div', 'vzb-hd');
    hd.append(el('div', 'vzb-title', d.title || ''));
    if (d.subtitle) hd.append(el('div', 'vzb-sub', d.subtitle));
    card.append(hd);
    if (value == null) return;
    const fig = el('div', 'vzb-fig');
    const n = figure(f, value);
    fig.append(n);
    if (cmp && cmp.value != null) {
      const line = el('span', 'vzb-delta'), vs = el('span', 'vzb-vs', cmpText || tx(d, '%s 대비', cmp.label || tx(d, '비교 기간')));
      line.append(badge(f, value, cmp.value, d.better), vs);
      fig.append(line);
      onSize(fig, () => card.classList.toggle('vzb-wrapped', vs.offsetTop >= n.offsetTop + n.offsetHeight - 1));   // 숫자 칸 아래에서 시작하면 내려간 것
    }
    card.append(fig);
  }
  // 같은 종류의 카드를 모두 찾아 그립니다. 코드를 두 번 붙여 넣어도 카드는 한 번만 그립니다.
  // 데이터가 바뀌면 다시 그립니다: 페이지의 스크립트가 카드 안 JSON(<script type="application/json">)의 글자를 바꾸면 그 카드만
  // 새 요소로 갈아 끼워 새로 그립니다(필터 · 기간 컨트롤과 잇는 길). 갈아 끼운 카드에는 data-vzb-live가 붙어, 움직이는 카드는
  // 처음부터 다시 재생하지 않고 마지막 장면을 바로 그립니다. 예전 요소가 페이지에서 빠지므로 그 카드의 움직임도 함께 끝납니다.
  function mount(type, draw) {
    const watch = (card, src) => {
      if (!src || !window.MutationObserver) return;
      const mo = new MutationObserver(() => {
        mo.disconnect();
        if (!card.isConnected) return;
        const next = card.cloneNode(false);
        next.removeAttribute('data-vzb-ready');
        next.setAttribute('data-vzb-live', '');
        next.append(src);
        card.replaceWith(next);
        run();
      });
      mo.observe(src, { childList: true, characterData: true, subtree: true });
    };
    const run = () => {
      document.querySelectorAll(`.vzb-card[data-vzb="${type}"]:not([data-vzb-ready])`).forEach((card) => {
        card.setAttribute('data-vzb-ready', '');
        const src = card.querySelector('script[type="application/json"]');
        const raw = src ? src.textContent : '';
        watch(card, src);
        let d;
        try { d = JSON.parse(raw); } catch (e) {
          card.append(el('p', 'vzb-error', tx(/"lang"\s*:\s*"en"/.test(raw) ? { lang: 'en' } : null, '카드 데이터(JSON)를 읽지 못했습니다: %s', e.message)));
          return;
        }
        card.setAttribute('role', 'group');
        if (d.title) card.setAttribute('aria-label', d.title);
        // 어두운 바탕: JSON에 "theme": "dark"를 적으면 Vizney 검정 바탕에 노랑을 첫째 색으로 그립니다(tokens.css의 .vzb-dark).
        card.classList.toggle('vzb-dark', d.theme === 'dark');
        try { draw(card, d, format(d)); } catch (e) {
          card.append(el('p', 'vzb-error', tx(d, '카드를 그리지 못했습니다: %s', e.message)));
        }
      });
    };
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
    else run();
  }
  // 크기가 바뀔 때마다 그래픽을 다시 그립니다. 차트는 카드에 남은 높이를 채웁니다.
  function onSize(node, fn) {
    let lw = 0, lh = 0;
    new ResizeObserver((entries) => {
      const w = Math.floor(entries[0].contentRect.width), h = Math.floor(entries[0].contentRect.height);
      if (w > 0 && h > 0 && (w !== lw || h !== lh)) { lw = w; lh = h; fn(w, h); }
    }).observe(node);
  }

  /* 툴팁과 조작: 마우스는 올리면 보이고 벗어나면 숨깁니다. 터치는 탭하면 보이고 다른 곳을 탭하면 닫힙니다.
     키보드는 그래프에 포커스한 뒤 화살표로 옮깁니다(가로 그래프는 좌우, 목록은 위아래). */
  let tip, tipDark = false;
  // value는 글자나 요소입니다. side가 참이면 점 위가 아니라 세로선 옆, 그래프 윗선에 맞춰 둡니다(선 · 막대 차트).
  function showTip(x, y, label, value, side) {
    if (!tip) {
      tip = el('div', 'vzb-tip');
      tip.setAttribute('aria-hidden', 'true');
      document.body.append(tip);
    }
    tip.classList.toggle('vzb-dark', tipDark);
    tip.replaceChildren(el('span', 'vzb-tip-label', label), typeof value === 'string' ? el('span', 'vzb-tip-value', value) : value);
    tip.classList.add('is-on');
    const w = tip.offsetWidth, h = tip.offsetHeight, W = window.innerWidth, H = window.innerHeight;
    if (side) {
      // side가 'left'이면 기준선 왼쪽을 먼저, 그 밖에는 오른쪽을 먼저 씁니다. 자리가 모자라면 반대쪽으로 넘깁니다.
      const right = x + 14 + w <= W - 8, left = x - w - 14 >= 8;
      tip.style.left = `${(side === 'left' ? left || !right : !right) ? Math.max(8, x - w - 14) : x + 14}px`;
      tip.style.top = `${Math.min(Math.max(8, y), H - h - 8)}px`;
      return;
    }
    tip.style.left = `${Math.min(Math.max(8, x - w / 2), W - w - 8)}px`;
    tip.style.top = `${y - h - 12 < 8 ? y + 16 : y - h - 12}px`;
  }
  function hideTip() { if (tip) tip.classList.remove('is-on'); }
  // 툴팁 내용: { value, note } 또는 { rows: [{ tone, value, name }], note }. 값이 먼저, 이름은 뒤에 옅게 적습니다.
  function tipBody(spec, label) {
    const body = el('span', 'vzb-tip-body');
    let text = label;
    if (spec.rows) {
      const grid = el('span', 'vzb-tip-rows');
      spec.rows.forEach((r) => {
        const row = el('span', 'vzb-tip-row');
        row.append(el('i', `vzb-tip-key vzb-${r.tone}`), el('b', null, r.value), el('span', null, r.name));
        grid.append(row);
        text += `, ${r.name} ${r.value}`;
      });
      body.append(grid);
    } else {
      body.append(el('span', 'vzb-tip-value', spec.value));
      text += ` ${spec.value}`;
    }
    // note는 한 줄 글 또는 줄마다 나눈 배열입니다.
    (Array.isArray(spec.note) ? spec.note : [spec.note]).filter(Boolean).forEach((line) => {
      body.append(el('span', 'vzb-tip-note', line));
      text += `, ${line}`;
    });
    return { node: body, text };
  }
  // 범례: [{ name, tone, box(면), dot(점), dim(옅게), tick(세로 눈금) }]. 선은 짧은 선, 막대와 면은 네모, 점은 동그라미로 표시합니다.
  function legendEl(items) {
    const ul = el('ul', 'vzb-legend');
    items.forEach((it) => {
      const li = el('li');
      li.append(el('i', `vzb-key vzb-${it.tone}${it.box ? ' is-box' : ''}${it.dot ? ' is-dot' : ''}${it.dim ? ' is-dim' : ''}${it.tick ? ' is-tick' : ''}`), it.name);
      ul.append(li);
    });
    return ul;
  }
  // 글자 폭(12px 기준, size를 주면 그 크기)과 SVG 글자
  let meter;
  function textW(s, weight, family, size) {
    meter = meter || document.createElement('canvas').getContext('2d');
    meter.font = `${weight} ${size || 12}px ${family}`;
    return meter.measureText(s).width;
  }
  function svgText(x, y, s, cls, anchor) {
    const t = svgEl('text', { x, y, class: cls, 'text-anchor': anchor || 'start' });
    t.textContent = s;
    return t;
  }
  // 차이를 부호와 함께 적습니다. 비율 지표는 %p입니다.
  function signed(f, diff) {
    const s = diff > 0 ? '+' : diff < 0 ? '−' : '';
    return f.rel ? `${s}${show(f, Math.abs(diff))}` : `${s}${pp(f, diff)}%p`;
  }
  // pick(clientX, clientY) → 순번, point(순번) → { x, y, label, value, text, side }(화면 좌표), clear() → 강조 해제
  // n은 개수나 개수를 돌려주는 함수입니다. vertical이면 위아래 화살표로 옮깁니다. 돌려주는 { show(순번), at() }로 키를 더 붙입니다.
  // 아무것도 고르지 않은 채 화살표를 누르면 기간 차트는 마지막(최근)부터, vertical이나 first면 첫 순번부터 짚습니다.
  function interact(plot, n, pick, point, clear, vertical, first) {
    let idx = -1, touched = false;
    const count = () => (typeof n === 'function' ? n() : n);
    const live = el('span', 'vzb-sr');
    live.setAttribute('aria-live', 'polite');
    plot.after(live);
    const go = (i) => {
      if (i < 0 || i >= count()) return;
      idx = i;
      const p = point(i);
      tipDark = !!plot.closest('.vzb-dark');
      showTip(p.x, p.y, p.label, p.value, p.side);
      live.textContent = p.text || `${p.label} ${p.value}`;
    };
    const end = () => {
      if (idx < 0) return;
      idx = -1;
      clear();
      hideTip();
    };
    const [back, next] = vertical ? ['ArrowUp', 'ArrowDown'] : ['ArrowLeft', 'ArrowRight'];
    // pick이 -1이면(조각 밖) 강조를 풉니다.
    const at = (e) => { const i = pick(e.clientX, e.clientY); if (i < 0) end(); else go(i); };
    plot.addEventListener('pointermove', (e) => { if (e.pointerType !== 'touch' || touched) at(e); });
    plot.addEventListener('pointerdown', (e) => { if (e.pointerType === 'touch') { touched = true; at(e); } });
    plot.addEventListener('pointerleave', (e) => { if (e.pointerType !== 'touch') end(); });
    // 문서와 창에 건 듣기는 그래프가 페이지에서 빠지면(카드를 다시 그리면) 스스로 풉니다.
    const away = (e) => { if (!plot.isConnected) document.removeEventListener('pointerdown', away, true); else if (touched && !plot.contains(e.target)) { touched = false; end(); } };
    document.addEventListener('pointerdown', away, true);
    plot.addEventListener('keydown', (e) => {
      if (e.key !== back && e.key !== next) return;
      e.preventDefault();
      const last = count() - 1;
      go(idx < 0 ? (vertical || first ? 0 : last) : Math.max(0, Math.min(last, idx + (e.key === next ? 1 : -1))));
    });
    plot.addEventListener('blur', end);
    const scrolled = () => { if (!plot.isConnected) window.removeEventListener('scroll', scrolled); else end(); };
    window.addEventListener('scroll', scrolled, { passive: true });
    return { show: go, at: () => idx, end };
  }

  // Waterfall Walk: 시작 값(start)에서 출발해 요인(labels)마다의 증감(values)을 앞 알약이 끝난 높이에 이어 붙이고, 끝 값은 카드가 계산해 끝 알약으로 세웁니다.
  // 막대는 양 끝이 둥근 알약입니다. 시작과 끝은 바닥(0)에서 서는 글자색(--vzb-ink), 요인은 결과를 좋게 만든 쪽이 초록(--vzb-good), 나쁘게 만든 쪽이
  // 빨강(--vzb-bad)이고(better: up · down), neutral이면 모두 첫째 데이터 색입니다. 알약마다 도착한 쪽 끝(머리)에 카드 바탕색 점을 뚫어,
  // 점이 위면 늘었고 아래면 줄었음을 색 없이도 알립니다. 알약 사이를 잇는 선은 없고, 이웃한 알약의 끝 높이가 맞닿아 이어집니다.
  // 요인 값은 늘린 알약 위 · 줄인 알약 아래에 부호와 함께 적고, 이웃 값과 겹치면 작은 값부터 생략합니다(툴팁과 표에는 남습니다).
  // 값이 알약 굵기보다 작으면 길이를 부풀리지 않고 그 값만큼 납작하게 그립니다. 이름이 칸보다 길면 두 줄로 엇갈려 적고, 그래도 길면 줄여 적습니다.
  // 큰 숫자는 끝 값, 증감 배지는 시작 값 대비입니다.
  // 움직임: 시작 알약과 큰 숫자가 놓여 있다가, 앞 알약 머리의 점만 한 공이 다음 칸으로 건너가 그 높이에서 알약으로 늘어납니다(요인마다 왼쪽부터 하나씩).
  // 큰 숫자는 보이는 길이를 그대로 셉니다. 마지막 요인 뒤에는 공이 끝 칸으로 건너가 바닥까지 쏟아지고, 증감 배지가 붙은 뒤 멈춥니다.
  // 카드가 화면에 보이면 한 번 움직이고, 다시 보기 단추로 되풀이합니다. 데이터가 바뀌어 다시 그릴 때(data-vzb-live)는 마지막 장면을 바로 그립니다.
  // 카드를 감싼 요소에 data-vzb-loop가 있으면(갤러리) 끝에서 잠시 쉬고 되풀이합니다.
  mount('waterfall-walk', (card, d, f) => {
    // 걸러서 남은 요인이 없을 때(labels가 빈 배열)는 오류가 아니라 빈 화면입니다.
    if (Array.isArray(d.labels) && !d.labels.length) {
      head(card, d, f, null);
      card.append(el('p', 'vzb-ww-none', tx(d, '표시할 항목이 없습니다.')));
      return;
    }
    const labels = Array.isArray(d.labels) ? d.labels.map(String) : [], raw = Array.isArray(d.values) ? d.values : [];
    if (!labels.length || raw.length !== labels.length) throw new Error(tx(d, 'labels와 values를 확인하세요.'));
    if (!fin(d.start)) throw new Error(tx(d, 'start에 시작 값을 숫자로 적으세요.'));
    const v0 = +d.start, want = d.better === 'down' ? -1 : d.better === 'neutral' ? 0 : 1;
    const given = (s) => s != null && s !== '';
    const startName = given(d.startLabel) ? String(d.startLabel) : tx(d, '시작'), endName = given(d.endLabel) ? String(d.endLabel) : tx(d, '끝');
    // 칸: 시작 · 요인들 · 끝. 요인은 앞 알약이 끝난 높이(from)에서 출발해 증감만큼 오르내립니다(to). 값이 null인 요인은 높이를 그대로 넘깁니다.
    let level = v0;
    const steps = labels.map((name, i) => {
      const v = fin(raw[i]) ? +raw[i] : null, from = level;
      if (v != null) level += v;
      return { kind: 'step', name, v, from, to: level };
    });
    const v1 = level;
    const cols = [{ kind: 'start', name: startName, v: v0, from: 0, to: v0 }, ...steps, { kind: 'end', name: endName, v: v1, from: 0, to: v1 }];
    const N = cols.length, S = steps.length;
    cols.forEach((c, k) => {
      c.k = k;
      c.bar = c.v != null && c.v !== 0;
      c.tone = c.kind !== 'step' ? 'total' : !c.bar ? 'none' : want === 0 ? 'plain' : Math.sign(c.v) === want ? 'good' : 'bad';
    });

    const live = card.hasAttribute('data-vzb-live');   // 데이터가 바뀌어 다시 그린 카드: 처음부터 재생하지 않고 마지막 장면을 그립니다
    head(card, d, f, null);
    const hd = card.querySelector('.vzb-hd');
    // 큰 숫자: 끝 값. 움직이는 동안은 지금 보이는 높이를 그대로 세고, 끝 알약이 다 차면 증감 배지와 '시작 대비'가 붙습니다.
    // 시작 값이 0 이하이면 비율이 뜻을 잃으므로 배지에 차이를 적습니다.
    const fig = el('div', 'vzb-fig vzb-ww-fig'), numEl = figure(f, live ? v1 : v0), nowText = numEl.firstChild;
    let unitEl = numEl.querySelector('.vzb-unit');
    const bd = badge(f, v1, v0, d.better), vs = el('span', 'vzb-vs', tx(d, '%s 대비', startName));
    if (!(v0 > 0)) bd.lastChild.textContent = show(f, Math.abs(v1 - v0));
    bd.hidden = vs.hidden = !live;
    fig.append(numEl, bd, vs);
    card.append(fig);
    const btn = el('button', 'vzb-ww-btn');
    btn.type = 'button';
    btn.setAttribute('aria-label', tx(d, '처음부터 다시'));
    const ic = svgEl('svg', { class: 'vzb-ww-ic', viewBox: '0 0 16 16', 'aria-hidden': 'true' });
    ic.append(svgEl('path', { d: 'M3.7 9a4.4 4.4 0 1 0 1.2-4.2' }), svgEl('path', { d: 'M4.6 2.6v2.6h2.6' }));
    btn.append(ic);
    hd.append(btn);   // 다시 보기 단추는 머리의 오른쪽 위
    const foot = el('div', 'vzb-foot vzb-fill vzb-ww');
    const plot = el('div', 'vzb-plot');
    plot.tabIndex = 0;
    plot.setAttribute('role', 'img');
    plot.setAttribute('aria-label', tx(d, '%1. %2 %3 → %4 %5, %6. 좌우 화살표로 항목을 옮깁니다. 모든 값은 이어지는 표에 있습니다.', d.title || '', startName, show(f, v0), endName, show(f, v1), signed(f, v1 - v0)));
    const svg = svgEl('svg', { class: 'vzb-ww-svg', 'aria-hidden': 'true' });
    plot.append(svg);
    foot.append(plot);
    card.append(foot);
    // 화면 낭독기용 표: 이름 · 값(요인은 증감) · 누계
    const sr = el('div', 'vzb-sr'), table = el('table'), hr = el('tr');
    hr.append(...[tx(d, '이름'), tx(d, '값'), tx(d, '누계')].map((k) => el('th', null, k)));
    table.append(el('caption', null, d.title || ''), hr);
    const valText = (c) => (c.kind !== 'step' ? show(f, c.v) : c.v == null ? tx(d, '데이터 없음') : signed(f, c.v));
    cols.forEach((c) => { const tr = el('tr'); tr.append(el('th', null, c.name), el('td', null, valText(c)), el('td', null, show(f, c.to))); table.append(tr); });
    sr.append(table);
    card.append(sr);

    // 그림: 바닥선(0) · 알약 · 머리의 점 · 값 · 이름. 알약은 출발한 높이(꼬리)에서 도착한 높이(머리)까지이고, 머리 쪽에 카드 바탕색 점을 뚫어
    // 어느 쪽으로 움직였는지를 색 없이도 알립니다(점이 위면 늘었고, 아래면 줄었습니다). 합계 알약은 바닥에서 서고 머리가 값입니다.
    const base = svgEl('line', { class: 'vzb-ww-base' }), barsG = svgEl('g', {});
    cols.forEach((c) => {
      c.g = svgEl('g', { class: `vzb-ww-col is-${c.tone}` });
      c.pill = svgEl('rect', { class: 'vzb-ww-bar' });
      c.dot = svgEl('circle', { class: 'vzb-ww-dot' });
      c.val = svgText(0, 0, valText(c), 'vzb-ww-val', 'middle');
      c.nm = svgText(0, 0, c.name, 'vzb-ww-name', 'middle');
      c.g.append(c.pill, c.dot, c.val, c.nm);
      barsG.append(c.g);
    });
    svg.append(base, barsG);
    const fam = getComputedStyle(card).fontFamily, fx = (v) => v.toFixed(1), clamp = (v) => Math.max(0, Math.min(1, v));
    // 이름을 maxW 안에 줄여 적습니다(끝에 …). 툴팁과 표에는 온전한 이름이 있습니다.
    const fit = (s, maxW) => {
      if (textW(s, 400, fam, 11) <= maxW) return s;
      let lo = 0, hi = s.length;
      while (lo < hi) { const m = (lo + hi + 1) >> 1; if (textW(`${s.slice(0, m)}…`, 400, fam, 11) <= maxW) lo = m; else hi = m - 1; }
      return `${s.slice(0, Math.max(1, lo))}…`;
    };
    // 시간표(밀리초): 시작 알약을 읽는 시간 · 요인 하나에 쓰는 시간(다음 칸으로 건너가기 HOP + 그 자리에서 늘어나기 GROW) · 끝 알약이 바닥까지 쏟아지는 시간 · 배지가 붙는 시간
    const T0 = 520, STEP = Math.max(340, Math.min(560, 3000 / Math.max(1, S))), HOP = Math.round(STEP * 0.42), GROW = STEP - HOP - 20, POUR = 480, BADGE = 240, ARC = 6;
    const easeOut = (k) => 1 - (1 - k) ** 3, easeInOut = (k) => (k < 0.5 ? 4 * k ** 3 : 1 - (-2 * k + 2) ** 3 / 2);
    cols.forEach((c) => {
      c.at = c.kind === 'start' ? -1e9 : T0 + (c.k - 1) * STEP;   // 앞 칸의 머리에서 이 칸으로 건너오기 시작하는 때
      c.dur = c.kind === 'end' ? POUR : GROW;
      c.done = c.at + HOP + c.dur;
    });
    const DONE = cols[N - 1].done, TOTAL = DONE + 60 + BADGE;
    let W = 0, H = 0, y = (v) => v, slot = 1, bw = 0, dotR = 0, t = 0;
    function layout(w, h) {
      W = w; H = h;
      svg.setAttribute('width', w);
      svg.setAttribute('height', h);
      svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
      slot = w / N;
      // 이름: 한 줄에 다 들어가면 한 줄, 아니면 두 줄로 엇갈려(이웃 칸까지 써서) 적고, 그래도 넘치면 줄여 적습니다.
      const nameW = Math.max(...cols.map((c) => textW(c.name, c.kind === 'step' ? 400 : 500, fam, 11))) + 6;
      const two = nameW > slot, NAMEH = two ? 36 : 22, room = (two ? 2 * slot : slot) - 6;
      // 높이: 0과 모든 알약의 양 끝이 들어가게 하고, 위아래에 값을 적을 자리를 남깁니다.
      const lvls = [0, ...cols.flatMap((c) => [c.from, c.to])], lo = Math.min(...lvls), hi = Math.max(...lvls), span = hi - lo || 1;
      // 아래쪽 자리는 내려간 알약의 값이나 0 아래의 합계가 이름 줄에 닿을 때만 남깁니다.
      const PADT = 16, yb = h - NAMEH, scale = (padb) => Math.max(1, yb - PADT - padb) / span;
      const s0 = scale(0), low = cols.some((c) => c.bar && c.to < c.from && PADT + (hi - c.to) * s0 + 17 > yb);
      const s = scale(low || lo < 0 ? 16 : 0);
      y = (v) => PADT + (hi - v) * s;
      // 알약 굵기: 칸의 60%, 30px까지. 증감이 합계에 비해 작아 알약이 짧으면(요인 길이의 가운데 값 기준) 굵기도 줄여, 절반 넘는 알약이 굵기보다 길게 합니다.
      const lens = steps.filter((c) => c.bar).map((c) => Math.abs(c.v) * s).sort((a, b) => a - b);
      const mid = lens.length ? (lens[(lens.length - 1) >> 1] + lens[lens.length >> 1]) / 2 : 1e9;
      bw = Math.max(Math.min(12, slot * 0.6), Math.min(30, slot * 0.6, mid / 1.3));
      dotR = Math.max(2, bw * 0.22);
      const y0 = y(0);
      base.setAttribute('x1', 0);
      base.setAttribute('x2', w);
      base.setAttribute('y1', fx(Math.round(y0) + 0.5));
      base.setAttribute('y2', fx(Math.round(y0) + 0.5));
      cols.forEach((c) => {
        c.cx = slot * (c.k + 0.5);
        c.yT = y(c.from);                      // 꼬리 끝(출발한 높이)
        c.yH = y(c.to);                        // 머리 끝(도착한 높이)
        c.len = Math.abs(c.yH - c.yT);
        c.dir = c.yH < c.yT ? -1 : 1;          // 머리가 위(-1)인지 아래(1)인지
        c.h0 = Math.min(bw, c.len);            // 건너올 때의 크기: 알약 굵기만 한 공. 값이 그보다 작으면 값만큼만
        c.hy = c.bar ? c.yH - c.dir * c.h0 / 2 : c.yH;   // 머리의 가운데(점이 놓이는 자리)
        c.nm.textContent = fit(c.name, room);
        c.nm.setAttribute('x', fx(c.cx));
        c.nm.setAttribute('y', fx(yb + 15 + (two && c.k % 2 ? 14 : 0)));
        // 값: 올라간 알약(과 바닥 위의 합계)은 머리 위에, 내려간 알약은 머리 아래에 적습니다.
        const down = c.to < c.from, tw = textW(c.val.textContent, c.kind === 'step' ? 500 : 700, fam, 11) + 4;
        c.vy = down ? c.yH + 14 : c.yH - 6;
        c.vx = Math.max(tw / 2, Math.min(w - tw / 2, c.cx));
        c.box = [c.vx - tw / 2, c.vx + tw / 2, down ? c.yH + 3 : c.yH - 16, down ? c.yH + 16 : c.yH - 3];
        c.val.setAttribute('x', fx(c.vx));
        c.val.setAttribute('y', fx(c.vy));
      });
      // 건너오는 공이 떠나는 자리: 앞쪽에서 가장 가까운, 알약이 있는 칸의 머리입니다.
      cols.forEach((c) => { let p = cols[c.k - 1]; while (p && !p.bar && p.k > 0) p = cols[p.k - 1]; c.src = p || c; });
      // 겹치는 값은 합계부터, 그다음 큰 증감부터 자리를 차지하고 나머지는 생략합니다.
      const placed = [];
      [...cols].sort((a, b) => (a.kind === 'step') - (b.kind === 'step') || Math.abs(b.v || 0) - Math.abs(a.v || 0)).forEach((c) => {
        c.valOn = c.kind !== 'step' || c.v != null;
        if (!c.valOn) return;
        const [l, r, tp, bt] = c.box;
        if (placed.some(([l2, r2, t2, b2]) => l < r2 && r > l2 && tp < b2 && bt > t2)) { c.valOn = false; return; }
        placed.push(c.box);
      });
      render();
    }
    // 한 칸의 진행: hop은 앞 칸의 머리에서 이 칸으로 건너온 정도, grow는 그 자리에서 늘어난 정도입니다. 시작 알약은 처음부터 다 서 있습니다.
    const phase = (c) => (c.kind === 'start' ? [1, 1] : [easeInOut(clamp((t - c.at) / HOP)), (c.kind === 'end' ? easeInOut : easeOut)(clamp((t - c.at - HOP) / c.dur))]);
    // 지금 보이는 높이(큰 숫자): 출발한 요인 가운데 마지막 요인의 알약이 지금 보이는 길이만큼입니다. 끝 알약이 쏟아지는 동안은 도착한 높이 그대로입니다.
    function shown() {
      let v = v0;
      steps.forEach((c) => {
        if (c.v == null || t < c.at) return;
        const [hop, grow] = phase(c);
        v = c.from + c.v * (c.len > 0 ? (hop * c.h0 + (c.len - c.h0) * grow) / c.len : grow);
      });
      return v;
    }
    const pillAt = (c, x, top, w, h, dx, dy, r) => {
      c.pill.setAttribute('x', fx(x));
      c.pill.setAttribute('y', fx(top));
      c.pill.setAttribute('width', fx(w));
      c.pill.setAttribute('height', fx(h));
      c.pill.setAttribute('rx', fx(Math.min(w, h) / 2));
      c.dot.setAttribute('cx', fx(dx));
      c.dot.setAttribute('cy', fx(dy));
      c.dot.setAttribute('r', fx(Math.max(0, r)));
    };
    let badgeOn = live;
    function render() {
      if (!W) return;
      cols.forEach((c) => {
        const [hop, grow] = phase(c), onAir = c.bar && t >= c.at;
        c.pill.setAttribute('opacity', onAir ? 1 : 0);
        c.dot.setAttribute('opacity', onAir ? 1 : 0);
        if (onAir && hop < 1) {
          // 건너오는 중: 앞 칸 머리의 점만 한 공이 이 칸의 색으로 떠나, 알약 굵기만큼 커지며 살짝 뛰어 건너옵니다.
          const land = c.kind === 'end' ? c.hy : c.yT + c.dir * c.h0 / 2;
          const sw = 2 * dotR + (bw - 2 * dotR) * hop, sh = Math.min(sw, c.h0);
          const sx = c.src.cx + (c.cx - c.src.cx) * hop, sy = c.src.hy + (land - c.src.hy) * hop - ARC * Math.sin(Math.PI * hop);
          pillAt(c, sx - sw / 2, sy - sh / 2, sw, sh, sx, sy, Math.min(dotR, sh / 2 - 2.5) * hop);
        } else if (onAir) {
          // 늘어나는 중: 요인은 꼬리를 출발한 높이에 둔 채 머리가 도착한 높이로 가고, 끝 알약은 머리를 도착한 높이에 둔 채 바닥까지 쏟아집니다.
          const hh = c.h0 + (c.len - c.h0) * grow, rr = Math.min(bw, hh) / 2;
          const top = c.kind === 'end' ? (c.dir < 0 ? c.yH : c.yH - hh) : (c.dir < 0 ? c.yT - hh : c.yT);
          const headY = c.kind === 'end' ? c.hy : c.yT + c.dir * (hh - rr);
          pillAt(c, c.cx - bw / 2, top, bw, hh, c.cx, headY, Math.min(dotR, rr - 2.5));
        }
        c.val.setAttribute('opacity', c.valOn ? fx(c.kind === 'start' ? 1 : clamp((t - c.done + 120) / 200)) : 0);
      });
      const [txt, unit] = f.parts(t >= DONE ? v1 : shown());
      nowText.data = txt;
      if (unit && !unitEl) { unitEl = el('span', 'vzb-unit'); numEl.append(unitEl); }
      if (unitEl) unitEl.textContent = unit || '';
      const ready = t >= DONE + 60;
      if (ready !== badgeOn) {
        badgeOn = ready;
        bd.hidden = vs.hidden = !ready;
        bd.classList.toggle('vzb-ww-in', ready);
        vs.classList.toggle('vzb-ww-in', ready);
      }
    }

    // 재생: 화면에 보이고(inView) 마우스 · 포커스로 잠시 멈추지 않았을 때(holds)만 움직입니다. 카드가 페이지에서 빠지면 멈춥니다.
    const loop = !!card.closest('[data-vzb-loop]');
    const holds = new Set();
    let playing = false, seen = false, inView = false, rest = 0, last = 0, raf = 0;
    const on = () => playing && inView && !holds.size && card.isConnected;
    function tick(stamp) {
      raf = 0;
      if (!on()) { last = 0; return; }
      const dt = last ? Math.min(100, stamp - last) : 16;
      last = stamp;
      if (rest > 0) { rest -= dt; if (rest <= 0) t = 0; }
      else {
        t = Math.min(TOTAL, t + dt);
        if (t >= TOTAL) { if (loop) rest = 2600; else playing = false; }
      }
      render();
      if (on()) raf = requestAnimationFrame(tick);
    }
    function kick() { if (!raf && on()) raf = requestAnimationFrame(tick); }
    function play() { t = 0; rest = 0; last = 0; playing = true; render(); kick(); }
    btn.addEventListener('click', play);
    plot.addEventListener('pointerenter', (e) => { if (e.pointerType === 'mouse') { holds.add('hover'); render(); } });
    plot.addEventListener('pointerleave', () => { holds.delete('hover'); kick(); });
    plot.addEventListener('focus', () => { if (plot.matches(':focus-visible')) { holds.add('focus'); render(); } });   // 키보드로 들어왔을 때만 멈춥니다
    plot.addEventListener('blur', () => { holds.delete('focus'); kick(); });
    // 툴팁: 알약을 짚으면 값을, 요인은 증감과 누계(앞 → 뒤)를, 끝 알약은 시작 대비 증감을 보여 주고 그 알약만 진하게 남깁니다.
    interact(plot, N,
      (cx) => Math.max(0, Math.min(N - 1, Math.floor((cx - plot.getBoundingClientRect().left) / slot))),
      (k) => {
        const c = cols[k], r = plot.getBoundingClientRect();
        svg.classList.add('is-hover');
        cols.forEach((x) => x.g.classList.toggle('is-on', x === c));
        const note = c.kind === 'step' ? (c.v == null ? null : tx(d, '누계 %1 → %2', show(f, c.from), show(f, c.to)))
          : c.kind === 'end' ? `${signed(f, v1 - v0)} · ${tx(d, '%s 대비', startName)}` : null;
        const tb = tipBody({ value: valText(c), note }, c.name);
        return { x: r.left + c.cx, y: r.top + Math.min(c.yT, c.yH) - 6, label: c.name, value: tb.node, text: tb.text };
      },
      () => { svg.classList.remove('is-hover'); cols.forEach((x) => x.g.classList.remove('is-on')); },
      false, true);

    if (live) t = 1e9;   // 마지막 장면에서 시작합니다
    onSize(plot, layout);
    if (document.fonts && document.fonts.ready) document.fonts.ready.then(() => { if (W) layout(W, H); });
    // 화면에 처음 보일 때 움직입니다. 화면 밖에 있는 동안은 멈춰 둡니다.
    const start = () => { if (!seen) { seen = true; if (!live) play(); } };
    if ('IntersectionObserver' in window) {
      const io = new IntersectionObserver((es) => {
        if (!card.isConnected) { io.disconnect(); return; }
        inView = es[es.length - 1].isIntersecting;
        if (inView) start();
        kick();
      }, { threshold: 0.3 });
      io.observe(card);
    } else { inView = true; start(); }
  });
})();
</script>
