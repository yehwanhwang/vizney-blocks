  // tx(d 또는 f, 원문, 값…): d.lang(format()이 f에 옮겨 둡니다)이 en이면 영어 문구를, 아니면 원문을 돌려줍니다.
  function tx(x, s, ...a) {
    let i = 0;
    const out = x && x.lang === 'en' && EN[s] !== undefined ? EN[s] : s.split('|')[0];
    return out.replace(/%(\d|s)/g, (m, k) => String(a[k === 's' ? i++ : k - 1]));
  }
  // 숫자 뒤의 단위: 영어 단위(days · orders)는 한 칸 띄우고, 기호와 한 글자 단위(% · K · 억)는 붙입니다.
  const withUnit = (n, u) => (/^[A-Za-z]{2}/.test(u) ? `${n} ${u}` : n + u);
  function el(tag, cls, text) {
    const n = document.createElement(tag);
    if (cls) n.className = cls;
    if (text != null) n.textContent = text;
    return n;
  }
  function svgEl(tag, attrs) {
    const n = document.createElementNS(SVGNS, tag);
    for (const k in attrs) n.setAttribute(k, attrs[k]);
    return n;
  }
  const nfs = {};
  function num(v, d) {
    nfs[d] = nfs[d] || new Intl.NumberFormat('ko-KR', { minimumFractionDigits: d, maximumFractionDigits: d });
    return nfs[d].format(v);
  }
  // 원 금액: 조 · 억 · 만으로 줄여 적습니다. 10만 원보다 작으면 87,324원처럼 원 단위 그대로 적어 객단가 · CAC가 "9만"으로
  // 뭉개지지 않게 합니다. decimals를 주면 줄인 숫자의 소수 자릿수를 그 값으로 맞춥니다(1억 8,600만 원은 2면 1.86억).
  function krw(v, d) {
    const a = Math.abs(v), k = (x) => (d != null ? d : x);
    if (a >= 1e12) return [num(v / 1e12, k(1)), '조'];
    if (a >= 1e8) return [num(v / 1e8, k(a >= 1e10 ? 0 : 1)), '억'];
    if (a >= 1e5) return [num(v / 1e4, k(0)), '만'];
    return [num(v, 0), '원'];
  }
  // 달러 금액: 원래 값을 받아 K · M · B로 줄이고 소수 한 자리로 맞춥니다($745.6K). 접미사는 단위처럼 작게 씁니다.
  function usd(v, d) {
    const a = Math.abs(v), s = v < 0 ? '−' : '';
    const [div, u] = a >= 1e9 ? [1e9, 'B'] : a >= 1e6 ? [1e6, 'M'] : a >= 1e3 ? [1e3, 'K'] : [1, ''];
    return [`${s}$${num(a / div, d != null ? d : div > 1 ? 1 : 0)}`, u];
  }
  // format: eok(억 단위 값) · krw(원 단위 값) · usd(달러 값) · count(unit으로 단위) · pct(증감을 %p로) · days
  // 축이나 탭처럼 카드 안의 다른 객체로 만들 때는 두 번째 인자로 카드 데이터를 주어 lang을 잇습니다.
  function format(d, root) {
    const dec = (v, fallback) => (d.decimals != null ? d.decimals : fallback);
    const F = (o) => Object.assign(o, { lang: (root || d).lang });
    switch (d.format) {
      case 'eok': return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) || Math.abs(v) >= 100 ? 0 : 1)), '억'], rel: true });
      case 'krw': return F({ parts: (v) => krw(v, d.decimals), rel: true });
      case 'usd': return F({ parts: (v) => usd(v, d.decimals), rel: true, usd: true });
      case 'pct': return F({ parts: (v) => [num(v, dec(v, 1)), '%'], rel: false, dp: d.decimals });
      case 'days': return F({ parts: (v) => [num(v, dec(v, 1)), tx(root || d, '일')], rel: true });
      default: return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) ? 0 : 1)), d.unit || ''], rel: true });
    }
  }
  const show = (f, v) => withUnit(...f.parts(v));
  // 그릴 수 있는 숫자인지(빈 값 · 빈 글자 · NaN 제외)
  const fin = (v) => v != null && v !== '' && isFinite(v);
  // 비율의 차이(%p) 숫자: 소수 자릿수는 decimals(비우면 한 자리)를 따릅니다. 0이 아닌 차이가 0.0으로 보일 만큼 작으면
  // 자리를 늘려(최대 세 자리) 1.12%에서 1.16%로 오른 것을 0.0%p가 아니라 0.03%p로 적습니다.
  function pp(f, diff) {
    const a = Math.abs(diff);
    let k = f.dp != null ? f.dp : 1;
    while (k < 3 && a > 1e-9 && Math.round(a * 10 ** k) === 0) k++;
    return num(a, k);
  }
  // 증감: 방향은 화살표로, 좋고 나쁨(tone)은 better(up · down · neutral)로 정합니다. 비율은 %p로 적습니다.
  function delta(f, value, base, better) {
    const diff = value - base;
    const dir = Math.abs(diff) < 1e-9 ? 0 : Math.sign(diff);
    const want = better === 'down' ? -1 : better === 'neutral' ? 0 : 1;
    const tone = dir === 0 || want === 0 ? 'flat' : dir === want ? 'good' : 'bad';
    let text;
    if (!f.rel) text = tx(f, '%s%p', pp(f, diff));
    else if (base) text = `${num(Math.abs(diff / base) * 100, 1)}%`;
    else text = show(f, Math.abs(diff));
    return { dir, tone, text };
  }
  function arrow(dir) {
    const s = svgEl('svg', { class: 'vzb-arrow', viewBox: '0 0 12 12', 'aria-hidden': 'true' });
    s.append(svgEl('path', { d: dir > 0 ? 'M3.5 8.5l5-5M4.5 3.5h4v4' : 'M3.5 3.5l5 5M8.5 4.5v4h-4' }));
    return s;
  }
  // 대표 수치(단위는 작게)
  function figure(f, value) {
    const n = el('div', 'vzb-num');
    const [text, unit] = f.parts(value);
    n.append(text);
    if (unit) n.append(el('span', 'vzb-unit', unit));
    return n;
  }
  // 증감 배지: 화살표는 오르내림, 색은 좋고 나쁨(better)입니다.
  function badge(f, value, base, better) {
    const dl = delta(f, value, base, better);
    const b = el('span', `vzb-badge vzb-${dl.tone}`);
    if (dl.dir) b.append(arrow(dl.dir), el('span', 'vzb-sr', tx(f, dl.dir > 0 ? '증가' : '감소')));
    b.append(dl.text);
    return b;
  }
  // 머리: 제목 · 기간 부제(subtitle) · 대표 수치와 바로 뒤의 증감 배지. value가 null이면 제목만 둡니다.
  // 배지는 숫자 바로 뒤에 붙어 있고, 카드가 좁아 한 줄에 다 들어가지 않으면 비교 문구만 숫자 아래 줄로 내려갑니다.
  // 내려갔는지는 카드의 vzb-wrapped로 알려, 높이가 1단위인 KPI - Number가 아래 여백으로 그 줄을 받게 합니다.
  function head(card, d, f, value, cmp, cmpText) {
    const hd = el('div', 'vzb-hd');
    hd.append(el('div', 'vzb-title', d.title || ''));
    if (d.subtitle) hd.append(el('div', 'vzb-sub', d.subtitle));
    card.append(hd);
    if (value == null) return;
    const fig = el('div', 'vzb-fig');
    const n = figure(f, value);
    fig.append(n);
    if (cmp && cmp.value != null) {
      const line = el('span', 'vzb-delta'), vs = el('span', 'vzb-vs', cmpText || tx(d, '%s 대비', cmp.label || tx(d, '비교 기간')));
      line.append(badge(f, value, cmp.value, d.better), vs);
      fig.append(line);
      onSize(fig, () => card.classList.toggle('vzb-wrapped', vs.offsetTop >= n.offsetTop + n.offsetHeight - 1));   // 숫자 칸 아래에서 시작하면 내려간 것
    }
    card.append(fig);
  }
  // 같은 종류의 카드를 모두 찾아 그립니다. 코드를 두 번 붙여 넣어도 카드는 한 번만 그립니다.
  // 데이터가 바뀌면 다시 그립니다: 페이지의 스크립트가 카드 안 JSON(<script type="application/json">)의 글자를 바꾸면 그 카드만
  // 새 요소로 갈아 끼워 새로 그립니다(필터 · 기간 컨트롤과 잇는 길). 갈아 끼운 카드에는 data-vzb-live가 붙어, 움직이는 카드는
  // 처음부터 다시 재생하지 않고 마지막 장면을 바로 그립니다. 예전 요소가 페이지에서 빠지므로 그 카드의 움직임도 함께 끝납니다.
  function mount(type, draw) {
    const watch = (card, src) => {
      if (!src || !window.MutationObserver) return;
      const mo = new MutationObserver(() => {
        mo.disconnect();
        if (!card.isConnected) return;
        const next = card.cloneNode(false);
        next.removeAttribute('data-vzb-ready');
        next.setAttribute('data-vzb-live', '');
        next.append(src);
        card.replaceWith(next);
        run();
      });
      mo.observe(src, { childList: true, characterData: true, subtree: true });
    };
    const run = () => {
      document.querySelectorAll(`.vzb-card[data-vzb="${type}"]:not([data-vzb-ready])`).forEach((card) => {
        card.setAttribute('data-vzb-ready', '');
        const src = card.querySelector('script[type="application/json"]');
        const raw = src ? src.textContent : '';
        watch(card, src);
        let d;
        try { d = JSON.parse(raw); } catch (e) {
          card.append(el('p', 'vzb-error', tx(/"lang"\s*:\s*"en"/.test(raw) ? { lang: 'en' } : null, '카드 데이터(JSON)를 읽지 못했습니다: %s', e.message)));
          return;
        }
        card.setAttribute('role', 'group');
        if (d.title) card.setAttribute('aria-label', d.title);
        // 모양만 보이기: 카드를 감싼 요소에 data-vzb-view="shape"가 있으면 이름표 없이 그림만 그립니다(갤러리 첫 화면).
        // 붙여 넣은 카드에는 이 속성이 없으므로 모두 그립니다. 값은 마우스를 올리거나 화살표 키로 그대로 볼 수 있습니다.
        if (card.closest('[data-vzb-view="shape"]')) card.classList.add('vzb-bare');
        // 어두운 바탕: JSON에 "theme": "dark"를 적으면 Vizney 검정 바탕에 노랑을 첫째 색으로 그립니다(tokens.css의 .vzb-dark).
        card.classList.toggle('vzb-dark', d.theme === 'dark');
        try { draw(card, d, format(d)); } catch (e) {
          card.append(el('p', 'vzb-error', tx(d, '카드를 그리지 못했습니다: %s', e.message)));
        }
      });
    };
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
    else run();
  }
  // 크기가 바뀔 때마다 그래픽을 다시 그립니다. 차트는 카드에 남은 높이를 채웁니다.
  function onSize(node, fn) {
    let lw = 0, lh = 0;
    new ResizeObserver((entries) => {
      const w = Math.floor(entries[0].contentRect.width), h = Math.floor(entries[0].contentRect.height);
      if (w > 0 && h > 0 && (w !== lw || h !== lh)) { lw = w; lh = h; fn(w, h); }
    }).observe(node);
  }

  /* 툴팁과 조작: 마우스는 올리면 보이고 벗어나면 숨깁니다. 터치는 탭하면 보이고 다른 곳을 탭하면 닫힙니다.
     키보드는 그래프에 포커스한 뒤 화살표로 옮깁니다(가로 그래프는 좌우, 목록은 위아래). */
  let tip, tipDark = false;
  // value는 글자나 요소입니다. side가 참이면 점 위가 아니라 세로선 옆, 그래프 윗선에 맞춰 둡니다(선 · 막대 차트).
  function showTip(x, y, label, value, side) {
    if (!tip) {
      tip = el('div', 'vzb-tip');
      tip.setAttribute('aria-hidden', 'true');
      document.body.append(tip);
    }
    tip.classList.toggle('vzb-dark', tipDark);
    tip.replaceChildren(el('span', 'vzb-tip-label', label), typeof value === 'string' ? el('span', 'vzb-tip-value', value) : value);
    tip.classList.add('is-on');
    const w = tip.offsetWidth, h = tip.offsetHeight, W = window.innerWidth, H = window.innerHeight;
    if (side) {
      // side가 'left'이면 기준선 왼쪽을 먼저, 그 밖에는 오른쪽을 먼저 씁니다. 자리가 모자라면 반대쪽으로 넘깁니다.
      const right = x + 14 + w <= W - 8, left = x - w - 14 >= 8;
      tip.style.left = `${(side === 'left' ? left || !right : !right) ? Math.max(8, x - w - 14) : x + 14}px`;
      tip.style.top = `${Math.min(Math.max(8, y), H - h - 8)}px`;
      return;
    }
    tip.style.left = `${Math.min(Math.max(8, x - w / 2), W - w - 8)}px`;
    tip.style.top = `${y - h - 12 < 8 ? y + 16 : y - h - 12}px`;
  }
  function hideTip() { if (tip) tip.classList.remove('is-on'); }
  // 툴팁 내용: { value, note } 또는 { rows: [{ tone, value, name }], note }. 값이 먼저, 이름은 뒤에 옅게 적습니다.
  function tipBody(spec, label) {
    const body = el('span', 'vzb-tip-body');
    let text = label;
    if (spec.rows) {
      const grid = el('span', 'vzb-tip-rows');
      spec.rows.forEach((r) => {
        const row = el('span', 'vzb-tip-row');
        row.append(el('i', `vzb-tip-key vzb-${r.tone}`), el('b', null, r.value), el('span', null, r.name));
        grid.append(row);
        text += `, ${r.name} ${r.value}`;
      });
      body.append(grid);
    } else {
      body.append(el('span', 'vzb-tip-value', spec.value));
      text += ` ${spec.value}`;
    }
    // note는 한 줄 글 또는 줄마다 나눈 배열입니다.
    (Array.isArray(spec.note) ? spec.note : [spec.note]).filter(Boolean).forEach((line) => {
      body.append(el('span', 'vzb-tip-note', line));
      text += `, ${line}`;
    });
    return { node: body, text };
  }
  // 범례: [{ name, tone, box(면), dot(점), dim(옅게), tick(세로 눈금) }]. 선은 짧은 선, 막대와 면은 네모, 점은 동그라미로 표시합니다.
  function legendEl(items) {
    const ul = el('ul', 'vzb-legend');
    items.forEach((it) => {
      const li = el('li');
      li.append(el('i', `vzb-key vzb-${it.tone}${it.box ? ' is-box' : ''}${it.dot ? ' is-dot' : ''}${it.dim ? ' is-dim' : ''}${it.tick ? ' is-tick' : ''}`), it.name);
      ul.append(li);
    });
    return ul;
  }
  // 글자 폭(12px 기준, size를 주면 그 크기)과 SVG 글자
  let meter;
  function textW(s, weight, family, size) {
    meter = meter || document.createElement('canvas').getContext('2d');
    meter.font = `${weight} ${size || 12}px ${family}`;
    return meter.measureText(s).width;
  }
  function svgText(x, y, s, cls, anchor) {
    const t = svgEl('text', { x, y, class: cls, 'text-anchor': anchor || 'start' });
    t.textContent = s;
    return t;
  }
  // 차이를 부호와 함께 적습니다. 비율 지표는 %p입니다.
  function signed(f, diff) {
    const s = diff > 0 ? '+' : diff < 0 ? '−' : '';
    return f.rel ? `${s}${show(f, Math.abs(diff))}` : `${s}${pp(f, diff)}%p`;
  }
  // pick(clientX, clientY) → 순번, point(순번) → { x, y, label, value, text, side }(화면 좌표), clear() → 강조 해제
  // n은 개수나 개수를 돌려주는 함수입니다. vertical이면 위아래 화살표로 옮깁니다. 돌려주는 { show(순번), at() }로 키를 더 붙입니다.
  // 아무것도 고르지 않은 채 화살표를 누르면 기간 차트는 마지막(최근)부터, vertical이나 first면 첫 순번부터 짚습니다.
  function interact(plot, n, pick, point, clear, vertical, first) {
    let idx = -1, touched = false;
    const count = () => (typeof n === 'function' ? n() : n);
    const live = el('span', 'vzb-sr');
    live.setAttribute('aria-live', 'polite');
    plot.after(live);
    const go = (i) => {
      if (i < 0 || i >= count()) return;
      idx = i;
      const p = point(i);
      tipDark = !!plot.closest('.vzb-dark');
      showTip(p.x, p.y, p.label, p.value, p.side);
      live.textContent = p.text || `${p.label} ${p.value}`;
    };
    const end = () => {
      if (idx < 0) return;
      idx = -1;
      clear();
      hideTip();
    };
    const [back, next] = vertical ? ['ArrowUp', 'ArrowDown'] : ['ArrowLeft', 'ArrowRight'];
    // pick이 -1이면(조각 밖) 강조를 풉니다.
    const at = (e) => { const i = pick(e.clientX, e.clientY); if (i < 0) end(); else go(i); };
    plot.addEventListener('pointermove', (e) => { if (e.pointerType !== 'touch' || touched) at(e); });
    plot.addEventListener('pointerdown', (e) => { if (e.pointerType === 'touch') { touched = true; at(e); } });
    plot.addEventListener('pointerleave', (e) => { if (e.pointerType !== 'touch') end(); });
    // 문서와 창에 건 듣기는 그래프가 페이지에서 빠지면(카드를 다시 그리면) 스스로 풉니다.
    const away = (e) => { if (!plot.isConnected) document.removeEventListener('pointerdown', away, true); else if (touched && !plot.contains(e.target)) { touched = false; end(); } };
    document.addEventListener('pointerdown', away, true);
    plot.addEventListener('keydown', (e) => {
      if (e.key !== back && e.key !== next) return;
      e.preventDefault();
      const last = count() - 1;
      go(idx < 0 ? (vertical || first ? 0 : last) : Math.max(0, Math.min(last, idx + (e.key === next ? 1 : -1))));
    });
    plot.addEventListener('blur', end);
    const scrolled = () => { if (!plot.isConnected) window.removeEventListener('scroll', scrolled); else end(); };
    window.addEventListener('scroll', scrolled, { passive: true });
    return { show: go, at: () => idx, end };
  }

  // Budget Bars: 항목마다 한도(limit, 예: 예산)를 세로 눈금으로 세워 두고, 지금까지 쓴 양(value)을 가로 막대로 채웁니다.
  // 눈금은 막대보다 길게 그려 막대 안에 들어가도 보이고, 첫 줄의 눈금 위에 이름(limitLabel, 비우면 한도)을 한 번 적습니다.
  // 한도 안은 회색, 한도를 넘긴 구간만 빨강(--vzb-bad)입니다. 값은 오른쪽 끝에 적고, 넘친 줄은 이름과 값을 굵게 적습니다.
  // 아래 줄에는 넘친 항목 수와 넘친 양의 합을 카드가 계산해 적습니다.
  // 움직임: 눈금이 먼저 서 있고, 막대가 위에서부터 하나씩 차오르며 값이 따라 올라갑니다. 마지막 막대가 다 차면 넘친 구간만
  // 세 번 깜빡이고 멈춥니다. 막대가 차기 전에는 막대 자리를 미리 그리지 않습니다.
  // 카드가 화면에 보이면 한 번 움직이고, 다시 보기 단추로 되풀이합니다. 동작 줄이기 설정이거나 데이터가 바뀌어 다시 그릴 때
  // (data-vzb-live)는 마지막 장면을 바로 그립니다. 카드를 감싼 요소에 data-vzb-loop가 있으면(갤러리) 끝에서 잠시 쉬고 되풀이합니다.
  // 항목은 받은 순서대로 max(비우면 6)개까지 그립니다. items가 빈 배열이면 '표시할 항목이 없습니다.'만 적습니다.
  mount('bar-budget', (card, d, f) => {
    // 걸러서 남은 항목이 없을 때(items가 빈 배열)는 오류가 아니라 빈 화면입니다.
    if (Array.isArray(d.items) && !d.items.length) {
      head(card, d, f, null);
      card.append(el('p', 'vzb-bb-none', tx(d, '표시할 항목이 없습니다.')));
      return;
    }
    const all = (d.items || []).filter((it) => it && it.name != null && fin(it.value) && fin(it.limit));
    if (!all.length) throw new Error(tx(d, 'items의 value와 limit을 확인하세요.'));
    const N = Math.max(1, Math.min(all.length, fin(d.max) && +d.max >= 1 ? Math.round(+d.max) : 6));
    const items = all.slice(0, N).map((it, i) => {
      const v = Math.max(0, +it.value), lim = Math.max(0, +it.limit);
      return { i, name: String(it.name), v, lim, over: Math.max(0, v - lim) };
    });
    const label = d.limitLabel || tx(d, '한도');
    const inText = f.lang === 'en' ? label.toLowerCase() : label;   // 문장 안의 영어 이름은 소문자로
    const spill = items.filter((it) => it.over > 0);
    const sum = spill.reduce((a, it) => a + it.over, 0);
    const top = Math.max(...items.map((it) => Math.max(it.v, it.lim))) || 1;
    const summary = !spill.length ? tx(d, '모두 %s 이내', inText)
      : f.rel ? tx(d, '%1개 중 %2개 %3 초과 · +%4', items.length, spill.length, inText, show(f, sum))
        : tx(d, '%1개 중 %2개 %3 초과', items.length, spill.length, inText);

    head(card, d, f, null);
    const foot = el('div', 'vzb-foot vzb-fill vzb-bb');
    const plot = el('div', 'vzb-plot');
    plot.tabIndex = 0;
    plot.setAttribute('role', 'img');
    plot.setAttribute('aria-label', tx(d, '%1. 항목 %2개 가운데 %3개가 %4 초과입니다. 위아래 화살표로 항목을 옮깁니다. 모든 값은 이어지는 표에 있습니다.', d.title || '', items.length, spill.length, inText));
    const svg = svgEl('svg', { class: 'vzb-bb-svg', 'aria-hidden': 'true' });
    plot.append(svg);
    const ctl = el('div', 'vzb-bb-ctl'), btn = el('button', 'vzb-bb-btn'), sumEl = el('p', 'vzb-bb-sum');
    btn.type = 'button';
    btn.setAttribute('aria-label', tx(d, '처음부터 다시'));
    const ic = svgEl('svg', { class: 'vzb-bb-ic', viewBox: '0 0 16 16', 'aria-hidden': 'true' });
    ic.append(svgEl('path', { d: 'M3.7 9a4.4 4.4 0 1 0 1.2-4.2' }), svgEl('path', { d: 'M4.6 2.6v2.6h2.6' }));
    btn.append(ic);
    if (spill.length) sumEl.append(el('i', 'vzb-bb-key'));
    sumEl.append(summary);
    ctl.append(btn, sumEl);
    foot.append(plot, ctl);
    card.append(foot);
    // 화면 낭독기용 표: 이름 · 값 · 한도 · 차이
    const sr = el('div', 'vzb-sr'), table = el('table'), hr = el('tr');
    hr.append(...[tx(d, '이름'), tx(d, '값'), label, tx(d, '차이')].map((k) => el('th', null, k)));
    table.append(el('caption', null, d.title || ''), hr);
    items.forEach((it) => { const tr = el('tr'); tr.append(el('th', null, it.name), el('td', null, show(f, it.v)), el('td', null, show(f, it.lim)), el('td', null, signed(f, it.v - it.lim))); table.append(tr); });
    sr.append(table);
    card.append(sr);

    // 값 글자는 숫자와 단위로 나눠 두고, 차오르는 동안 숫자만 같은 단위로 올립니다(5,800만으로 가는 길에 원 단위로 바뀌지 않게).
    items.forEach((it) => {
      const [text, unit] = f.parts(it.v), m = /^(\D*)([\d,]*\.?\d*)(.*)$/.exec(text);
      const end = parseFloat(m[2].replace(/,/g, '')) || 0, dec = (m[2].split('.')[1] || '').length;
      it.unitText = unit;
      it.count = (k) => (k >= 1 ? text : m[1] + num(end * k, dec) + m[3]);
    });
    // 줄: 회색 막대(한도 안) · 빨간 막대(넘친 구간) · 눈금(한도) · 이름(왼쪽) · 값(오른쪽 끝)
    const limText = svgText(0, 0, label, 'vzb-bb-lim', 'middle');
    items.forEach((it) => {
      it.g = svgEl('g', { class: 'vzb-bb-row' });
      it.fill = svgEl('path', { class: 'vzb-bb-fill' });
      it.red = svgEl('path', { class: 'vzb-bb-over' });
      it.tick = svgEl('rect', { class: 'vzb-bb-tick', width: 2, rx: 1 });
      it.nm = svgText(0, 0, it.name, 'vzb-bb-name', 'end');
      it.val = svgText(0, 0, '', 'vzb-bb-val', 'end');
      it.num = document.createTextNode('');
      it.val.append(it.num, Object.assign(svgEl('tspan', { class: 'vzb-bb-unit', dx: /^[A-Za-z]{2}/.test(it.unitText) ? 3 : 1 }), { textContent: it.unitText }));
      it.g.append(it.fill, it.red, it.tick, it.nm, it.val);
      svg.append(it.g);
    });
    svg.append(limText);
    const fam = getComputedStyle(card).fontFamily, fx = (v) => v.toFixed(1), clamp = (v) => Math.max(0, Math.min(1, v));
    // 가로 막대: round가 참이면 오른쪽 끝만 둥글게, 아니면 각지게(눈금에 닿는 끝)
    const bar = (x, y, w, h, round) => {
      if (w < 0.5) return '';
      const r = round ? Math.min(4, w, h / 2) : 0;
      return `M${fx(x)},${fx(y)}H${fx(x + w - r)}A${r},${r} 0 0 1 ${fx(x + w)},${fx(y + r)}V${fx(y + h - r)}A${r},${r} 0 0 1 ${fx(x + w - r)},${fx(y + h)}H${fx(x)}Z`;
    };
    // 시간표(밀리초): 눈금만 보이는 처음 · 줄 사이 간격 · 한 줄이 차는 시간 · 다 찬 뒤 쉬는 시간 · 깜빡임 한 번 · 깜빡이는 횟수
    const T0 = 320, STAG = 320, DUR = 520, WAIT = 260, BLINK = 440, TIMES = 3, GAP = 2, TOPPAD = 16;
    const easeOut = (k) => 1 - (1 - k) ** 3;
    let W = 0, H = 0, x0 = 0, s = 1, rowH = 0, bh = 0, t = 0, FULL = 0, TOTAL = 0;
    function layout(w, h) {
      W = w; H = h;
      svg.setAttribute('width', w);
      svg.setAttribute('height', h);
      svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
      x0 = Math.ceil(Math.max(...items.map((it) => textW(it.name, 700, fam)))) + 10;
      const vW = Math.ceil(Math.max(...items.map((it) => textW(show(f, it.v), 700, fam)))) + 4;
      s = Math.max(1, w - x0 - vW - 12) / top;
      rowH = Math.min(32, Math.max(14, (h - TOPPAD) / N));
      bh = Math.max(6, Math.min(16, Math.round(rowH * 0.5)));
      const th = Math.min(rowH - 2, bh + 8);   // 눈금은 막대보다 위아래로 4px씩 깁니다
      items.forEach((it) => {
        it.y = TOPPAD + it.i * rowH + (rowH - bh) / 2;
        it.yc = it.y + bh / 2;
        it.xl = x0 + it.lim * s;
        it.tick.setAttribute('x', fx(it.xl - 1));
        it.tick.setAttribute('y', fx(it.yc - th / 2));
        it.tick.setAttribute('height', fx(th));
        it.nm.setAttribute('x', x0 - 8);
        it.nm.setAttribute('y', fx(it.yc + 4));
        it.val.setAttribute('x', w);
        it.val.setAttribute('y', fx(it.yc + 4));
        it.start = T0 + it.i * STAG;
      });
      // 눈금 이름은 첫 줄의 눈금 위에 한 번만 적습니다. 그래프 밖으로 나가지 않게 당깁니다.
      const lw = textW(label, 400, fam, 11) / 2;
      limText.setAttribute('x', fx(Math.max(lw, Math.min(w - lw, items[0].xl))));
      limText.setAttribute('y', fx(items[0].yc - th / 2 - 5));
      FULL = T0 + (N - 1) * STAG + DUR;
      TOTAL = FULL + (spill.length ? WAIT + BLINK * TIMES : 240);
      render();
    }
    function render() {
      if (!W) return;
      // 깜빡임: 다 찬 뒤 넘친 구간만 옅어졌다 돌아오기를 되풀이하고, 진한 상태로 끝납니다.
      const b = (t - FULL - WAIT) / BLINK;
      const glow = b <= 0 || b >= TIMES || holds.size ? 1 : 1 - 0.84 * (0.5 - 0.5 * Math.cos(2 * Math.PI * b));   // 멈춰서 값을 읽는 동안은 진하게
      const alert = t >= FULL + WAIT;
      items.forEach((it) => {
        const k = clamp((t - it.start) / DUR), e = easeOut(k), xv = x0 + it.v * s * e;
        // 한도에 닿은 막대는 눈금 앞에서 각지게 끊고, 넘친 만큼은 눈금 뒤에서 빨강으로 잇습니다.
        const hit = it.v >= it.lim && xv >= it.xl - GAP;
        it.fill.setAttribute('d', bar(x0, it.y, (hit ? it.xl - GAP : xv) - x0, bh, !hit));
        it.red.setAttribute('d', it.over > 0 ? bar(it.xl + GAP, it.y, xv - it.xl - GAP, bh, true) : '');
        it.red.setAttribute('opacity', fx(glow));
        it.num.nodeValue = it.count(e);
        it.val.setAttribute('opacity', fx(clamp((t - it.start) / 140)));
        it.g.classList.toggle('is-over', it.over > 0 && alert);
      });
      sumEl.style.opacity = fx(clamp((t - (spill.length ? FULL + WAIT : FULL)) / 240));
    }

    // 재생: 화면에 보이고(inView) 마우스 · 포커스로 잠시 멈추지 않았을 때(holds)만 움직입니다. 카드가 페이지에서 빠지면 멈춥니다.
    const still = !!(window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches);
    const live = card.hasAttribute('data-vzb-live');   // 데이터가 바뀌어 다시 그린 카드: 처음부터 재생하지 않습니다
    const loop = !!card.closest('[data-vzb-loop]');
    const holds = new Set();
    let playing = false, seen = false, inView = false, rest = 0, last = 0, raf = 0;
    const on = () => playing && inView && !holds.size && card.isConnected;
    function tick(stamp) {
      raf = 0;
      if (!on()) { last = 0; return; }
      const dt = last ? Math.min(100, stamp - last) : 16;
      last = stamp;
      if (rest > 0) { rest -= dt; if (rest <= 0) t = 0; }
      else {
        t = Math.min(TOTAL, t + dt);
        if (t >= TOTAL) { if (loop) rest = 2600; else playing = false; }
      }
      render();
      if (on()) raf = requestAnimationFrame(tick);
    }
    function kick() { if (!raf && on()) raf = requestAnimationFrame(tick); }
    function play() { t = 0; rest = 0; last = 0; playing = true; render(); kick(); }
    btn.addEventListener('click', play);
    plot.addEventListener('pointerenter', (e) => { if (e.pointerType === 'mouse') { holds.add('hover'); render(); } });
    plot.addEventListener('pointerleave', () => { holds.delete('hover'); kick(); });
    plot.addEventListener('focus', () => { if (plot.matches(':focus-visible')) { holds.add('focus'); render(); } });   // 키보드로 들어왔을 때만 멈춥니다
    plot.addEventListener('blur', () => { holds.delete('focus'); kick(); });
    // 툴팁: 줄을 짚으면 쓴 양, 한도, 넘치거나 남은 양과 한도 대비 비율을 보여 주고 그 줄만 진하게 남깁니다.
    interact(plot, items.length,
      (cx, cy) => {
        const i = Math.floor((cy - plot.getBoundingClientRect().top - TOPPAD) / rowH);
        return i >= 0 && i < items.length ? i : -1;
      },
      (k) => {
        const it = items[k], r = plot.getBoundingClientRect();
        svg.classList.add('is-hover');
        items.forEach((x) => x.g.classList.toggle('is-on', x === it));
        const gap = show(f, Math.abs(it.v - it.lim)), share = it.lim > 0 ? num((it.v / it.lim) * 100, 0) : null;
        const note = [`${label} ${show(f, it.lim)}`];
        if (share == null) note.push(tx(d, '%s 초과|한도', gap));
        else note.push(it.v > it.lim ? tx(d, '%1 초과 · %2의 %3%', gap, inText, share) : it.v < it.lim ? tx(d, '%1 남음 · %2의 %3%', gap, inText, share) : tx(d, '%1의 %2', inText, `${share}%`));
        const tb = tipBody({ value: show(f, it.v), note }, it.name);
        return { x: r.left + x0 + Math.max(it.v, it.lim) * s, y: r.top + it.y - 6, label: it.name, value: tb.node, text: tb.text, side: true };
      },
      () => { svg.classList.remove('is-hover'); items.forEach((x) => x.g.classList.remove('is-on')); },
      true, true);

    btn.hidden = still;
    if (still || live) t = 1e9;   // 마지막 장면에서 시작합니다
    onSize(plot, layout);
    if (document.fonts && document.fonts.ready) document.fonts.ready.then(() => { if (W) layout(W, H); });
    // 화면에 처음 보일 때 움직입니다. 화면 밖에 있는 동안은 멈춰 둡니다.
    const start = () => { if (!seen) { seen = true; if (!still && !live) play(); } };
    if ('IntersectionObserver' in window) {
      const io = new IntersectionObserver((es) => {
        if (!card.isConnected) { io.disconnect(); return; }
        inView = es[es.length - 1].isIntersecting;
        if (inView) start();
        kick();
      }, { threshold: 0.3 });
      io.observe(card);
    } else { inView = true; start(); }
  });
})();
