  };
  // tx(d 또는 f, 원문, 값…): d.lang(format()이 f에 옮겨 둡니다)이 en이면 영어 문구를, 아니면 원문을 돌려줍니다.
  function tx(x, s, ...a) {
    let i = 0;
    const out = x && x.lang === 'en' && EN[s] !== undefined ? EN[s] : s.split('|')[0];
    return out.replace(/%(\d|s)/g, (m, k) => String(a[k === 's' ? i++ : k - 1]));
  }
  // 숫자 뒤의 단위: 영어 단위(days · orders)는 한 칸 띄우고, 기호와 한 글자 단위(% · K · 억)는 붙입니다.
  const withUnit = (n, u) => (/^[A-Za-z]{2}/.test(u) ? `${n} ${u}` : n + u);
  function el(tag, cls, text) {
    const n = document.createElement(tag);
    if (cls) n.className = cls;
    if (text != null) n.textContent = text;
    return n;
  }
  function svgEl(tag, attrs) {
    const n = document.createElementNS(SVGNS, tag);
    for (const k in attrs) n.setAttribute(k, attrs[k]);
    return n;
  }
  const nfs = {};
  function num(v, d) {
    nfs[d] = nfs[d] || new Intl.NumberFormat('ko-KR', { minimumFractionDigits: d, maximumFractionDigits: d });
    return nfs[d].format(v);
  }
  function krw(v) {
    const a = Math.abs(v);
    if (a >= 1e12) return [num(v / 1e12, 1), '조'];
    if (a >= 1e8) return [num(v / 1e8, a >= 1e10 ? 0 : 1), '억'];
    if (a >= 1e4) return [num(v / 1e4, 0), '만'];
    return [num(v, 0), '원'];
  }
  // 달러 금액: 원래 값을 받아 K · M · B로 줄이고 소수 한 자리로 맞춥니다($745.6K). 접미사는 단위처럼 작게 씁니다.
  function usd(v, d) {
    const a = Math.abs(v), s = v < 0 ? '−' : '';
    const [div, u] = a >= 1e9 ? [1e9, 'B'] : a >= 1e6 ? [1e6, 'M'] : a >= 1e3 ? [1e3, 'K'] : [1, ''];
    return [`${s}$${num(a / div, d != null ? d : div > 1 ? 1 : 0)}`, u];
  }
  // format: eok(억 단위 값) · krw(원 단위 값) · usd(달러 값) · count(unit으로 단위) · pct(증감을 %p로) · days
  // 축이나 탭처럼 카드 안의 다른 객체로 만들 때는 두 번째 인자로 카드 데이터를 주어 lang을 잇습니다.
  function format(d, root) {
    const dec = (v, fallback) => (d.decimals != null ? d.decimals : fallback);
    const F = (o) => Object.assign(o, { lang: (root || d).lang });
    switch (d.format) {
      case 'eok': return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) || Math.abs(v) >= 100 ? 0 : 1)), '억'], rel: true });
      case 'krw': return F({ parts: krw, rel: true });
      case 'usd': return F({ parts: (v) => usd(v, d.decimals), rel: true, usd: true });
      case 'pct': return F({ parts: (v) => [num(v, dec(v, 1)), '%'], rel: false });
      case 'days': return F({ parts: (v) => [num(v, dec(v, 1)), tx(root || d, '일')], rel: true });
      default: return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) ? 0 : 1)), d.unit || ''], rel: true });
    }
  }
  const show = (f, v) => withUnit(...f.parts(v));
  // 그릴 수 있는 숫자인지(빈 값 · 빈 글자 · NaN 제외)
  const fin = (v) => v != null && v !== '' && isFinite(v);
  // 증감: 방향은 화살표로, 좋고 나쁨(tone)은 better(up · down · neutral)로 정합니다. 비율은 %p로 적습니다.
  function delta(f, value, base, better) {
    const diff = value - base;
    const dir = Math.abs(diff) < 1e-9 ? 0 : Math.sign(diff);
    const want = better === 'down' ? -1 : better === 'neutral' ? 0 : 1;
    const tone = dir === 0 || want === 0 ? 'flat' : dir === want ? 'good' : 'bad';
    let text;
    if (!f.rel) text = tx(f, '%s%p', num(Math.abs(diff), 1));
    else if (base) text = `${num(Math.abs(diff / base) * 100, 1)}%`;
    else text = show(f, Math.abs(diff));
    return { dir, tone, text };
  }
  function arrow(dir) {
    const s = svgEl('svg', { class: 'vzb-arrow', viewBox: '0 0 12 12', 'aria-hidden': 'true' });
    s.append(svgEl('path', { d: dir > 0 ? 'M3.5 8.5l5-5M4.5 3.5h4v4' : 'M3.5 3.5l5 5M8.5 4.5v4h-4' }));
    return s;
  }
  // 대표 수치(단위는 작게)

  }
  // 증감 배지: 화살표는 오르내림, 색은 좋고 나쁨(better)입니다.
  function badge(f, value, base, better) {
    const dl = delta(f, value, base, better);
    const b = el('span', `vzb-badge vzb-${dl.tone}`);
    if (dl.dir) b.append(arrow(dl.dir), el('span', 'vzb-sr', tx(f, dl.dir > 0 ? '증가' : '감소')));
    b.append(dl.text);
    return b;
  }
  // 머리: 제목 · 기간 부제(subtitle) · 대표 수치와 바로 뒤의 증감 배지. value가 null이면 제목만 둡니다.
  function head(card, d, f, value, cmp, cmpText) {
    const hd = el('div', 'vzb-hd');

  // 같은 종류의 카드를 모두 찾아 그립니다. 코드를 두 번 붙여 넣어도 카드는 한 번만 그립니다.
  function mount(type, draw) {
    const run = () => {
      document.querySelectorAll(`.vzb-card[data-vzb="${type}"]:not([data-vzb-ready])`).forEach((card) => {
        card.setAttribute('data-vzb-ready', '');
        const src = card.querySelector('script[type="application/json"]');
        const raw = src ? src.textContent : '';
        let d;
        try { d = JSON.parse(raw); } catch (e) {
          card.append(el('p', 'vzb-error', tx(/"lang"\s*:\s*"en"/.test(raw) ? { lang: 'en' } : null, '카드 데이터(JSON)를 읽지 못했습니다: %s', e.message)));
          return;
        }
        card.setAttribute('role', 'group');
        if (d.title) card.setAttribute('aria-label', d.title);
        try { draw(card, d, format(d)); } catch (e) {
          card.append(el('p', 'vzb-error', tx(d, '카드를 그리지 못했습니다: %s', e.message)));
        }
      });
    };
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
    else run();
  }
  // 크기가 바뀔 때마다 그래픽을 다시 그립니다. 차트는 카드에 남은 높이를 채웁니다.
  function onSize(node, fn) {
    let lw = 0, lh = 0;
    new ResizeObserver((entries) => {
      const w = Math.floor(entries[0].contentRect.width), h = Math.floor(entries[0].contentRect.height);
      if (w > 0 && h > 0 && (w !== lw || h !== lh)) { lw = w; lh = h; fn(w, h); }
    }).observe(node);
  }

  /* 툴팁과 조작: 마우스는 올리면 보이고 벗어나면 숨깁니다. 터치는 탭하면 보이고 다른 곳을 탭하면 닫힙니다.
     키보드는 그래프에 포커스한 뒤 화살표로 옮깁니다(가로 그래프는 좌우, 목록은 위아래). */
  let tip;
  // value는 글자나 요소입니다. side가 참이면 점 위가 아니라 세로선 옆, 그래프 윗선에 맞춰 둡니다(선 · 막대 차트).
  function showTip(x, y, label, value, side) {
    if (!tip) {
      tip = el('div', 'vzb-tip');
      tip.setAttribute('aria-hidden', 'true');
      document.body.append(tip);
    }
    tip.replaceChildren(el('span', 'vzb-tip-label', label), typeof value === 'string' ? el('span', 'vzb-tip-value', value) : value);
    tip.classList.add('is-on');
    const w = tip.offsetWidth, h = tip.offsetHeight, W = window.innerWidth, H = window.innerHeight;
    if (side) {
      // side가 'left'이면 기준선 왼쪽을 먼저, 그 밖에는 오른쪽을 먼저 씁니다. 자리가 모자라면 반대쪽으로 넘깁니다.
      const right = x + 14 + w <= W - 8, left = x - w - 14 >= 8;
      tip.style.left = `${(side === 'left' ? left || !right : !right) ? Math.max(8, x - w - 14) : x + 14}px`;
      tip.style.top = `${Math.min(Math.max(8, y), H - h - 8)}px`;
      return;
    }
    tip.style.left = `${Math.min(Math.max(8, x - w / 2), W - w - 8)}px`;
    tip.style.top = `${y - h - 12 < 8 ? y + 16 : y - h - 12}px`;
  }
  function hideTip() { if (tip) tip.classList.remove('is-on'); }
  // 툴팁 내용: { value, note } 또는 { rows: [{ tone, value, name }], note }. 값이 먼저, 이름은 뒤에 옅게 적습니다.
  function tipBody(spec, label) {
    const body = el('span', 'vzb-tip-body');

    if (spec.rows) {

  // 범례: [{ name, tone, box(면), dot(점), dim(옅게), tick(세로 눈금) }]. 선은 짧은 선, 막대와 면은 네모, 점은 동그라미로 표시합니다.

    items.forEach((it) => {
      const li = el('li');
      li.append(el('i', `vzb-key vzb-${it.tone}${it.box ? ' is-box' : ''}${it.dot ? ' is-dot' : ''}${it.dim ? ' is-dim' : ''}${it.tick ? ' is-tick' : ''}`), it.name);
      ul.append(li);

  }
  // 글자 폭(12px 기준, size를 주면 그 크기)과 SVG 글자
  let meter;
  function textW(s, weight, family, size) {
    meter = meter || document.createElement('canvas').getContext('2d');
    meter.font = `${weight} ${size || 12}px ${family}`;
    return meter.measureText(s).width;
  }
  function svgText(x, y, s, cls, anchor) {
    const t = svgEl('text', { x, y, class: cls, 'text-anchor': anchor || 'start' });
    t.textContent = s;
    return t;
  }
  // 차이를 부호와 함께 적습니다. 비율 지표는 %p입니다.
  function signed(f, diff) {
    const s = diff > 0 ? '+' : diff < 0 ? '−' : '';
    return f.rel ? `${s}${show(f, Math.abs(diff))}` : `${s}${num(Math.abs(diff), 1)}%p`;
  }
  // pick(clientX, clientY) → 순번, point(순번) → { x, y, label, value, text, side }(화면 좌표), clear() → 강조 해제
  // n은 개수나 개수를 돌려주는 함수입니다. vertical이면 위아래 화살표로 옮깁니다. 돌려주는 { show(순번), at() }로 키를 더 붙입니다.
  // 아무것도 고르지 않은 채 화살표를 누르면 기간 차트는 마지막(최근)부터, vertical이나 first면 첫 순번부터 짚습니다.
  function interact(plot, n, pick, point, clear, vertical, first) {
    let idx = -1, touched = false;
    const count = () => (typeof n === 'function' ? n() : n);
    const live = el('span', 'vzb-sr');
    live.setAttribute('aria-live', 'polite');
    plot.after(live);
    const go = (i) => {
      if (i < 0 || i >= count()) return;
      idx = i;
      const p = point(i);
      showTip(p.x, p.y, p.label, p.value, p.side);
      live.textContent = p.text || `${p.label} ${p.value}`;
    };
    const end = () => {
      if (idx < 0) return;
      idx = -1;
      clear();
      hideTip();
    };
    const [back, next] = vertical ? ['ArrowUp', 'ArrowDown'] : ['ArrowLeft', 'ArrowRight'];
    // pick이 -1이면(조각 밖) 강조를 풉니다.
    const at = (e) => { const i = pick(e.clientX, e.clientY); if (i < 0) end(); else go(i); };
    plot.addEventListener('pointermove', (e) => { if (e.pointerType !== 'touch' || touched) at(e); });
    plot.addEventListener('pointerdown', (e) => { if (e.pointerType === 'touch') { touched = true; at(e); } });
    plot.addEventListener('pointerleave', (e) => { if (e.pointerType !== 'touch') end(); });
    document.addEventListener('pointerdown', (e) => { if (touched && !plot.contains(e.target)) { touched = false; end(); } }, true);
    plot.addEventListener('keydown', (e) => {
      if (e.key !== back && e.key !== next) return;
      e.preventDefault();
      const last = count() - 1;
      go(idx < 0 ? (vertical || first ? 0 : last) : Math.max(0, Math.min(last, idx + (e.key === next ? 1 : -1))));
    });
    plot.addEventListener('blur', end);
    window.addEventListener('scroll', end, { passive: true });
    return { show: go, at: () => idx, end };
  }

  /* 가로 막대 공통: 항목마다 한 줄. 큰 순서로 정렬하고(sort: false면 적은 순서), max를 넘는 항목은 '기타 n개' 한 줄로 묶습니다.
     이름을 막대 안에 얹는 줄(overlay)과 이름 · 막대 · 값을 칸으로 나눈 줄(column)이 있습니다.
     줄에 마우스를 올리거나, 목록에 포커스한 뒤 위아래 화살표를 누르면 툴팁으로 자세한 값을 보여줍니다. */

  // 막대 길이(0–100%)
  const share = (v, max) => `${(Math.max(0, Math.min(1, max > 0 ? v / max : 0)) * 100).toFixed(2)}%`;
  // 이름을 막대 안에 얹은 칸. mark가 있으면 그 자리(비교 값)에 세로 눈금을 긋습니다.
  function overlayBar(name, width, mark) {
    const bar = el('div', 'vzb-li-bar');
    const fill = el('i', 'vzb-li-fill');
    fill.style.width = width;
    bar.append(fill);
    if (mark != null) {
      const m = el('i', 'vzb-li-mark');
      m.style.left = mark;
      bar.append(m);
    }
    bar.append(el('span', 'vzb-li-name', name));
    return bar;
  }
  // 묶인 '기타' 줄의 이름 칸(막대 없음)
  const restName = (it) => el('span', 'vzb-li-name is-rest', it.name);
  // 묶을 항목들의 이름과 값의 합. keys에 적은 값도 함께 더합니다.
  function foldRest(d, rest, keys) {
    const it = { rest: true, name: tx(d, '기타 %s개', rest.length), note: rest.map((x) => x.name).join('·') };
    keys.forEach((k) => { it[k] = rest.reduce((a, x) => a + (+x[k] || 0), 0); });
    return it;
  }

  // cfg: items · layout(overlay | column) · cols(열 너비, CSS grid) · columns([이름 열 제목, 값 열 제목]) · legend · top
  //   max(보여줄 줄 수, 넘으면 기타로 묶음) · order(정렬 함수) · fold(rest) · prepare(shown) → ctx · cells(item, ctx) → 칸 요소들
  //   tip(item, ctx) → { value, note } 또는 { rows, note } · f(값 표기)
  // 돌려주는 show(cfg 일부)로 같은 카드에 다른 목록을 다시 그립니다(탭 카드).
  function listCard(card, d, f, cfg) {
    head(card, d, f, null);
    if (cfg.top) card.append(cfg.top);
    if (cfg.legend) card.append(legendEl(cfg.legend));
    const foot = el('div', 'vzb-foot vzb-li');
    const cols = el('div', 'vzb-li-cols');
    const list = el('div', 'vzb-li-list');
    list.tabIndex = 0;
    list.setAttribute('role', 'list');
    foot.append(cols, list);
    card.append(foot);
    let S, ctx, rows = [], shown = [];

    function use(next) {
      S = Object.assign({ f }, cfg, next);
      let items = (S.items || []).filter((it) => it && it.name != null);
      if (S.order) items = items.slice().sort(S.order);
      else if (S.sort !== false) items = items.slice().sort((a, b) => (+b.value || 0) - (+a.value || 0));
      const max = S.max || 5;
      shown = items.length > max + 1 && S.fold ? items.slice(0, max).concat([S.fold(items.slice(max))]) : items;
      cols.replaceChildren(...(S.columns || []).map((t) => el('span', null, t)));
      list.className = `vzb-li-list is-${S.layout || 'overlay'}`;
      list.style.gridTemplateColumns = S.cols || 'minmax(0, 1fr) auto';
      list.setAttribute('aria-label', tx(d, '%s 목록, %s줄. 위아래 화살표로 항목의 자세한 값을 봅니다.', S.name || d.title || '', shown.length));
      ctx = Object.assign({ f: S.f }, S.prepare ? S.prepare(shown) : {});
      rows = shown.map((it) => {
        const row = el('div', `vzb-li-row${it.rest ? ' is-rest' : ''}`);
        row.setAttribute('role', 'listitem');
        row.append(...S.cells(it, ctx));
        return row;
      });
      list.replaceChildren(...rows);
    }
    use({});

    const api = interact(list, () => rows.length,
      (cx, cy) => {
        let best = 0, gap = Infinity;
        rows.forEach((r, i) => {
          const b = r.getBoundingClientRect(), dist = cy < b.top ? b.top - cy : cy > b.bottom ? cy - b.bottom : 0;
          if (dist < gap) { gap = dist; best = i; }
        });
        return best;
      },
      (i) => {
        rows.forEach((r, k) => r.classList.toggle('is-on', k === i));
        const it = shown[i], t = tipBody(S.tip(it, ctx), it.rest && it.note ? `${it.name} (${it.note})` : it.name);
        const b = rows[i].getBoundingClientRect();
        return { x: b.left + b.width / 2, y: b.top, label: it.rest && it.note ? `${it.name} (${it.note})` : it.name, value: t.node, text: t.text };
      },
      () => rows.forEach((r) => r.classList.remove('is-on')),
      true);
    return {
      show(next) {
        api.end();
        use(next);
      },
    };
  }

  // Bar - Ranking: 이름을 막대 안에 얹고 값은 오른쪽에. 툴팁에 전체 중 비중을 적습니다.
  function basicList(card, d, f, cfg) {
    return listCard(card, d, f, Object.assign({
      layout: 'overlay',
      cols: 'minmax(0, 1fr) auto',
      fold: (rest) => foldRest(d, rest, ['value']),
      prepare: (shown) => ({
        max: Math.max(0, ...shown.filter((x) => !x.rest).map((x) => +x.value || 0)),
        total: shown.reduce((a, x) => a + (+x.value || 0), 0),
      }),
      cells: (it, c) => [it.rest ? restName(it) : overlayBar(it.name, share(it.value, c.max)), el('b', 'vzb-li-val', show(c.f, it.value))],
      tip: (it, c) => ({ value: show(c.f, it.value), note: c.total ? tx(d, '전체의 %s', `${num((it.value / c.total) * 100, 1)}%`) : '' }),
    }, cfg));
  }

  // Bar - Compare: 이름을 얹은 막대 위에 비교 값(prev)을 세로 눈금으로 긋고, 오른쪽에 값과 KPI와 같은 증감 배지를 둡니다.
  // 묶인 기타 줄도 합계끼리 비교한 증감을 적습니다.
  mount('bar-compare', (card, d, f) => {
    const now = d.currentLabel || tx(d, '이번 기간'), base = d.compareLabel || tx(d, '비교 기간|범례');
    listCard(card, d, f, {
      items: d.items || [],
      max: d.max,
      sort: d.sort,
      layout: 'overlay',
      cols: 'minmax(0, 1fr) auto auto',
      legend: [{ name: now, tone: 'fs', box: true }, { name: base, tone: 'ti', tick: true }],
      fold: (rest) => foldRest(d, rest, ['value', 'prev']),
      prepare: (shown) => ({ max: Math.max(0, ...shown.filter((x) => !x.rest).map((x) => Math.max(+x.value || 0, +x.prev || 0))) }),
      cells: (it, c) => {
        const b = it.prev != null ? badge(f, it.value, it.prev, d.better) : el('span');
        b.classList.add('vzb-li-badge');
        const bar = it.rest ? restName(it) : overlayBar(it.name, share(it.value, c.max), it.prev == null ? null : share(it.prev, c.max));
        return [bar, el('b', 'vzb-li-val', show(f, it.value)), b];
      },
      tip: (it) => {
        const rows = [{ tone: 'fs', name: now, value: show(f, it.value) }];
        if (it.prev == null) return { rows };
        rows.push({ tone: 'f0', name: base, value: show(f, it.prev) });
        const dl = delta(f, it.value, it.prev, d.better);
        return { rows, note: tx(d, '%s 대비 %s', base, `${dl.dir > 0 ? '+' : dl.dir < 0 ? '−' : ''}${dl.text}`) };
      },
    });
  });
})();
</script>
