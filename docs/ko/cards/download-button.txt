    '%1. %2 %3 %4, %5 %6. 좌우 화살표로 기간을 옮기고 Enter로 짚을 선을 바꿉니다. 모든 값은 이어지는 표에 있습니다.': '%1. In %2, %3 %4 and %5 %6. Use the left and right arrows to move between periods and Enter to switch the highlighted line. Every value is in the table that follows.',
    '%1. %2 %3 %4. 좌우 화살표로 기간을 옮깁니다. 모든 값은 이어지는 표에 있습니다.': '%1. In %2, %3 %4. Use the left and right arrows to move between periods. Every value is in the table that follows.',
    '카드 데이터(JSON)를 읽지 못했습니다: %s': 'Could not read the card data (JSON): %s', '카드를 그리지 못했습니다: %s': 'Could not draw the card: %s',
  };
  // tx(d 또는 f, 원문, 값…): d.lang(format()이 f에 옮겨 둡니다)이 en이면 영어 문구를, 아니면 원문을 돌려줍니다.
  function tx(x, s, ...a) {
    let i = 0;
    const out = x && x.lang === 'en' && EN[s] !== undefined ? EN[s] : s.split('|')[0];
    return out.replace(/%(\d|s)/g, (m, k) => String(a[k === 's' ? i++ : k - 1]));
  }
  // 숫자 뒤의 단위: 영어 단위(days · orders)는 한 칸 띄우고, 기호와 한 글자 단위(% · K · 억)는 붙입니다.
  const withUnit = (n, u) => (/^[A-Za-z]{2}/.test(u) ? `${n} ${u}` : n + u);
  function el(tag, cls, text) {
    const n = document.createElement(tag);
    if (cls) n.className = cls;
    if (text != null) n.textContent = text;
    return n;
  }
  function svgEl(tag, attrs) {
    const n = document.createElementNS(SVGNS, tag);
    for (const k in attrs) n.setAttribute(k, attrs[k]);
    return n;
  }
  const nfs = {};
  function num(v, d) {
    nfs[d] = nfs[d] || new Intl.NumberFormat('ko-KR', { minimumFractionDigits: d, maximumFractionDigits: d });
    return nfs[d].format(v);
  }
  // 원 금액: 조 · 억 · 만으로 줄여 적습니다. 10만 원보다 작으면 87,324원처럼 원 단위 그대로 적어 객단가 · CAC가 "9만"으로
  // 뭉개지지 않게 합니다. decimals를 주면 줄인 숫자의 소수 자릿수를 그 값으로 맞춥니다(1억 8,600만 원은 2면 1.86억).
  function krw(v, d) {
    const a = Math.abs(v), k = (x) => (d != null ? d : x);
    if (a >= 1e12) return [num(v / 1e12, k(1)), '조'];
    if (a >= 1e8) return [num(v / 1e8, k(a >= 1e10 ? 0 : 1)), '억'];
    if (a >= 1e5) return [num(v / 1e4, k(0)), '만'];
    return [num(v, 0), '원'];
  }
  // 달러 금액: 원래 값을 받아 K · M · B로 줄이고 소수 한 자리로 맞춥니다($745.6K). 접미사는 단위처럼 작게 씁니다.
  function usd(v, d) {
    const a = Math.abs(v), s = v < 0 ? '-' : '';
    const [div, u] = a >= 1e9 ? [1e9, 'B'] : a >= 1e6 ? [1e6, 'M'] : a >= 1e3 ? [1e3, 'K'] : [1, ''];
    return [`${s}$${num(a / div, d != null ? d : div > 1 ? 1 : 0)}`, u];
  }
  // format: eok(억 단위 값) · krw(원 단위 값) · usd(달러 값) · count(unit으로 단위) · pct(증감을 %p로) · days
  // 축이나 탭처럼 카드 안의 다른 객체로 만들 때는 두 번째 인자로 카드 데이터를 주어 lang을 잇습니다.
  function format(d, root) {
    const dec = (v, fallback) => (d.decimals != null ? d.decimals : fallback);
    const F = (o) => Object.assign(o, { lang: (root || d).lang });
    switch (d.format) {
      case 'eok': return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) || Math.abs(v) >= 100 ? 0 : 1)), '억'], rel: true });
      case 'krw': return F({ parts: (v) => krw(v, d.decimals), rel: true });
      case 'usd': return F({ parts: (v) => usd(v, d.decimals), rel: true, usd: true });
      case 'pct': return F({ parts: (v) => [num(v, dec(v, 1)), '%'], rel: false, dp: d.decimals });
      case 'days': return F({ parts: (v) => [num(v, dec(v, 1)), tx(root || d, '일')], rel: true });
      default: return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) ? 0 : 1)), d.unit || ''], rel: true });
    }
  }
  const show = (f, v) => withUnit(...f.parts(v));
  // 그릴 수 있는 숫자인지(빈 값 · 빈 글자 · NaN 제외)
  const fin = (v) => v != null && v !== '' && isFinite(v);
  // 비율의 차이(%p) 숫자: 소수 자릿수는 decimals(비우면 한 자리)를 따릅니다. 0이 아닌 차이가 0.0으로 보일 만큼 작으면
  // 자리를 늘려(최대 세 자리) 1.12%에서 1.16%로 오른 것을 0.0%p가 아니라 0.03%p로 적습니다.
  function pp(f, diff) {
    const a = Math.abs(diff);
    let k = f.dp != null ? f.dp : 1;
    while (k < 3 && a > 1e-9 && Math.round(a * 10 ** k) === 0) k++;
    return num(a, k);
  }
  // 증감: 방향은 화살표로, 좋고 나쁨(tone)은 better(up · down · neutral)로 정합니다. 비율은 %p로 적습니다.
  function delta(f, value, base, better) {
    const diff = value - base;
    const dir = Math.abs(diff) < 1e-9 ? 0 : Math.sign(diff);
    const want = better === 'down' ? -1 : better === 'neutral' ? 0 : 1;
    const tone = dir === 0 || want === 0 ? 'flat' : dir === want ? 'good' : 'bad';
    let text;
    if (!f.rel) text = tx(f, '%s%p', pp(f, diff));
    else if (base) text = `${num(Math.abs(diff / base) * 100, 1)}%`;
    else text = show(f, Math.abs(diff));
    return { dir, tone, text };
  }
  function arrow(dir) {
    const s = svgEl('svg', { class: 'vzb-arrow', viewBox: '0 0 12 12', 'aria-hidden': 'true' });
    s.append(svgEl('path', { d: dir > 0 ? 'M3.5 8.5l5-5M4.5 3.5h4v4' : 'M3.5 3.5l5 5M8.5 4.5v4h-4' }));
    return s;
  }
  // 대표 수치(단위는 작게)
  function figure(f, value) {
    const n = el('div', 'vzb-num');
    const [text, unit] = f.parts(value);
    n.append(text);
    if (unit) n.append(el('span', 'vzb-unit', unit));
    return n;
  }
  // 증감 배지: 화살표는 오르내림, 색은 좋고 나쁨(better)입니다.
  function badge(f, value, base, better) {
    const dl = delta(f, value, base, better);
    const b = el('span', `vzb-badge vzb-${dl.tone}`);
    if (dl.dir) b.append(arrow(dl.dir), el('span', 'vzb-sr', tx(f, dl.dir > 0 ? '증가' : '감소')));
    b.append(dl.text);
    return b;
  }
  // 머리: 제목 · 기간 부제(subtitle) · 대표 수치와 바로 뒤의 증감 배지. value가 null이면 제목만 둡니다.
  // 배지는 숫자 바로 뒤에 붙어 있고, 카드가 좁아 한 줄에 다 들어가지 않으면 비교 문구만 숫자 아래 줄로 내려갑니다.
  // 내려갔는지는 카드의 vzb-wrapped로 알려, 높이가 1단위인 카드가 아래 여백으로 그 줄을 받을 수 있게 합니다.
  function head(card, d, f, value, cmp, cmpText) {
    const hd = el('div', 'vzb-hd');
    hd.append(el('div', 'vzb-title', d.title || ''));
    if (d.subtitle) hd.append(el('div', 'vzb-sub', d.subtitle));
    card.append(hd);
    if (value == null) return;
    const fig = el('div', 'vzb-fig');
    const n = figure(f, value);
    fig.append(n);
    if (cmp && cmp.value != null) {
      const line = el('span', 'vzb-delta'), vs = el('span', 'vzb-vs', cmpText || tx(d, '%s 대비', cmp.label || tx(d, '비교 기간')));
      line.append(badge(f, value, cmp.value, d.better), vs);
      fig.append(line);
      onSize(fig, () => card.classList.toggle('vzb-wrapped', vs.offsetTop >= n.offsetTop + n.offsetHeight - 1));   // 숫자 칸 아래에서 시작하면 내려간 것
    }
    card.append(fig);
  }
  // 움직임 값(tokens.css의 --vzb-follow … --vzb-rest). 스프링은 '세기 잦아듦' 두 숫자이고, 시간은 초(s)나 밀리초(ms)입니다.
  // 역할: Follow(끄는 동안 손에 붙음) · Settle(놓이거나 열릴 때 자리를 잡음) · Pop(눌렸다 튀어 오름) · Move(값의 자리로 감, 넘치지 않음) ·
  // Fade(투명도와 색) · Press(눌리는 순간) · Lead(시작 전의 뜸) · Seq(차례의 전체 시간) · Rest(되풀이 전의 쉼). 읽지 못한 값은 아래 기본값을 씁니다.
  const MOTION = { follow: '1000 63', settle: '340 26', pop: '520 21', move: '200 28.3', fade: '0.14s', press: '0.08s', lead: '400ms', seq: '1200ms', rest: '2600ms' };
  // 스프링이 0에서 1로 갈 때의 자리(t는 초). 잦아듦이 2√세기에 닿으면 넘치지 않고 곧장 갑니다.
  function springAt(k, c) {
    const w = Math.sqrt(k), z = c / (2 * w);
    if (z < 0.999) { const wd = w * Math.sqrt(1 - z * z); return (t) => 1 - Math.exp(-z * w * t) * (Math.cos(wd * t) + ((z * w) / wd) * Math.sin(wd * t)); }
    return (t) => 1 - (1 + w * t) * Math.exp(-w * t);
  }
  // 스프링을 CSS 전환으로 옮긴 [시간, 곡선]: 목표의 0.5% 안에 든 때까지를 시간으로 잡고, 그 사이를 33점으로 잇습니다.
  function springCss(k, c) {
    const f = springAt(k, c), pts = [];
    let d = 0;
    for (let t = 0; t <= 3; t += 0.005) if (Math.abs(f(t) - 1) > 0.005) d = t;
    d = Math.round((d + 0.01) * 100) / 100;
    for (let i = 0; i <= 32; i++) pts.push(i === 0 ? 0 : i === 32 ? 1 : +f((d * i) / 32).toFixed(3));
    return [`${d}s`, `linear(${pts.join(', ')})`];
  }
  // motion(card): 그 카드에 적용되는 움직임 값. pair(이름)은 스프링의 [세기, 잦아듦], ms(이름)은 밀리초,
  // curve(이름)은 { at(지난 밀리초) → 0에서 1까지의 자리, dur: 다 가는 데 걸리는 밀리초 }, css(이름)은 CSS 전환의 [시간, 곡선]입니다.
  function motion(card) {
    const cs = getComputedStyle(card), raw = (n) => cs.getPropertyValue(`--vzb-${n}`).trim().replace(/\s+/g, ' ');
    const two = (v) => { const p = v.split(' ').map(Number); return p.length === 2 && p[0] > 0 && p[1] > 0 ? p : null; };
    const pair = (n) => two(raw(n)) || two(MOTION[n]);
    const time = (v) => { const x = parseFloat(v); return isFinite(x) && x >= 0 && /s$/.test(v) ? (/ms$/.test(v) ? x : x * 1000) : null; };
    const ms = (n) => { const v = time(raw(n)); return v == null ? time(MOTION[n]) : v; };
    const css = (n) => springCss(...pair(n));
    const curve = (n) => { const f = springAt(...pair(n)); return { at: (t) => (t <= 0 ? 0 : f(t / 1000)), dur: parseFloat(css(n)[0]) * 1000 }; };
    return { pair, ms, css, curve };
  }
  // 스프링 값을 :root나 카드에서 바꿔 적었으면, CSS 전환이 쓰는 시간과 곡선(--vzb-…-t · -e)을 그 값으로 다시 계산해 카드에 적습니다.
  function tune(card) {
    const m = motion(card);
    for (const n of ['settle', 'pop', 'move']) {
      if (m.pair(n).join(' ') === MOTION[n]) continue;
      const [t, e] = m.css(n);
      card.style.setProperty(`--vzb-${n}-t`, t);
      card.style.setProperty(`--vzb-${n}-e`, e);
    }
  }
  // 스프링으로 목표에 다가가는 값: s = { x: 지금 자리, v: 속도 }를 dt(밀리초)만큼 옮깁니다. 가던 중에 목표가 바뀌어도 속도를 이어 갑니다.
  function springTo(s, to, k, c, dt) {
    let n = Math.max(1, Math.ceil(dt / 4));
    const h = dt / 1000 / n;
    while (n--) { s.v += (-k * (s.x - to) - c * s.v) * h; s.x += s.v * h; }
  }
  // 장면 잇기: 데이터가 바뀌어 카드를 갈아 끼울 때, 앞 카드가 화면에 그리고 있던 장면(값과 자리)을 새 카드에 넘깁니다.
  // 카드는 keep(card, 장면을 돌려주는 함수)로 맡기고, 새 카드는 before(card)로 받아 그 장면에서 새 값으로 옮겨 갑니다.
  const SCENE = new WeakMap(), FROM = new WeakMap();
  const keep = (card, fn) => SCENE.set(card, fn);
  const before = (card) => { const s = FROM.get(card); FROM.delete(card); return s || null; };
  // 같은 종류의 카드를 모두 찾아 그립니다. 코드를 두 번 붙여 넣어도 카드는 한 번만 그립니다.
  // 데이터가 바뀌면 다시 그립니다: 페이지의 스크립트가 카드 안 JSON(<script type="application/json">)의 글자를 바꾸면 그 카드만
  // 새 요소로 갈아 끼워 새로 그립니다(필터 · 기간 컨트롤과 잇는 길). 갈아 끼운 카드에는 data-vzb-live가 붙어, 움직이는 카드는
  // 처음부터 다시 재생하지 않습니다. 앞 카드가 장면을 맡겨 두었으면(keep) 그 장면에서 새 값으로 옮겨 가고, 아니면 마지막 장면을 바로 그립니다.
  // 예전 요소가 페이지에서 빠지므로 그 카드의 움직임도 함께 끝납니다.
  // 대비 최소선: 페이지가 바꾼 색으로 글자가 바탕과 4.5:1, 포커스 링이 바탕과 3:1에 못 미치면 콘솔에 한 줄로 알립니다.
  // 그림은 페이지 값 그대로 둡니다(브랜드 색을 카드가 고치지 않습니다). 색은 1×1 캔버스에 칠해 읽으므로 color-mix · oklch도 읽힙니다.
  const CHECKS = [['ink', 'card', 4.5], ['ink-2', 'card', 4.5], ['muted', 'card', 4.5], ['good-ink', 'card', 4.5], ['bad-ink', 'card', 4.5],
    ['alert-ink', 'card', 4.5], ['ink-2', 'track', 4.5], ['card', 'ink', 4.5], ['tip-ink', 'tip', 4.5], ['focus', 'card', 3]];
  // 같은 알림은 페이지에서 한 번만: 카드마다 core.js가 따로 들어 있으므로, 공유 기호(Symbol.for)로 document에 한 벌만 둡니다.
  const WARNED = Symbol.for('vizney-blocks.contrast');
  function checkContrast(card) {
    let ctx;
    try { ctx = document.createElement('canvas').getContext('2d', { willReadFrequently: true }); } catch (e) { return; }
    if (!ctx) return;
    const probe = el('i');
    probe.style.cssText = 'position:absolute;visibility:hidden';
    card.append(probe);
    const read = (name) => {
      probe.style.color = name === 'focus' ? 'var(--vzb-focus, var(--vzb-primary))' : `var(--vzb-${name})`;
      return getComputedStyle(probe).color;
    };
    const lum = (bg, fg) => {
      ctx.clearRect(0, 0, 1, 1);
      ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, 1, 1);
      ctx.fillStyle = bg; ctx.fillRect(0, 0, 1, 1);
      if (fg) { ctx.fillStyle = fg; ctx.fillRect(0, 0, 1, 1); }
      const [r, g, b] = ctx.getImageData(0, 0, 1, 1).data;
      const c = (v) => { v /= 255; return v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4; };
      return 0.2126 * c(r) + 0.7152 * c(g) + 0.0722 * c(b);
    };
    const low = [];
    CHECKS.forEach(([fg, bg, need]) => {
      const B = read(bg), a = lum(B), b = lum(B, read(fg));
      const ratio = (Math.max(a, b) + 0.05) / (Math.min(a, b) + 0.05);
      if (ratio < need - 0.005) low.push(`${fg === 'focus' ? '--vzb-focus' : `--vzb-${fg}`} on --vzb-${bg} ${ratio.toFixed(2)}:1 (needs ${need}:1)`);
    });
    probe.remove();
    const msg = low.join('; ');
    const warned = document[WARNED] || (document[WARNED] = new Set());
    if (!low.length || warned.has(msg)) return;
    warned.add(msg);
    console.warn(`Vizney Blocks: low contrast with the page's --vzb-* colors: ${msg}`);
  }
  function mount(type, draw) {
    const watch = (card, src) => {
      if (!src || !window.MutationObserver) return;
      const mo = new MutationObserver(() => {
        mo.disconnect();
        if (!card.isConnected) return;
        const next = card.cloneNode(false);
        next.removeAttribute('data-vzb-ready');
        next.setAttribute('data-vzb-live', '');
        next.append(src);
        const scene = SCENE.get(card);
        if (scene) { try { FROM.set(next, scene()); } catch (e) { /* 장면을 못 넘기면 마지막 장면을 바로 그립니다 */ } }
        card.replaceWith(next);
        run();
      });
      mo.observe(src, { childList: true, characterData: true, subtree: true });
    };
    const run = () => {
      document.querySelectorAll(`.vzb-card[data-vzb="${type}"]:not([data-vzb-ready])`).forEach((card) => {
        card.setAttribute('data-vzb-ready', '');
        const src = card.querySelector('script[type="application/json"]');
        const raw = src ? src.textContent : '';
        watch(card, src);
        let d;
        try { d = JSON.parse(raw); } catch (e) {
          card.append(el('p', 'vzb-error', tx(/"lang"\s*:\s*"en"/.test(raw) ? { lang: 'en' } : null, '카드 데이터(JSON)를 읽지 못했습니다: %s', e.message)));
          return;
        }
        card.setAttribute('role', 'group');
        if (d.title) card.setAttribute('aria-label', d.title);
        // 어두운 바탕: JSON에 "theme": "dark"를 적으면 Vizney 검정 바탕에 노랑을 첫째 색으로 그립니다(tokens.css의 .vzb-dark).
        card.classList.toggle('vzb-dark', d.theme === 'dark');
        tune(card);
        checkContrast(card);
        try { draw(card, d, format(d)); } catch (e) {
          card.append(el('p', 'vzb-error', tx(d, '카드를 그리지 못했습니다: %s', e.message)));
        }
      });
    };
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
    else run();
  }
  // 서식 길이: 페이지가 :root에 적은 --vzb-<name>(0 · 4px · 0.5rem 등)을 px 숫자로 읽습니다. 비워 두었으면(initial) fallback을 돌려줍니다.
  // 스크립트가 그리는 SVG(막대 끝, 말풍선, 칸)가 CSS와 같은 모서리 변수를 읽게 하는 길입니다. 길이는 카드에 잠깐 넣은 요소의 폭으로 재서, rem · calc도 읽힙니다.
  function len(card, name, fallback) {
    if (!getComputedStyle(card).getPropertyValue(`--vzb-${name}`).trim()) return fallback;
    const probe = el('i');
    probe.style.cssText = `position:absolute;visibility:hidden;width:var(--vzb-${name})`;
    card.append(probe);
    const v = parseFloat(getComputedStyle(probe).width);
    probe.remove();
    return isFinite(v) && v >= 0 ? v : fallback;
  }
  // 크기가 바뀔 때마다 그래픽을 다시 그립니다. 차트는 카드에 남은 높이를 채웁니다.
  function onSize(node, fn) {
    let lw = 0, lh = 0;
    new ResizeObserver((entries) => {
      const w = Math.floor(entries[0].contentRect.width), h = Math.floor(entries[0].contentRect.height);
      if (w > 0 && h > 0 && (w !== lw || h !== lh)) { lw = w; lh = h; fn(w, h); }
    }).observe(node);
  }

  /* 굴러가는 숫자(컨트롤 공통): 숫자 한 자리마다 0–9를 세로로 쌓은 칸을 두고 새 값의 자리로 굴립니다(구독 수가 오르는 단추처럼).
     roll(box, text): box 안의 글자를 text로 바꿉니다. 자리 수와 쉼표 자리가 그대로면 칸이 제자리에서 구르고, 바뀌면 칸을 새로 만들어
     0에서 굴러 올라옵니다. 움직임은 roll.css의 transition이 그리고, box는 화면 낭독기에 읽히지 않으므로 값은 부르는 쪽이 따로 알립니다. */
  function roll(box, text) {
    const chars = [...String(text)], shape = chars.map((c) => (/\d/.test(c) ? '0' : c)).join('');
    box.setAttribute('aria-hidden', 'true');
    if (box.dataset.shape !== shape) {
      box.dataset.shape = shape;
      box.replaceChildren(...chars.map((c) => {
        if (!/\d/.test(c)) return el('span', 'vzb-rl-ch', c);
        const col = el('span', 'vzb-rl-col'), strip = el('span', 'vzb-rl-strip');
        for (let k = 0; k < 10; k++) strip.append(el('span', null, String(k)));
        col.append(strip);
        return col;
      }));
      box.getBoundingClientRect();   // 새 칸은 0에 선 채로 그려진 뒤 굴러갑니다
    }
    [...box.children].forEach((node, i) => { if (node.classList.contains('vzb-rl-col')) node.firstChild.style.transform = `translateY(${-10 * +chars[i]}%)`; });
  }

  // Download Button: 화면에 있는 차트 카드(target)의 데이터를 내려받는 단추입니다. 카드 머리 없이 단추만 그립니다. 처음에는 내려받기 아이콘만 있고, 마우스를 올리거나
  // 키보드로 들어오거나 아이콘을 누르면 단추가 옆으로 늘어나며 받을 형식(formats: csv · json · png, 비우면 셋 다)이 하나씩 나옵니다.
  // 늘어난 단추는 옆 내용 위로 겹쳐 떠서 자리를 밀어내지 않고, 오른쪽에 펼칠 자리가 없으면 아이콘은 그대로 둔 채 왼쪽으로 펼칩니다(align: end면 왼쪽부터).
  // csv · json은 target 카드의 지금 JSON(컨트롤로 거른 뒤)에서 표를 만들어 받습니다: items(객체 배열), 없으면 labels와 values(한도
  // limit · 목표 target이 있으면 열로 더함), labels와 series, 그 밖에는 JSON에서 처음 나오는 객체 배열입니다. csv 앞에는 엑셀용 BOM을 붙입니다.
  // png는 target 카드를 보이는 그대로 그림으로 받습니다(글꼴은 기기에 설치된 것으로 그려질 수 있습니다).
  // 받기 전에 카드에서 vzb-download 이벤트(bubbles, cancelable, detail: { name, target, format, filename, rows })를 보냅니다. 페이지가
  // preventDefault()를 부르면 카드는 파일을 만들지 않습니다(서버에서 원본을 받는 화면 등).
  // 움직임(download-button.css): 형식은 늘어나는 단추 안에서 차례로 나오고, 받으면 단추가 접히며 화살표가 받침으로 떨어져 체크로 바뀌고
  // 작은 점들이 터집니다. 옆에는 받은 형식과 행 수가 굴러 올라왔다가 1.8초 뒤 사라집니다. 받을 데이터가 없으면 단추가 좌우로 떱니다.
  // 열린 상태에서 아이콘을 다시 누르면 첫 형식으로 받습니다. Esc는 닫고, 좌우 화살표는 형식 사이를 옮깁니다.
  // 카드를 감싼 요소에 data-vzb-loop가 있으면(갤러리) 파일은 만들지 않고 고르는 모습만 되풀이합니다.
  mount('download-button', (card, d, f) => {
    const FORMATS = ['csv', 'json', 'png'];
    const formats = (Array.isArray(d.formats) ? d.formats.map((x) => String(x).toLowerCase()).filter((x) => FORMATS.includes(x)) : FORMATS);
    const list = formats.length ? formats : FORMATS;
    const base = String(d.filename || 'data').replace(/\.(csv|json|png)$/i, '');
    const label = d.label || tx(d, '내려받기');
    const foot = el('div', 'vzb-dl'), row = el('div', 'vzb-dl-row'), menu = el('div', 'vzb-dl-menu'), say = el('p', 'vzb-sr');
    menu.setAttribute('role', 'group');
    menu.setAttribute('aria-label', label);
    menu.setAttribute('data-vzb-input', '');
    say.setAttribute('aria-live', 'polite');
    const btn = el('button', 'vzb-dl-btn');
    btn.type = 'button';
    btn.setAttribute('aria-label', label);
    btn.setAttribute('aria-expanded', 'false');
    // 아이콘: 화살표 · 체크 · 받침과, 받았을 때 터지는 점 여덟 개(데이터 색 넷을 두 번씩)
    const ic = el('span', 'vzb-dl-ic'), burst = el('span', 'vzb-dl-burst');
    for (let i = 0; i < 8; i++) {
      const dot = el('i', `vzb-dl-c${(i % 4) + 1}`);
      dot.style.setProperty('--vzb-a', `${i * 45 + 12}deg`);
      dot.style.setProperty('--vzb-r', `${i % 2 ? 26 : 20}px`);
      burst.append(dot);
    }
    const svg = svgEl('svg', { class: 'vzb-dl-svg', viewBox: '0 0 20 20', 'aria-hidden': 'true' }), arrow = svgEl('g', { class: 'vzb-dl-arrow' });
    arrow.append(svgEl('path', { d: 'M10 3.2v8.6' }), svgEl('path', { d: 'M6.6 8.6l3.4 3.4 3.4-3.4' }));
    svg.append(arrow, svgEl('path', { class: 'vzb-dl-check', d: 'M6.4 9.6l2.5 2.5 4.8-5.2' }), svgEl('path', { class: 'vzb-dl-tray', d: 'M3.6 13.4v1.6a1.6 1.6 0 0 0 1.6 1.6h9.6a1.6 1.6 0 0 0 1.6-1.6v-1.6' }));
    ic.append(burst, svg);
    btn.append(ic);
    const opts = el('span', 'vzb-dl-opts');
    const items = list.map((fmt, i) => {
      const o = el('button', 'vzb-dl-opt', fmt.toUpperCase());
      o.type = 'button';
      o.tabIndex = -1;
      o.dataset.fmt = fmt;
      o.style.setProperty('--vzb-i', i);
      o.setAttribute('aria-label', tx(d, '%s로 받기', fmt.toUpperCase()));
      opts.append(o);
      return o;
    });
    menu.append(btn, opts);
    const meta = el('span', 'vzb-dl-meta'), got = el('span', 'vzb-dl-got'), what = el('span', 'vzb-dl-what'), count = el('span', 'vzb-rl'), unit = el('span', 'vzb-dl-unit');
    got.append(what, count, unit);
    got.setAttribute('aria-hidden', 'true');
    meta.append(got);
    const anchor = el('div', 'vzb-dl-anchor');   // 아이콘 자리(44px). 늘어난 단추는 이 자리에서 옆 내용 위로 겹쳐 뜹니다
    anchor.append(menu);
    row.append(anchor, meta);
    const alignEnd = d.align === 'end';   // 화면 오른쪽 위처럼 오른쪽 끝에 둘 때: 아이콘을 오른쪽 끝에 붙이고 왼쪽으로 펼칩니다
    foot.classList.toggle('is-end', alignEnd);
    foot.append(row, say);
    card.append(foot);

    // 열고 닫기: 마우스를 올리면 열고, 내리면 잠시 뒤 닫습니다. 키보드로 아이콘에 들어오거나 아이콘을 눌러도 엽니다.
    let open = false, shut = 0, since = 0, hush = false;   // hush: Esc로 닫고 아이콘에 포커스를 돌려줄 때는 다시 열지 않습니다
    // 펼칠 방향: 아이콘 오른쪽(왼쪽 맞춤이면)이나 왼쪽(align end면)으로 화면 안에 다 들어가면 그쪽, 아니면 반대쪽입니다. 아이콘은 그 자리에 있습니다.
    function side() {
      const r = btn.getBoundingClientRect(), vw = document.documentElement.clientWidth || window.innerWidth, need = 44 + opts.scrollWidth;
      const right = vw - 8 - r.left, left = r.right - 8;
      return alignEnd ? left >= need || right < need : right < need && left > right;
    }
    function setOpen(v) {
      clearTimeout(shut);
      if (open === v) return;
      if (v) foot.classList.toggle('is-left', side());
      open = v;
      since = performance.now();
      foot.classList.toggle('is-open', v);
      btn.setAttribute('aria-expanded', String(v));
      items.forEach((o) => { o.tabIndex = v ? 0 : -1; });
    }
    menu.addEventListener('pointerenter', (e) => { if (e.pointerType === 'mouse') { idle = performance.now() + 8000; setOpen(true); } });
    menu.addEventListener('pointerleave', (e) => { if (e.pointerType === 'mouse' && !menu.contains(document.activeElement)) shut = setTimeout(() => setOpen(false), 280); });
    btn.addEventListener('focus', () => { if (hush) hush = false; else if (btn.matches(':focus-visible')) setOpen(true); });
    menu.addEventListener('focusout', (e) => { if (!menu.contains(e.relatedTarget)) shut = setTimeout(() => setOpen(false), 120); });
    menu.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') { setOpen(false); if (document.activeElement !== btn) { hush = true; btn.focus(); } return; }
      if (e.key !== 'ArrowRight' && e.key !== 'ArrowLeft') return;
      const all = [btn, ...items], i = all.indexOf(document.activeElement);
      if (i < 0) return;
      e.preventDefault();
      setOpen(true);
      all[Math.max(0, Math.min(all.length - 1, i + (e.key === 'ArrowRight' ? 1 : -1)))].focus();
    });
    document.addEventListener('pointerdown', function away(e) {
      if (!card.isConnected) { document.removeEventListener('pointerdown', away, true); return; }
      if (open && !menu.contains(e.target)) setOpen(false);
    }, true);

    // 받을 표: target 카드의 지금 JSON에서 만듭니다.
    const target = () => (d.target ? document.querySelector(`.vzb-card[data-vzb="${String(d.target).replace(/["\\]/g, '')}"]`) : null);
    const isRows = (a) => Array.isArray(a) && a.length > 0 && a.every((r) => r && typeof r === 'object' && !Array.isArray(r));
    function rowsOf(t) {
      const src = t && t.querySelector('script[type="application/json"]');
      if (!src) return null;
      let data;
      try { data = JSON.parse(src.textContent); } catch (e) { return null; }
      if (isRows(data.items)) return data.items;
      if (Array.isArray(data.labels) && data.labels.length && Array.isArray(data.values)) {
        return data.labels.map((l, i) => {
          const r = { label: l, value: data.values[i] };
          ['limit', 'target'].forEach((k) => { if (fin(data[k])) r[k] = +data[k]; });
          return r;
        });
      }
      if (Array.isArray(data.labels) && data.labels.length && isRows(data.series)) {
        return data.labels.map((l, i) => Object.assign({ label: l }, ...data.series.map((s) => ({ [s.name]: Array.isArray(s.values) ? s.values[i] : null }))));
      }
      return Object.values(data).find(isRows) || null;
    }
    // CSV: 행마다 숫자 · 글자 필드를 처음 나온 순서대로 열로 둡니다. 쉼표 · 따옴표 · 줄바꿈이 든 칸은 따옴표로 감쌉니다.
    function csv(rows) {
      const cols = [];
      rows.forEach((r) => Object.keys(r).forEach((k) => { if (!cols.includes(k) && (r[k] == null || typeof r[k] !== 'object')) cols.push(k); }));
      const cell = (v) => { const s = v == null ? '' : String(v); return /[",\r\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s; };
      return `\ufeff${[cols.map(cell).join(','), ...rows.map((r) => cols.map((k) => cell(r[k])).join(','))].join('\r\n')}\r\n`;
    }
    // PNG: 카드를 보이는 그대로 복제해 계산된 스타일을 글자로 옮기고, SVG 그림 안에 넣어 캔버스에 그립니다(외부 라이브러리 없이).
    function png(t) {
      const r = t.getBoundingClientRect(), w = Math.ceil(r.width), h = Math.ceil(r.height), clone = t.cloneNode(true);
      const from = [t, ...t.querySelectorAll('*')], to = [clone, ...clone.querySelectorAll('*')];
      from.forEach((node, i) => {
        const out = to[i];
        if (!out || !out.setAttribute) return;
        if (node.tagName === 'SCRIPT' || node.closest('button')) { out.setAttribute('style', 'visibility:hidden'); return; }   // 단추(다시 보기 등)는 자리만 남기고 그리지 않습니다
        const cs = getComputedStyle(node);
        let css = '';
        for (let k = 0; k < cs.length; k++) css += `${cs[k]}:${cs.getPropertyValue(cs[k])};`;
        out.setAttribute('style', css);
      });
      clone.querySelectorAll('script').forEach((s) => s.remove());
      clone.style.margin = '0';
      const xml = new XMLSerializer().serializeToString(clone);
      const src = `data:image/svg+xml;charset=utf-8,${encodeURIComponent(`<svg xmlns="http://www.w3.org/2000/svg" width="${w}" height="${h}"><foreignObject width="100%" height="100%">${xml}</foreignObject></svg>`)}`;
      return new Promise((done, fail) => {
        const img = new Image();
        img.onload = () => {
          try {
            const k = 2, c = document.createElement('canvas');
            c.width = w * k;
            c.height = h * k;
            const g = c.getContext('2d');
            g.scale(k, k);
            g.drawImage(img, 0, 0);
            c.toBlob((b) => (b ? done({ blob: b, w: w * k, h: h * k }) : fail(new Error('png'))), 'image/png');
          } catch (e) { fail(e); }
        };
        img.onerror = fail;
        img.src = src;
      });
    }
    function save(blob, name) {
      const url = URL.createObjectURL(blob), a = el('a');
      a.href = url;
      a.download = name;
      a.hidden = true;
      document.body.append(a);
      a.click();
      a.remove();
      setTimeout(() => URL.revokeObjectURL(url), 2000);
    }
    // 상태: '' 처음, 'done' 받음, 'none' 받을 데이터 없음. 같은 상태를 다시 줘도 움직임이 처음부터 다시 돌도록 클래스를 지웠다 붙입니다.
    let back = 0;
    function mark(state, fmt, n, px) {
      clearTimeout(back);
      foot.classList.remove('is-done', 'is-none');
      if (!state) return;
      void foot.offsetWidth;
      foot.classList.add(`is-${state}`);
      if (state === 'done') {
        what.textContent = `${fmt.toUpperCase()} · `;
        if (px) { roll(count, '0'); requestAnimationFrame(() => roll(count, String(px[0]))); unit.textContent = `×${px[1]}px`; }
        else { unit.textContent = (d.lang === 'en' ? ' ' : '') + tx(d, n === 1 ? '행|1' : '행'); roll(count, '0'); requestAnimationFrame(() => roll(count, num(n, 0))); }
      } else {
        what.textContent = tx(d, '데이터 없음');
        count.replaceChildren();
        unit.textContent = '';
      }
      back = setTimeout(() => { if (card.isConnected) foot.classList.remove('is-done', 'is-none'); }, state === 'done' ? 1800 : 1400);
    }
    async function run(fmt, demo) {
      setOpen(false);
      const t = target(), rows = rowsOf(t);
      if (demo) { mark('done', fmt, rows ? rows.length : 13, fmt === 'png' && t ? [Math.ceil(t.offsetWidth * 2), Math.ceil(t.offsetHeight * 2)] : null); return; }
      if (!t || (fmt !== 'png' && !rows)) { mark('none'); say.textContent = tx(d, '받을 데이터가 없습니다.'); return; }
      const name = `${base}.${fmt}`;
      const ev = new CustomEvent('vzb-download', { bubbles: true, cancelable: true, detail: { name: d.name || '', target: d.target || '', format: fmt, filename: name, rows: rows || [] } });
      const go = card.dispatchEvent(ev);
      if (fmt === 'png') {
        try {
          const p = await png(t);
          if (go) save(p.blob, name);
          mark('done', fmt, 0, [p.w, p.h]);
          say.textContent = tx(d, '%s로 받았습니다.', name);
        } catch (e) { mark('none'); say.textContent = tx(d, '받을 데이터가 없습니다.'); }
        return;
      }
      if (go) save(new Blob([fmt === 'csv' ? csv(rows) : JSON.stringify(rows, null, 2)], { type: fmt === 'csv' ? 'text/csv;charset=utf-8' : 'application/json' }), name);
      mark('done', fmt, rows.length);
      say.textContent = rows.length === 1 ? tx(d, '%s로 1행을 받았습니다.', name) : tx(d, '%1로 %2행을 받았습니다.', name, rows.length);
    }
    let idle = 0;
    btn.addEventListener('click', () => {
      idle = performance.now() + 8000;
      if (!open) setOpen(true);
      else if (performance.now() - since > 250) run(list[0]);   // 열린 상태에서 아이콘을 다시 누르면 첫 형식으로 받습니다
    });
    items.forEach((o) => o.addEventListener('click', () => { idle = performance.now() + 8000; run(o.dataset.fmt); }));

    // 갤러리(data-vzb-loop): 화면에 보이는 동안, 손을 대기 전과 손을 뗀 지 8초 뒤부터 열어서 형식을 고르는 모습을 되풀이합니다(파일은 만들지 않습니다).
    if (card.closest('[data-vzb-loop]')) {
      let seen = true, k = 0;
      if ('IntersectionObserver' in window) new IntersectionObserver((es) => { seen = es[es.length - 1].isIntersecting; }).observe(card);
      const timer = setInterval(() => {
        if (!card.isConnected) { clearInterval(timer); return; }
        if (!seen || performance.now() < idle || open) return;
        const o = items[k++ % items.length];
        setOpen(true);
        setTimeout(() => { if (performance.now() >= idle) o.classList.add('is-pick'); }, 700);
        setTimeout(() => { o.classList.remove('is-pick'); if (performance.now() >= idle) run(o.dataset.fmt, true); }, 1150);
      }, 4200);
    }
  });
})();
