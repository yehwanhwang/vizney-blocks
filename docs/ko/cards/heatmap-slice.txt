  function krw(v, d) {

    if (a >= 1e12) return [num(v / 1e12, k(1)), '조'];

      case 'usd': return F({ parts: (v) => usd(v, d.decimals), rel: true, usd: true });
      case 'pct': return F({ parts: (v) => [num(v, dec(v, 1)), '%'], rel: false, dp: d.decimals });
      case 'days': return F({ parts: (v) => [num(v, dec(v, 1)), tx(root || d, '일')], rel: true });
      default: return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) ? 0 : 1)), d.unit || ''], rel: true });
    }

  // 자리를 늘려(최대 세 자리) 1.12%에서 1.16%로 오른 것을 0.0%p가 아니라 0.03%p로 적습니다.

    const a = Math.abs(diff);

  function delta(f, value, base, better) {
    const diff = value - base;
    const dir = Math.abs(diff) < 1e-9 ? 0 : Math.sign(diff);
    const want = better === 'down' ? -1 : better === 'neutral' ? 0 : 1;
    const tone = dir === 0 || want === 0 ? 'flat' : dir === want ? 'good' : 'bad';
    let text;
    if (!f.rel) text = tx(f, '%s%p', pp(f, diff));
    else if (base) text = `${num(Math.abs(diff / base) * 100, 1)}%`;
    else text = show(f, Math.abs(diff));
    return { dir, tone, text };
  }

  // 움직임 값(tokens.css의 --vzb-follow … --vzb-rest). 스프링은 '세기 잦아듦' 두 숫자이고, 시간은 초(s)나 밀리초(ms)입니다.
  // 역할: Follow(끄는 동안 손에 붙음) · Settle(놓이거나 열릴 때 자리를 잡음) · Pop(눌렸다 튀어 오름) · Move(값의 자리로 감, 넘치지 않음) ·
  // Fade(투명도와 색) · Press(눌리는 순간) · Lead(시작 전의 뜸) · Seq(차례의 전체 시간) · Rest(되풀이 전의 쉼). 읽지 못한 값은 아래 기본값을 씁니다.
  const MOTION = { follow: '1000 63', settle: '340 26', pop: '520 21', move: '200 28.3', fade: '0.14s', press: '0.08s', lead: '400ms', seq: '1200ms', rest: '2600ms' };
  // 스프링이 0에서 1로 갈 때의 자리(t는 초). 잦아듦이 2√세기에 닿으면 넘치지 않고 곧장 갑니다.
  function springAt(k, c) {
    const w = Math.sqrt(k), z = c / (2 * w);
    if (z < 0.999) { const wd = w * Math.sqrt(1 - z * z); return (t) => 1 - Math.exp(-z * w * t) * (Math.cos(wd * t) + ((z * w) / wd) * Math.sin(wd * t)); }
    return (t) => 1 - (1 + w * t) * Math.exp(-w * t);
  }
  // 스프링을 CSS 전환으로 옮긴 [시간, 곡선]: 목표의 0.5% 안에 든 때까지를 시간으로 잡고, 그 사이를 33점으로 잇습니다.
  function springCss(k, c) {
    const f = springAt(k, c), pts = [];
    let d = 0;
    for (let t = 0; t <= 3; t += 0.005) if (Math.abs(f(t) - 1) > 0.005) d = t;
    d = Math.round((d + 0.01) * 100) / 100;
    for (let i = 0; i <= 32; i++) pts.push(i === 0 ? 0 : i === 32 ? 1 : +f((d * i) / 32).toFixed(3));
    return [`${d}s`, `linear(${pts.join(', ')})`];
  }
  // motion(card): 그 카드에 적용되는 움직임 값. pair(이름)은 스프링의 [세기, 잦아듦], ms(이름)은 밀리초,
  // curve(이름)은 { at(지난 밀리초) → 0에서 1까지의 자리, dur: 다 가는 데 걸리는 밀리초 }, css(이름)은 CSS 전환의 [시간, 곡선]입니다.
  function motion(card) {
    const cs = getComputedStyle(card), raw = (n) => cs.getPropertyValue(`--vzb-${n}`).trim().replace(/\s+/g, ' ');
    const two = (v) => { const p = v.split(' ').map(Number); return p.length === 2 && p[0] > 0 && p[1] > 0 ? p : null; };
    const pair = (n) => two(raw(n)) || two(MOTION[n]);
    const time = (v) => { const x = parseFloat(v); return isFinite(x) && x >= 0 && /s$/.test(v) ? (/ms$/.test(v) ? x : x * 1000) : null; };
    const ms = (n) => { const v = time(raw(n)); return v == null ? time(MOTION[n]) : v; };
    const css = (n) => springCss(...pair(n));
    const curve = (n) => { const f = springAt(...pair(n)); return { at: (t) => (t <= 0 ? 0 : f(t / 1000)), dur: parseFloat(css(n)[0]) * 1000 }; };
    return { pair, ms, css, curve };
  }

  function tune(card) {
    const m = motion(card);
    for (const n of ['settle', 'pop', 'move']) {
      if (m.pair(n).join(' ') === MOTION[n]) continue;
      const [t, e] = m.css(n);
      card.style.setProperty(`--vzb-${n}-t`, t);
      card.style.setProperty(`--vzb-${n}-e`, e);
    }
  }
  // 스프링으로 목표에 다가가는 값: s = { x: 지금 자리, v: 속도 }를 dt(밀리초)만큼 옮깁니다. 가던 중에 목표가 바뀌어도 속도를 이어 갑니다.
  function springTo(s, to, k, c, dt) {
    let n = Math.max(1, Math.ceil(dt / 4));
    const h = dt / 1000 / n;
    while (n--) { s.v += (-k * (s.x - to) - c * s.v) * h; s.x += s.v * h; }
  }

  const CHECKS = [['ink', 'card', 4.5], ['ink-2', 'card', 4.5], ['muted', 'card', 4.5], ['good-ink', 'card', 4.5], ['bad-ink', 'card', 4.5],
    ['alert-ink', 'card', 4.5], ['ink-2', 'track', 4.5], ['card', 'ink', 4.5], ['tip-ink', 'tip', 4.5], ['focus', 'card', 3]];
  // 같은 알림은 페이지에서 한 번만: 카드마다 core.js가 따로 들어 있으므로, 공유 기호(Symbol.for)로 document에 한 벌만 둡니다.
  const WARNED = Symbol.for('vizney-blocks.contrast');
  function checkContrast(card) {
    let ctx;
    try { ctx = document.createElement('canvas').getContext('2d', { willReadFrequently: true }); } catch (e) { return; }
    if (!ctx) return;
    const probe = el('i');
    probe.style.cssText = 'position:absolute;visibility:hidden';
    card.append(probe);
    const read = (name) => {
      probe.style.color = name === 'focus' ? 'var(--vzb-focus, var(--vzb-primary))' : `var(--vzb-${name})`;
      return getComputedStyle(probe).color;
    };
    const lum = (bg, fg) => {
      ctx.clearRect(0, 0, 1, 1);
      ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, 1, 1);
      ctx.fillStyle = bg; ctx.fillRect(0, 0, 1, 1);
      if (fg) { ctx.fillStyle = fg; ctx.fillRect(0, 0, 1, 1); }
      const [r, g, b] = ctx.getImageData(0, 0, 1, 1).data;
      const c = (v) => { v /= 255; return v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4; };
      return 0.2126 * c(r) + 0.7152 * c(g) + 0.0722 * c(b);
    };
    const low = [];
    CHECKS.forEach(([fg, bg, need]) => {
      const B = read(bg), a = lum(B), b = lum(B, read(fg));
      const ratio = (Math.max(a, b) + 0.05) / (Math.min(a, b) + 0.05);
      if (ratio < need - 0.005) low.push(`${fg === 'focus' ? '--vzb-focus' : `--vzb-${fg}`} on --vzb-${bg} ${ratio.toFixed(2)}:1 (needs ${need}:1)`);
    });
    probe.remove();
    const msg = low.join('; ');
    const warned = document[WARNED] || (document[WARNED] = new Set());
    if (!low.length || warned.has(msg)) return;
    warned.add(msg);
    console.warn(`Vizney Blocks: low contrast with the page's --vzb-* colors: ${msg}`);
  }
  function mount(type, draw) {
    const watch = (card, src) => {
      if (!src || !window.MutationObserver) return;
      const mo = new MutationObserver(() => {
        mo.disconnect();
        if (!card.isConnected) return;
        const next = card.cloneNode(false);
        next.removeAttribute('data-vzb-ready');
        next.setAttribute('data-vzb-live', '');
        next.append(src);
        const scene = SCENE.get(card);
        if (scene) { try { FROM.set(next, scene()); } catch (e) { /* 장면을 못 넘기면 마지막 장면을 바로 그립니다 */ } }
        card.replaceWith(next);
        run();
      });
      mo.observe(src, { childList: true, characterData: true, subtree: true });
    };
    const run = () => {
      document.querySelectorAll(`.vzb-card[data-vzb="${type}"]:not([data-vzb-ready])`).forEach((card) => {
        card.setAttribute('data-vzb-ready', '');
        const src = card.querySelector('script[type="application/json"]');
        const raw = src ? src.textContent : '';
        watch(card, src);
        let d;
        try { d = JSON.parse(raw); } catch (e) {
          card.append(el('p', 'vzb-error', tx(/"lang"\s*:\s*"en"/.test(raw) ? { lang: 'en' } : null, '카드 데이터(JSON)를 읽지 못했습니다: %s', e.message)));
          return;
        }
        card.setAttribute('role', 'group');
        if (d.title) card.setAttribute('aria-label', d.title);
        // 어두운 바탕: JSON에 "theme": "dark"를 적으면 Vizney 검정 바탕에 노랑을 첫째 색으로 그립니다(tokens.css의 .vzb-dark).
        card.classList.toggle('vzb-dark', d.theme === 'dark');
        tune(card);
        checkContrast(card);
        try { draw(card, d, format(d)); } catch (e) {
          card.append(el('p', 'vzb-error', tx(d, '카드를 그리지 못했습니다: %s', e.message)));
        }
      });
    };
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
    else run();
  }
  // 서식 길이: 페이지가 :root에 적은 --vzb-<name>(0 · 4px · 0.5rem 등)을 px 숫자로 읽습니다. 비워 두었으면(initial) fallback을 돌려줍니다.
  // 스크립트가 그리는 SVG(막대 끝, 말풍선, 칸)가 CSS와 같은 모서리 변수를 읽게 하는 길입니다. 길이는 카드에 잠깐 넣은 요소의 폭으로 재서, rem · calc도 읽힙니다.
  function len(card, name, fallback) {
    if (!getComputedStyle(card).getPropertyValue(`--vzb-${name}`).trim()) return fallback;
    const probe = el('i');
    probe.style.cssText = `position:absolute;visibility:hidden;width:var(--vzb-${name})`;
    card.append(probe);
    const v = parseFloat(getComputedStyle(probe).width);
    probe.remove();
    return isFinite(v) && v >= 0 ? v : fallback;
  }
  // 크기가 바뀔 때마다 그래픽을 다시 그립니다. 차트는 카드에 남은 높이를 채웁니다.
  function onSize(node, fn) {
    let lw = 0, lh = 0;
    new ResizeObserver((entries) => {
      const w = Math.floor(entries[0].contentRect.width), h = Math.floor(entries[0].contentRect.height);
      if (w > 0 && h > 0 && (w !== lw || h !== lh)) { lw = w; lh = h; fn(w, h); }
    }).observe(node);
  }

  function showTip(x, y, label, value, side) {
    if (!tip) {
      tip = el('div', 'vzb-tip');
      tip.setAttribute('aria-hidden', 'true');
      document.body.append(tip);
    }
    tip.classList.toggle('vzb-dark', tipDark);
    tip.replaceChildren(el('span', 'vzb-tip-label', label), typeof value === 'string' ? el('span', 'vzb-tip-value', value) : value);
    tip.classList.add('is-on');
    const w = tip.offsetWidth, h = tip.offsetHeight, W = window.innerWidth, H = window.innerHeight;
    if (side) {
      // side가 'left'이면 짚은 자리의 왼쪽을 먼저, 그 밖에는 오른쪽을 먼저 씁니다. 자리가 모자라면 반대쪽으로 넘기고, 높이는 짚은 자리에 가운데를 맞춥니다.
      // 짚은 표시의 위(머리의 대표 수치나 레이블)를 덮지 않아야 하는 차트가 씁니다.
      const right = x + 14 + w <= W - 8, left = x - w - 14 >= 8;
      tip.style.left = `${(side === 'left' ? left || !right : !right) ? Math.max(8, x - w - 14) : x + 14}px`;
      tip.style.top = `${Math.min(Math.max(8, y - h / 2), H - h - 8)}px`;
      return;
    }
    tip.style.left = `${Math.min(Math.max(8, x - w / 2), W - w - 8)}px`;
    tip.style.top = `${y - h - 12 < 8 ? y + 16 : y - h - 12}px`;
  }
  function hideTip() { if (tip) tip.classList.remove('is-on'); }
  // 툴팁 내용: { value, tone, note } 또는 { rows: [{ tone, value, name }], note }, { value, flow: [{ value, name }], note }. 값이 먼저, 이름은 뒤에 옅게 적습니다.
  // 증감은 차트와 같은 뜻의 색으로 적습니다: tone이 'good'이면 초록, 'bad'면 빨강. note의 줄도 { text, tone }으로 색을 줄 수 있습니다.
  function tipBody(spec, label) {
    const body = el('span', 'vzb-tip-body');
    let text = label;
    // value는 짚은 값(big이면 18px로 크게), rows는 그 값을 이루는 항목들(견본 · 값 · 이름)입니다. 둘을 함께 주면 값 아래에 항목을 적습니다.
    // 견본은 tone(색 이름)으로 칠하고, box면 네모로 그립니다. flow는 앞에서 뒤로 넘어가는 값들을 화살표(→)로 한 줄에 잇습니다(예: 앞 단계 → 뒤 단계).
    if (spec.value != null) {
      body.append(el('span', `vzb-tip-value${spec.big ? ' is-big' : ''}${spec.tone ? ` is-${spec.tone}` : ''}`, spec.value));
      text += ` ${spec.value}`;
    }
    if (spec.rows) {
      const grid = el('span', 'vzb-tip-rows');
      spec.rows.forEach((r) => {
        const row = el('span', 'vzb-tip-row'), key = el('i', `vzb-tip-key${r.tone ? ` vzb-${r.tone}` : ''}${r.box ? ' is-box' : ''}`);
        row.append(key, el('b', null, r.value), el('span', null, r.name));
        grid.append(row);
        text += `, ${r.name} ${r.value}`;
      });
      body.append(grid);
    }
    if (spec.flow) {
      const line = el('span', 'vzb-tip-flow');
      spec.flow.forEach((r, i) => {
        if (i) line.append(el('span', 'vzb-tip-arrow', '→'));
        const item = el('span', 'vzb-tip-flow-item');
        item.append(el('b', null, r.value), el('span', null, r.name));
        line.append(item);
      });
      body.append(line);
      text += `, ${spec.flow.map((r) => `${r.name} ${r.value}`).join(' → ')}`;
    }
    // note는 한 줄 글 또는 줄마다 나눈 배열입니다.
    (Array.isArray(spec.note) ? spec.note : [spec.note]).filter(Boolean).forEach((line) => {
      const o = typeof line === 'object' ? line : { text: line };
      body.append(el('span', `vzb-tip-note${o.tone ? ` is-${o.tone}` : ''}`, o.text));
      text += `, ${o.text}`;
    });
    return { node: body, text };
  }

  // 차이를 부호와 함께 적습니다. 비율 지표는 %p입니다.
  function signed(f, diff) {
    const s = diff > 0 ? '+' : diff < 0 ? '-' : '';   // 줄어든 값은 하이픈(-)으로 적습니다. 복사해 표 계산기에 붙여도 음수로 읽힙니다
    return f.rel ? `${s}${show(f, Math.abs(diff))}` : `${s}${pp(f, diff)}%p`;
  }
  // pick(clientX, clientY) → 순번, point(순번) → { x, y, label, value, text, side }(화면 좌표), clear() → 강조 해제
  // n은 개수나 개수를 돌려주는 함수입니다. vertical이면 위아래 화살표로 옮깁니다. 돌려주는 { show(순번), at() }로 키를 더 붙입니다.
  // 아무것도 고르지 않은 채 화살표를 누르면 기간 차트는 마지막(최근)부터, vertical이나 first면 첫 순번부터 짚습니다.
  function interact(plot, n, pick, point, clear, vertical, first) {
    let idx = -1, touched = false;
    const count = () => (typeof n === 'function' ? n() : n);
    const live = el('span', 'vzb-sr');
    live.setAttribute('aria-live', 'polite');
    plot.after(live);
    const go = (i) => {
      if (i < 0 || i >= count()) return;
      idx = i;
      const p = point(i);
      tipDark = !!plot.closest('.vzb-dark');
      showTip(p.x, p.y, p.label, p.value, p.side);
      live.textContent = p.text || `${p.label} ${p.value}`;
    };
    const end = () => {
      if (idx < 0) return;
      idx = -1;
      clear();
      hideTip();
    };
    const [back, next] = vertical ? ['ArrowUp', 'ArrowDown'] : ['ArrowLeft', 'ArrowRight'];
    // pick이 -1이면(조각 밖) 강조를 풉니다.
    const at = (e) => { const i = pick(e.clientX, e.clientY); if (i < 0) end(); else go(i); };
    plot.addEventListener('pointermove', (e) => { if (e.pointerType !== 'touch' || touched) at(e); });
    plot.addEventListener('pointerdown', (e) => { if (e.pointerType === 'touch') { touched = true; at(e); } });
    plot.addEventListener('pointerleave', (e) => { if (e.pointerType !== 'touch') end(); });
    // 문서와 창에 건 듣기는 그래프가 페이지에서 빠지면(카드를 다시 그리면) 스스로 풉니다.
    const away = (e) => { if (!plot.isConnected) document.removeEventListener('pointerdown', away, true); else if (touched && !plot.contains(e.target)) { touched = false; end(); } };
    document.addEventListener('pointerdown', away, true);
    plot.addEventListener('keydown', (e) => {
      if (e.key !== back && e.key !== next) return;
      e.preventDefault();
      const last = count() - 1;
      go(idx < 0 ? (vertical || first ? 0 : last) : Math.max(0, Math.min(last, idx + (e.key === next ? 1 : -1))));
    });
    plot.addEventListener('blur', end);
    const scrolled = () => { if (!plot.isConnected) window.removeEventListener('scroll', scrolled); else end(); };
    window.addEventListener('scroll', scrolled, { passive: true });
    return { show: go, at: () => idx, end };
  }

  // Heatmap Slice: 두 갈래(요일 × 시간대 같은 rows × cols)로 나눈 값을 칸의 진하기로 그리고, 짚은 칸의 행과 열을 막대로 세워 보여 주는 히트맵입니다.
  // 칸은 첫째 데이터 색의 다섯 단계로 칠합니다(0에서 가장 큰 값까지를 같은 폭으로 끊습니다). 짚은 칸의 행과 열은 단계 그대로, 나머지 칸은 같은 색을 옅게 깔아 짚은 줄이 또렷합니다.
  // 색으로는 어디에 몰리는지를 읽고, 크기는 막대로 견줍니다.
  // 짚은 칸의 행은 그리드 위에 열마다 세로 막대로, 짚은 칸의 열은 그리드 오른쪽에 행마다 가로 막대로 섭니다. 두 막대는 같은 눈금(0에서 가장 큰 값)을 써서 서로 견줄 수 있습니다.
  // 막대는 그 막대가 나타내는 칸과 같은 색이고, 짚은 칸과 그 칸의 막대에는 글자색 테두리를 두릅니다.
  // 처음에는 아무 칸도 짚지 않은 전체 장면입니다: 모든 칸을 단계 그대로 칠하고, 위 막대는 열마다의 평균, 오른쪽 막대는 행마다의 평균, 머리의 큰 숫자는 전체 평균입니다.
  // focus(["행 이름", "열 이름"])를 적으면 그 칸을 짚은 채로 시작합니다. 칸을 짚으면 머리의 큰 숫자는 그 칸의 값이 되고, 뒤에 그 값이 전체 평균의 몇 배인지와 어느 칸인지(행 · 열 이름)를 적습니다(카드가 계산합니다).
  // 움직임: 칸에 마우스를 올리면 그 칸을 잠시 짚어 보이고, 누르거나(터치 포함) 포커스한 뒤 화살표를 누르면 그 칸에 짚은 자리가 머뭅니다. 마우스가 그래프를 벗어나면 머물던 칸으로(없으면 전체로) 돌아갑니다. 머물던 칸을 다시 누르거나 Esc를 누르면 전체로 돌아갑니다. 색이 그 행과 열로 옮겨 가고 테두리가 그 칸으로 미끄러져 가며(Settle),
  // 막대는 새 행 · 열의 값으로 옮겨 갑니다(Move: 넘치지 않음). 행을 옮기면 위 막대가, 열을 옮기면 오른쪽 막대가 바뀌어 줄마다 모양이 어떻게 다른지 보입니다.
  // 화면에 보이면 막대가 바닥에서 한 번 자랍니다. 데이터가 바뀌어 다시 그릴 때(data-vzb-live)는 칸의 진하기와 막대가 화면에 있던 값에서 새 값으로 옮겨 가고, 짚은 칸은 이어 갑니다.
  // rows · cols나 카드 크기가 달라지면 옮겨 가지 않고 바로 그립니다. 카드를 감싼 요소에 data-vzb-loop가 있으면(갤러리) 손을 대기 전까지 전체에서 시작해 행마다 가장 높은 칸을 차례로 짚어 보입니다.
  mount('heatmap-slice', (card, d, f) => {
    // 걸러서 남은 것이 없을 때(rows · cols · values가 빈 배열)는 오류가 아니라 빈 화면입니다.
    const empty = (a) => Array.isArray(a) && !a.length;
    if (empty(d.rows) || empty(d.cols) || empty(d.values)) {
      head(card, d, f, null);
      card.append(el('p', 'vzb-hl-none', tx(d, '표시할 항목이 없습니다.')));
      return;
    }
    const rows = Array.isArray(d.rows) ? d.rows.map(String) : [], cols = Array.isArray(d.cols) ? d.cols.map(String) : [], R = rows.length, C = cols.length, N = R * C;
    if (!N || !Array.isArray(d.values) || d.values.length !== R || d.values.some((r) => !Array.isArray(r) || r.length !== C || r.some((v) => v != null && (!fin(v) || v < 0)))) throw new Error(tx(d, 'rows · cols와 values를 확인하세요.'));
    const vals = d.values.flatMap((r) => r.map((v) => (v == null ? null : +v))), nums = vals.filter((v) => v != null);
    if (!nums.length) throw new Error(tx(d, 'rows · cols와 values를 확인하세요.'));
    const vmax = Math.max(...nums), avg = nums.reduce((a, b) => a + b, 0) / nums.length;
    const frac = (v) => (v == null || vmax <= 0 ? 0 : v / vmax);
    // 칸의 단계: 0에서 가장 큰 값까지를 같은 폭의 다섯 단계로 끊습니다. 단계마다 첫째 데이터 색을 섞는 몫(STEPS)이 칸의 진하기입니다.
    const STEPS = [0.09, 0.29, 0.51, 0.75, 1], stepOf = (v) => Math.min(4, Math.floor(frac(v) * 5)), tone = (v) => (v == null ? 0 : STEPS[stepOf(v)]);
    const shade = (node, a) => node.style.setProperty('--vzb-p', `${(a * 100).toFixed(1)}%`);   // 색으로 바꾸는 식은 heatmap-slice.css에 있습니다
    const toneTo = vals.map(tone), key = `${rows.join('\n')}\t${cols.join('\n')}`;

    const mean = (a) => { const n = a.filter((v) => v != null); return n.length ? n.reduce((x, y) => x + y, 0) / n.length : null; };
    const rowAvg = rows.map((_, r) => mean(vals.slice(r * C, r * C + C))), colAvg = cols.map((_, c) => mean(rows.map((_, r) => vals[r * C + c])));
    const peak = vals.reduce((b, v, i) => (v != null && (b < 0 || v > vals[b]) ? i : b), -1), where = (i) => `${rows[Math.floor(i / C)]} · ${cols[i % C]}`;

    const find = (p) => { const r = Array.isArray(p) ? rows.indexOf(String(p[0])) : -1, c = Array.isArray(p) ? cols.indexOf(String(p[1])) : -1; return r < 0 || c < 0 ? -1 : r * C + c; };
    const savedAttr = card.getAttribute('data-vzb-focus');

    const rowOf = () => (at < 0 ? -1 : Math.floor(at / C)), colOf = () => (at < 0 ? -1 : at % C);
    const valAt = () => (at < 0 ? avg : vals[at] || 0);   // 머리의 숫자가 가는 값

    // 머리: 제목 · 부제, 그 아래 짚은 칸의 값 · 평균 대비 · 어느 칸인지(전체 장면에서는 전체 평균)
    head(card, d, f, null);
    const fig = el('div', 'vzb-fig'), numEl = el('div', 'vzb-num vzb-hl-num'), numT = document.createTextNode(''), unitEl = el('span', 'vzb-unit');
    const dl = el('span', 'vzb-delta'), bd = el('span', 'vzb-badge vzb-flat'), vs = el('span', 'vzb-vs');
    numEl.append(numT, unitEl);
    dl.append(bd, vs);
    fig.append(numEl, dl);
    card.append(fig);
    const whole = nums.every(Number.isInteger);
    let figKey = '';
    function setFig(v) {
      const none = at >= 0 && vals[at] == null, [a, u] = none ? [tx(d, '데이터 없음'), ''] : f.parts(whole ? Math.round(v) : v), times = at < 0 || none || avg <= 0 ? '' : tx(d, '평균의 %s배', num(v / avg, 1));
      const k = `${a}|${u}|${times}|${at}`;
      if (k === figKey) return;
      figKey = k;
      numT.nodeValue = a;
      unitEl.textContent = u;
      unitEl.style.display = u ? '' : 'none';
      bd.textContent = times;
      bd.style.display = times ? '' : 'none';
      vs.textContent = at < 0 ? tx(d, '전체 평균') : where(at);
    }

    const foot = el('div', 'vzb-foot vzb-fill vzb-hl');
    const plot = el('div', 'vzb-plot vzb-hl-plot');
    plot.tabIndex = 0;
    plot.setAttribute('role', 'img');
    const svg = svgEl('svg', { class: 'vzb-hl-svg', 'aria-hidden': 'true' });
    plot.append(svg);
    foot.append(plot);
    card.append(foot);
    const say = el('span', 'vzb-sr');
    say.setAttribute('aria-live', 'polite');
    card.append(say);
    const told = (i) => (i < 0 ? `${tx(d, '전체 평균')} ${show(f, whole ? Math.round(avg) : avg)}` : vals[i] == null ? `${where(i)} ${tx(d, '데이터 없음')}` : `${where(i)} ${show(f, vals[i])}${avg > 0 ? `, ${tx(d, '평균의 %s배', num(vals[i] / avg, 1))}` : ''}`);
    const describe = () => plot.setAttribute('aria-label', tx(d, '%1. %2. 화살표로 칸을 옮기면 그 행과 열의 값이 막대로 바뀌고, Esc를 누르면 전체로 돌아갑니다. 모든 값은 이어지는 표에 있습니다.', d.title || '', told(at)));
    // 화면 낭독기용 표
    const sr = el('div', 'vzb-sr'), table = el('table'), hr = el('tr');
    hr.append(el('th', null, ''), ...cols.map((c) => el('th', null, c)));
    table.append(el('caption', null, d.title || ''), hr);
    rows.forEach((name, r) => {
      const tr = el('tr');
      tr.append(el('th', null, name), ...cols.map((_, c) => { const v = vals[r * C + c]; return el('td', null, v == null ? tx(d, '데이터 없음') : show(f, v)); }));
      table.append(tr);
    });
    sr.append(table);
    card.append(sr);

    // 그림: 위 막대(짚은 행) · 칸 · 오른쪽 막대(짚은 열) · 짚은 행과 열의 테두리 · 행 이름(왼쪽) · 열 이름(아래) · 범례(오른쪽 아래)
    const baseT = svgEl('line', { class: 'vzb-hl-base' }), baseR = svgEl('line', { class: 'vzb-hl-base' }), cellsG = svgEl('g', {});
    const cells = vals.map((v) => { const c = svgEl('rect', { class: `vzb-hl-cell${v == null ? ' is-null' : ''}` }); cellsG.append(c); return c; });
    const topB = cols.map(() => svgEl('path', { class: 'vzb-hl-bar' })), sideB = rows.map(() => svgEl('path', { class: 'vzb-hl-bar' }));
    const box = svgEl('rect', { class: 'vzb-hl-box' });
    const rowT = rows.map((name) => svgText(0, 0, name, 'vzb-hl-name', 'end')), colT = cols.map((name) => svgText(0, 0, name, 'vzb-hl-name', 'middle'));
    const topCap = svgText(0, 0, '', 'vzb-hl-name is-on', 'end'), sideCap = svgText(0, 0, '', 'vzb-hl-name is-on', 'middle');
    const legG = svgEl('g', {}), legLo = svgText(0, 0, tx(d, '적음'), 'vzb-hl-name', 'end'), legHi = svgText(0, 0, tx(d, '많음'), 'vzb-hl-name', 'start');
    const legB = STEPS.map((a) => { const b = svgEl('rect', { class: 'vzb-hl-cell is-on', width: 10, height: 10 }); shade(b, a); return b; });
    legG.append(legLo, ...legB, legHi);
    svg.append(baseT, baseR, cellsG, ...topB, ...sideB, box, ...rowT, ...colT, topCap, sideCap, legG);

    const fx = (v) => v.toFixed(1);
    foot.style.minHeight = `${R * 14 + 96}px`;   // 행 이름(12px)이 겹치지 않게 한 행은 14px 밑으로 내려가지 않습니다. 자리가 모자라면 카드가 그만큼 길어집니다
    // 칸의 진하기와 막대의 길이는 값이라 Move로 옮기고, 테두리의 자리는 값이 아니라 Settle로 미끄러집니다.
    const M = motion(card), [MK, MC] = M.pair('move'), [SK, SC] = M.pair('settle'), NAMEH = 20, LEGH = 20, GAP = 2;
    let W = 0, H = 0, GL = 0, cw = 1, ch = 1, RC = 2, RB = 3, every = 1, TOPH = 40, SIDEW = 60, gy = 0, sx = 0, wait = 0, idle = 0;
    // 스프링: 위 막대(열마다) · 오른쪽 막대(행마다)의 길이(0~1), 테두리의 자리(px), 머리의 숫자(값). 다시 그릴 때는 칸의 진하기(sa)도 옮깁니다.
    const tb = cols.map(() => ({ x: 0, v: 0 })), sb = rows.map(() => ({ x: 0, v: 0 })), bx = { x: 0, v: 0 }, by = { x: 0, v: 0 }, kv = { x: valAt(), v: 0 };
    let from = before(card), sa = null, grown = false, placed = false;
    const tbTo = (c) => (grown ? frac(at < 0 ? colAvg[c] : vals[rowOf() * C + c]) : 0), sbTo = (r) => (grown ? frac(at < 0 ? rowAvg[r] : vals[r * C + colOf()]) : 0);
    // 열 이름: 몇 칸마다 하나씩 적되, 짚은 열의 이름은 늘 굵게 적고 그와 겹칠 이웃 이름은 뺍니다.
    function names() {
      const all = at < 0, pr = rowOf(), pc = colOf();
      rowT.forEach((tn, r) => tn.classList.toggle('is-on', r === pr));
      cells.forEach((c, i) => c.classList.toggle('is-on', all || Math.floor(i / C) === pr || i % C === pc));   // 짚은 행과 열의 칸만 단계 그대로 칠합니다. 전체 장면에서는 모든 칸이 단계 그대로입니다
      colT.forEach((tn, c) => {
        tn.classList.toggle('is-on', c === pc);
        tn.setAttribute('opacity', c === pc || (c % every === 0 && (all || Math.abs(c - pc) >= every)) ? 1 : 0);
      });
      // 막대는 그 막대가 나타내는 값(짚은 줄의 칸, 전체 장면에서는 평균)과 같은 단계의 색입니다(가장 낮은 단계는 바탕에 묻히므로 둘째 단계로 칠합니다).
      const barTone = (v) => STEPS[Math.max(1, v == null ? 1 : stepOf(v))];
      topB.forEach((b, c) => { b.classList.toggle('is-on', c === pc); shade(b, barTone(all ? colAvg[c] : vals[pr * C + c])); });
      sideB.forEach((b, r) => { b.classList.toggle('is-on', r === pr); shade(b, barTone(all ? rowAvg[r] : vals[r * C + pc])); });
      topCap.textContent = all ? tx(d, '전체') : rows[pr];
      sideCap.textContent = all ? tx(d, '전체') : cols[pc];
      box.classList.toggle('is-off', all);   // 전체 장면에서는 테두리를 흐려 지웁니다
    }
    function layout(w, h) {
      W = w; H = h;
      svg.setAttribute('width', w);
      svg.setAttribute('height', h);
      svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
      const fam = getComputedStyle(card).fontFamily;
      GL = Math.ceil(Math.max(...rows.map((x) => textW(x, 700, fam, 12)))) + 8;
      const colW = Math.ceil(Math.max(...cols.map((x) => textW(x, 700, fam, 12))));
      SIDEW = Math.max(colW + 4, Math.min(96, Math.round(w * 0.14)));
      TOPH = Math.max(26, Math.min(64, Math.round(h * 0.2)));
      gy = TOPH + 8;
      cw = Math.max(4, (w - GL - 12 - SIDEW) / C);
      ch = Math.max(6, (h - gy - NAMEH - LEGH) / R);
      sx = GL + C * cw + 12;
      RC = Math.min(len(card, 'radius-mark', 3), Math.min(cw, ch) / 4);   // 칸의 모서리는 짧은 변의 1/4을 넘기지 않습니다(네모가 점으로 읽히지 않게)
      RB = len(card, 'radius-mark', 3);
      const g = Math.min(GAP, cw / 4, ch / 4);
      cells.forEach((c, i) => {
        c.setAttribute('x', fx(GL + (i % C) * cw + g / 2)); c.setAttribute('y', fx(gy + Math.floor(i / C) * ch + g / 2));
        c.setAttribute('width', fx(cw - g)); c.setAttribute('height', fx(ch - g));
        c.setAttribute('rx', fx(RC));
      });
      baseT.setAttribute('x1', GL); baseT.setAttribute('x2', fx(GL + C * cw)); baseT.setAttribute('y1', TOPH + 0.5); baseT.setAttribute('y2', TOPH + 0.5);
      baseR.setAttribute('x1', fx(sx - 0.5)); baseR.setAttribute('x2', fx(sx - 0.5)); baseR.setAttribute('y1', gy); baseR.setAttribute('y2', fx(gy + R * ch));
      box.setAttribute('width', fx(cw)); box.setAttribute('height', fx(ch)); box.setAttribute('rx', fx(RC + 1));
      rowT.forEach((tn, r) => { tn.setAttribute('x', GL - 8); tn.setAttribute('y', fx(gy + r * ch + ch / 2 + 4)); });
      every = Math.max(1, Math.ceil((colW + 10) / cw));
      colT.forEach((tn, c) => { tn.setAttribute('x', fx(GL + c * cw + cw / 2)); tn.setAttribute('y', fx(gy + R * ch + 15)); });
      topCap.setAttribute('x', GL - 8); topCap.setAttribute('y', TOPH - 2);
      sideCap.setAttribute('x', fx(sx + SIDEW / 2)); sideCap.setAttribute('y', fx(gy + R * ch + 15));

      const hiW = Math.ceil(textW(legHi.textContent, 400, fam, 12)), y = h - 11;

      legLo.setAttribute('x', x - 6); legLo.setAttribute('y', y + 9);

      legHi.setAttribute('x', x + 4); legHi.setAttribute('y', y + 9);
      if (from) { begin(from); from = null; }

      if (!placed || !busy()) { if (at >= 0) { bx.x = GL + colOf() * cw; by.x = gy + rowOf() * ch; } bx.v = by.v = 0; placed = true; }
      names();
      render();
      kick();
    }
    // 앞 장면에서 시작합니다. rows · cols나 카드 크기가 달라졌으면 옮겨 가지 않고 마지막 장면을 바로 그립니다.
    function begin(prev) {
      if (!prev || !prev.inView || prev.W !== W || prev.H !== H || prev.key !== key || !Array.isArray(prev.a)) return;
      sa = toneTo.map((_, i) => ({ x: prev.a[i] || 0, v: (prev.av && prev.av[i]) || 0 }));
      tb.forEach((sp, c) => { sp.x = prev.tb[c] || 0; sp.v = (prev.tbv && prev.tbv[c]) || 0; });
      sb.forEach((sp, r) => { sp.x = prev.sb[r] || 0; sp.v = (prev.sbv && prev.sbv[r]) || 0; });
      bx.x = prev.bx; by.x = prev.by; bx.v = prev.bxv || 0; by.v = prev.byv || 0;
      if (prev.at === at && prev.kv != null) { kv.x = prev.kv; kv.v = prev.kvv || 0; }
      placed = true;
    }
    const near = (sp, to, eps, veps) => Math.abs(sp.x - to) <= eps && Math.abs(sp.v) <= veps;
    function busy() {

      return wait > 0 || !!sa || (at >= 0 && (!near(bx, GL + pc * cw, 0.2, 4) || !near(by, gy + pr * ch, 0.2, 4))) || !near(kv, valAt(), (vmax || 1) * 0.0006, (vmax || 1) * 0.01)
        || tb.some((sp, c) => !near(sp, tbTo(c), 0.002, 0.02)) || sb.some((sp, r) => !near(sp, sbTo(r), 0.002, 0.02));
    }
    function step(dt) {
      const pr = rowOf(), pc = colOf();
      if (wait > 0) { wait -= dt; if (wait <= 0) grown = true; }
      tb.forEach((sp, c) => { springTo(sp, tbTo(c), MK, MC, dt); if (near(sp, tbTo(c), 0.002, 0.02)) { sp.x = tbTo(c); sp.v = 0; } });

      const land = (sp, to, k, c, eps, veps) => { springTo(sp, to, k, c, dt); if (near(sp, to, eps, veps)) { sp.x = to; sp.v = 0; } };
      if (at >= 0) { land(bx, GL + pc * cw, SK, SC, 0.2, 4); land(by, gy + pr * ch, SK, SC, 0.2, 4); } else { bx.v = by.v = 0; }   // 전체 장면에서는 테두리가 있던 자리에서 흐려집니다
      land(kv, valAt(), MK, MC, (vmax || 1) * 0.0006, (vmax || 1) * 0.01);
      if (sa) {
        let moving = false;
        sa.forEach((sp, i) => { springTo(sp, toneTo[i], MK, MC, dt); if (near(sp, toneTo[i], 0.002, 0.02)) { sp.x = toneTo[i]; sp.v = 0; } else moving = true; });
        if (!moving) sa = null;
      }
    }
    function render() {
      if (!W) return;
      cells.forEach((c, i) => { const a = sa ? sa[i].x : toneTo[i]; if (c.vzbA !== a) { c.vzbA = a; shade(c, a); } });
      // 위 막대: 열 폭에 맞춰(최대 28px) 바닥선에서 위로. 오른쪽 막대: 행 높이에 맞춰(최대 18px) 왼쪽 선에서 오른쪽으로. 끝만 둥글립니다.
      const bw = Math.min(28, cw - Math.min(GAP, cw / 4)), bh = Math.min(18, ch - Math.min(GAP, ch / 4));
      topB.forEach((b, c) => {
        const hgt = Math.max(0, tb[c].x) * TOPH, x = GL + c * cw + (cw - bw) / 2, r = Math.min(RB, bw / 2, hgt);
        b.setAttribute('d', hgt < 0.3 ? '' : `M${fx(x)},${TOPH}V${fx(TOPH - hgt + r)}a${fx(r)},${fx(r)} 0 0 1 ${fx(r)},${fx(-r)}H${fx(x + bw - r)}a${fx(r)},${fx(r)} 0 0 1 ${fx(r)},${fx(r)}V${TOPH}Z`);
      });
      sideB.forEach((b, k) => {
        const wid = Math.max(0, sb[k].x) * SIDEW, y = gy + k * ch + (ch - bh) / 2, r = Math.min(RB, bh / 2, wid);
        b.setAttribute('d', wid < 0.3 ? '' : `M${fx(sx)},${fx(y)}H${fx(sx + wid - r)}a${fx(r)},${fx(r)} 0 0 1 ${fx(r)},${fx(r)}V${fx(y + bh - r)}a${fx(r)},${fx(r)} 0 0 1 ${fx(-r)},${fx(r)}H${fx(sx)}Z`);
      });
      box.setAttribute('x', fx(bx.x)); box.setAttribute('y', fx(by.x));
      setFig(kv.x);
    }
    // 지금 화면의 장면: 데이터가 바뀌어 카드를 갈아 끼울 때 새 카드에 넘깁니다.
    keep(card, () => {
      if (!W) return from;
      return {
        inView, W, H, key, at, a: sa ? sa.map((sp) => sp.x) : toneTo.slice(), av: sa ? sa.map((sp) => sp.v) : null,
        tb: tb.map((sp) => sp.x), tbv: tb.map((sp) => sp.v), sb: sb.map((sp) => sp.x), sbv: sb.map((sp) => sp.v),
        bx: bx.x, by: by.x, bxv: bx.v, byv: by.v, kv: kv.x, kvv: kv.v,
      };
    });

    const live = card.hasAttribute('data-vzb-live');   // 데이터가 바뀌어 다시 그린 카드: 막대가 바닥에서 다시 자라지 않습니다
    let inView = false, seen = false, stamp0 = 0, raf = 0;
    function tick(stamp) {
      raf = 0;
      if (!card.isConnected || !inView) { stamp0 = 0; return; }
      const dt = stamp0 ? Math.min(100, stamp - stamp0) : 16;
      stamp0 = stamp;
      step(dt);
      render();
      if (busy()) raf = requestAnimationFrame(tick); else stamp0 = 0;
    }
    function kick() { if (!raf && W && inView && busy()) raf = requestAnimationFrame(tick); }
    // 칸 짚기: 테두리와 막대가 그 칸의 행과 열로 옮겨 갑니다. 짚은 칸은 카드 요소에 적어 두어, 데이터가 바뀌어 다시 그려도 이어 갑니다.
    // i가 -1이면 아무 칸도 짚지 않은 전체 장면으로 돌아갑니다.
    function pick(i, quiet, stay) {
      if (i < -1 || i >= N) return;
      if (stay && i !== pin) { pin = i; card.setAttribute('data-vzb-focus', i < 0 ? 'null' : JSON.stringify([rows[Math.floor(i / C)], cols[i % C]])); }
      if (i === at) return;
      const was = at;
      at = i;
      if (was < 0 && W) { bx.x = GL + colOf() * cw; by.x = gy + rowOf() * ch; bx.v = by.v = 0; }   // 전체에서 처음 짚을 때는 테두리가 미끄러져 오지 않고 그 칸에서 나타납니다
      kv.x = valAt(); kv.v = 0;   // 다른 칸의 값이므로 세지 않고 바로 적습니다
      names();
      describe();
      if (!quiet) say.textContent = told(at);
      render();
      kick();
    }
    const cellAt = (e) => {
      const r = plot.getBoundingClientRect(), c = Math.floor((e.clientX - r.left - GL) / cw), w = Math.floor((e.clientY - r.top - gy) / ch);
      return c < 0 || c >= C || w < 0 || w >= R ? -1 : w * C + c;
    };
    const touch = () => { idle = performance.now() + 6000; };
    // 칸 위에서만 누를 수 있다는 것을 포인터로 알립니다(칸 사이 틈도 칸으로 봅니다).
    plot.addEventListener('pointermove', (e) => { const i = cellAt(e); plot.classList.toggle('is-over', i >= 0); if (e.pointerType !== 'touch' && i >= 0) { touch(); pick(i, true); } });
    // 누른 칸에 머뭅니다. 머물던 칸을 다시 누르면 풀려 전체로 돌아갑니다(마우스는 그 칸에 올라가 있으므로 그래프를 벗어날 때 돌아갑니다).
    plot.addEventListener('pointerdown', (e) => {
      const i = cellAt(e);
      if (i < 0) return;
      touch();
      if (i !== pin) { pick(i, true, true); return; }
      pin = -1;
      card.setAttribute('data-vzb-focus', 'null');
      if (e.pointerType === 'touch') pick(-1, true);
    });
    plot.addEventListener('pointerleave', (e) => { plot.classList.remove('is-over'); if (e.pointerType !== 'touch') pick(pin, true); });
    plot.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') { if (at < 0 && pin < 0) return; e.preventDefault(); touch(); pick(-1, false, true); return; }
      const pr = rowOf(), pc = colOf();
      // 전체 장면에서 처음 누른 화살표는 가장 높은 칸을 짚습니다.
      const to = at < 0 ? { ArrowLeft: peak, ArrowRight: peak, ArrowUp: peak, ArrowDown: peak, Home: peak, End: peak }[e.key]
        : { ArrowLeft: pc > 0 ? at - 1 : -2, ArrowRight: pc < C - 1 ? at + 1 : -2, ArrowUp: pr > 0 ? at - C : -2, ArrowDown: pr < R - 1 ? at + C : -2, Home: pr * C, End: pr * C + C - 1 }[e.key];
      if (to === undefined) return;
      e.preventDefault();
      touch();
      pick(to, false, true);
    });
    describe();

    if (live || !('IntersectionObserver' in window)) grown = true;
    if (from) inView = !!from.inView;
    onSize(plot, layout);
    if (document.fonts && document.fonts.ready) document.fonts.ready.then(() => { if (W) layout(W, H); });
    // 화면에 처음 보이면 뜸(Lead) 뒤에 막대가 바닥에서 자랍니다. 화면 밖에 있는 동안은 멈춰 둡니다.
    if ('IntersectionObserver' in window) {
      const io = new IntersectionObserver((es) => {
        if (!card.isConnected) { io.disconnect(); return; }
        inView = es[es.length - 1].isIntersecting;
        if (inView && !seen) { seen = true; if (!grown) wait = M.ms('lead') || 1; }
        if (!inView && sa) { sa = null; render(); }   // 화면 밖에서는 옮겨 가기를 끝낸 장면으로 둡니다
        kick();
      }, { threshold: 0.3 });
      io.observe(card);
    } else { inView = true; }
    // 갤러리(data-vzb-loop): 화면에 보이는 동안, 손을 대기 전과 손을 뗀 지 6초 뒤부터 전체 장면에서 시작해 행마다 가장 높은 칸을 차례로 짚어 보이고 전체로 돌아옵니다.
    if (card.closest('[data-vzb-loop]')) {
      const tops = rows.map((_, r) => { let b = -1; for (let c = 0; c < C; c++) { const i = r * C + c; if (vals[i] != null && (b < 0 || vals[i] > vals[b])) b = i; } return b; }).filter((i) => i >= 0);
      const tour = [-1, ...tops];
      let k = Math.max(0, tour.indexOf(at));
      const timer = setInterval(() => {
        if (!card.isConnected) { clearInterval(timer); return; }
        if (!inView || !grown || performance.now() < idle || tour.length < 2) return;
        k = (k + 1) % tour.length;
        pin = tour[k];   // 갤러리의 둘러보기는 카드 요소에 적지 않습니다
        pick(tour[k], true);
      }, M.ms('rest') || 2600);
    }
  });
})();
</script>
