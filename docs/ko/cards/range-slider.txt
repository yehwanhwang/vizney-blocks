  // tx(d 또는 f, 원문, 값…): d.lang(format()이 f에 옮겨 둡니다)이 en이면 영어 문구를, 아니면 원문을 돌려줍니다.
  function tx(x, s, ...a) {
    let i = 0;
    const out = x && x.lang === 'en' && EN[s] !== undefined ? EN[s] : s.split('|')[0];
    return out.replace(/%(\d|s)/g, (m, k) => String(a[k === 's' ? i++ : k - 1]));
  }
  // 숫자 뒤의 단위: 영어 단위(days · orders)는 한 칸 띄우고, 기호와 한 글자 단위(% · K · 억)는 붙입니다.
  const withUnit = (n, u) => (/^[A-Za-z]{2}/.test(u) ? `${n} ${u}` : n + u);
  function el(tag, cls, text) {
    const n = document.createElement(tag);
    if (cls) n.className = cls;
    if (text != null) n.textContent = text;
    return n;
  }
  function svgEl(tag, attrs) {
    const n = document.createElementNS(SVGNS, tag);
    for (const k in attrs) n.setAttribute(k, attrs[k]);
    return n;

  const nfs = {};
  function num(v, d) {
    nfs[d] = nfs[d] || new Intl.NumberFormat('ko-KR', { minimumFractionDigits: d, maximumFractionDigits: d });
    return nfs[d].format(v);
  }
  // 원 금액: 조 · 억 · 만으로 줄여 적습니다. 10만 원보다 작으면 87,324원처럼 원 단위 그대로 적어 객단가 · CAC가 "9만"으로
  // 뭉개지지 않게 합니다. decimals를 주면 줄인 숫자의 소수 자릿수를 그 값으로 맞춥니다(1억 8,600만 원은 2면 1.86억).
  function krw(v, d) {
    const a = Math.abs(v), k = (x) => (d != null ? d : x);
    if (a >= 1e12) return [num(v / 1e12, k(1)), '조'];
    if (a >= 1e8) return [num(v / 1e8, k(a >= 1e10 ? 0 : 1)), '억'];
    if (a >= 1e5) return [num(v / 1e4, k(0)), '만'];
    return [num(v, 0), '원'];
  }
  // 달러 금액: 원래 값을 받아 K · M · B로 줄이고 소수 한 자리로 맞춥니다($745.6K). 접미사는 단위처럼 작게 씁니다.
  function usd(v, d) {
    const a = Math.abs(v), s = v < 0 ? '-' : '';
    const [div, u] = a >= 1e9 ? [1e9, 'B'] : a >= 1e6 ? [1e6, 'M'] : a >= 1e3 ? [1e3, 'K'] : [1, ''];
    return [`${s}$${num(a / div, d != null ? d : div > 1 ? 1 : 0)}`, u];
  }
  // format: eok(억 단위 값) · krw(원 단위 값) · usd(달러 값) · count(unit으로 단위) · pct(증감을 %p로) · days
  // 축이나 탭처럼 카드 안의 다른 객체로 만들 때는 두 번째 인자로 카드 데이터를 주어 lang을 잇습니다.
  function format(d, root) {
    const dec = (v, fallback) => (d.decimals != null ? d.decimals : fallback);
    const F = (o) => Object.assign(o, { lang: (root || d).lang });
    switch (d.format) {
      case 'eok': return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) || Math.abs(v) >= 100 ? 0 : 1)), '억'], rel: true });
      case 'krw': return F({ parts: (v) => krw(v, d.decimals), rel: true });
      case 'usd': return F({ parts: (v) => usd(v, d.decimals), rel: true, usd: true });
      case 'pct': return F({ parts: (v) => [num(v, dec(v, 1)), '%'], rel: false, dp: d.decimals });
      case 'days': return F({ parts: (v) => [num(v, dec(v, 1)), tx(root || d, '일')], rel: true });
      default: return F({ parts: (v) => [num(v, dec(v, Number.isInteger(v) ? 0 : 1)), d.unit || ''], rel: true });
    }
  }
  const show = (f, v) => withUnit(...f.parts(v));
  // 그릴 수 있는 숫자인지(빈 값 · 빈 글자 · NaN 제외)
  const fin = (v) => v != null && v !== '' && isFinite(v);
  // 비율의 차이(%p) 숫자: 소수 자릿수는 decimals(비우면 한 자리)를 따릅니다. 0이 아닌 차이가 0.0으로 보일 만큼 작으면
  // 자리를 늘려(최대 세 자리) 1.12%에서 1.16%로 오른 것을 0.0%p가 아니라 0.03%p로 적습니다.
  function pp(f, diff) {
    const a = Math.abs(diff);
    let k = f.dp != null ? f.dp : 1;
    while (k < 3 && a > 1e-9 && Math.round(a * 10 ** k) === 0) k++;
    return num(a, k);
  }
  // 증감: 방향은 화살표로, 좋고 나쁨(tone)은 better(up · down · neutral)로 정합니다. 비율은 %p로 적습니다.
  function delta(f, value, base, better) {
    const diff = value - base;
    const dir = Math.abs(diff) < 1e-9 ? 0 : Math.sign(diff);
    const want = better === 'down' ? -1 : better === 'neutral' ? 0 : 1;
    const tone = dir === 0 || want === 0 ? 'flat' : dir === want ? 'good' : 'bad';
    let text;
    if (!f.rel) text = tx(f, '%s%p', pp(f, diff));
    else if (base) text = `${num(Math.abs(diff / base) * 100, 1)}%`;
    else text = show(f, Math.abs(diff));
    return { dir, tone, text };
  }
  function arrow(dir) {
    const s = svgEl('svg', { class: 'vzb-arrow', viewBox: '0 0 12 12', 'aria-hidden': 'true' });
    s.append(svgEl('path', { d: dir > 0 ? 'M3.5 8.5l5-5M4.5 3.5h4v4' : 'M3.5 3.5l5 5M8.5 4.5v4h-4' }));
    return s;
  }
  // 대표 수치(단위는 작게)
  function figure(f, value) {

    n.append(text);
    if (unit) n.append(el('span', 'vzb-unit', unit));

  // 증감 배지: 화살표는 오르내림, 색은 좋고 나쁨(better)입니다.
  function badge(f, value, base, better) {

    const b = el('span', `vzb-badge vzb-${dl.tone}`);

  // 머리: 제목 · 기간 부제(subtitle) · 대표 수치와 바로 뒤의 증감 배지. value가 null이면 제목만 둡니다.
  // 배지는 숫자 바로 뒤에 붙어 있고, 카드가 좁아 한 줄에 다 들어가지 않으면 비교 문구만 숫자 아래 줄로 내려갑니다.
  // 내려갔는지는 카드의 vzb-wrapped로 알려, 높이가 1단위인 카드가 아래 여백으로 그 줄을 받을 수 있게 합니다.
  function head(card, d, f, value, cmp, cmpText) {

  // 움직임 값(tokens.css의 --vzb-follow … --vzb-rest). 스프링은 '세기 잦아듦' 두 숫자이고, 시간은 초(s)나 밀리초(ms)입니다.
  // 역할: Follow(끄는 동안 손에 붙음) · Settle(놓이거나 열릴 때 자리를 잡음) · Pop(눌렸다 튀어 오름) · Move(값의 자리로 감, 넘치지 않음) ·
  // Fade(투명도와 색) · Press(눌리는 순간) · Lead(시작 전의 뜸) · Seq(차례의 전체 시간) · Rest(되풀이 전의 쉼). 읽지 못한 값은 아래 기본값을 씁니다.
  const MOTION = { follow: '1000 63', settle: '340 26', pop: '520 21', move: '200 28.3', fade: '0.14s', press: '0.08s', lead: '400ms', seq: '1200ms', rest: '2600ms' };
  // 스프링이 0에서 1로 갈 때의 자리(t는 초). 잦아듦이 2√세기에 닿으면 넘치지 않고 곧장 갑니다.
  function springAt(k, c) {
    const w = Math.sqrt(k), z = c / (2 * w);
    if (z < 0.999) { const wd = w * Math.sqrt(1 - z * z); return (t) => 1 - Math.exp(-z * w * t) * (Math.cos(wd * t) + ((z * w) / wd) * Math.sin(wd * t)); }
    return (t) => 1 - (1 + w * t) * Math.exp(-w * t);
  }
  // 스프링을 CSS 전환으로 옮긴 [시간, 곡선]: 목표의 0.5% 안에 든 때까지를 시간으로 잡고, 그 사이를 33점으로 잇습니다.
  function springCss(k, c) {
    const f = springAt(k, c), pts = [];
    let d = 0;
    for (let t = 0; t <= 3; t += 0.005) if (Math.abs(f(t) - 1) > 0.005) d = t;
    d = Math.round((d + 0.01) * 100) / 100;
    for (let i = 0; i <= 32; i++) pts.push(i === 0 ? 0 : i === 32 ? 1 : +f((d * i) / 32).toFixed(3));
    return [`${d}s`, `linear(${pts.join(', ')})`];
  }
  // motion(card): 그 카드에 적용되는 움직임 값. pair(이름)은 스프링의 [세기, 잦아듦], ms(이름)은 밀리초,
  // curve(이름)은 { at(지난 밀리초) → 0에서 1까지의 자리, dur: 다 가는 데 걸리는 밀리초 }, css(이름)은 CSS 전환의 [시간, 곡선]입니다.
  function motion(card) {
    const cs = getComputedStyle(card), raw = (n) => cs.getPropertyValue(`--vzb-${n}`).trim().replace(/\s+/g, ' ');
    const two = (v) => { const p = v.split(' ').map(Number); return p.length === 2 && p[0] > 0 && p[1] > 0 ? p : null; };
    const pair = (n) => two(raw(n)) || two(MOTION[n]);
    const time = (v) => { const x = parseFloat(v); return isFinite(x) && x >= 0 && /s$/.test(v) ? (/ms$/.test(v) ? x : x * 1000) : null; };
    const ms = (n) => { const v = time(raw(n)); return v == null ? time(MOTION[n]) : v; };
    const css = (n) => springCss(...pair(n));
    const curve = (n) => { const f = springAt(...pair(n)); return { at: (t) => (t <= 0 ? 0 : f(t / 1000)), dur: parseFloat(css(n)[0]) * 1000 }; };
    return { pair, ms, css, curve };
  }
  // 스프링 값을 :root나 카드에서 바꿔 적었으면, CSS 전환이 쓰는 시간과 곡선(--vzb-…-t · -e)을 그 값으로 다시 계산해 카드에 적습니다.
  function tune(card) {
    const m = motion(card);
    for (const n of ['settle', 'pop', 'move']) {
      if (m.pair(n).join(' ') === MOTION[n]) continue;
      const [t, e] = m.css(n);
      card.style.setProperty(`--vzb-${n}-t`, t);
      card.style.setProperty(`--vzb-${n}-e`, e);
    }
  }
  // 스프링으로 목표에 다가가는 값: s = { x: 지금 자리, v: 속도 }를 dt(밀리초)만큼 옮깁니다. 가던 중에 목표가 바뀌어도 속도를 이어 갑니다.
  function springTo(s, to, k, c, dt) {
    let n = Math.max(1, Math.ceil(dt / 4));
    const h = dt / 1000 / n;
    while (n--) { s.v += (-k * (s.x - to) - c * s.v) * h; s.x += s.v * h; }
  }
  // 장면 잇기: 데이터가 바뀌어 카드를 갈아 끼울 때, 앞 카드가 화면에 그리고 있던 장면(값과 자리)을 새 카드에 넘깁니다.
  // 카드는 keep(card, 장면을 돌려주는 함수)로 맡기고, 새 카드는 before(card)로 받아 그 장면에서 새 값으로 옮겨 갑니다.
  const SCENE = new WeakMap(), FROM = new WeakMap();
  const keep = (card, fn) => SCENE.set(card, fn);
  const before = (card) => { const s = FROM.get(card); FROM.delete(card); return s || null; };
  // 같은 종류의 카드를 모두 찾아 그립니다. 코드를 두 번 붙여 넣어도 카드는 한 번만 그립니다.
  // 데이터가 바뀌면 다시 그립니다: 페이지의 스크립트가 카드 안 JSON(<script type="application/json">)의 글자를 바꾸면 그 카드만
  // 새 요소로 갈아 끼워 새로 그립니다(필터 · 기간 컨트롤과 잇는 길). 갈아 끼운 카드에는 data-vzb-live가 붙어, 움직이는 카드는
  // 처음부터 다시 재생하지 않습니다. 앞 카드가 장면을 맡겨 두었으면(keep) 그 장면에서 새 값으로 옮겨 가고, 아니면 마지막 장면을 바로 그립니다.
  // 예전 요소가 페이지에서 빠지므로 그 카드의 움직임도 함께 끝납니다.
  // 대비 최소선: 페이지가 바꾼 색으로 글자가 바탕과 4.5:1, 포커스 링이 바탕과 3:1에 못 미치면 콘솔에 한 줄로 알립니다.
  // 그림은 페이지 값 그대로 둡니다(브랜드 색을 카드가 고치지 않습니다). 색은 1×1 캔버스에 칠해 읽으므로 color-mix · oklch도 읽힙니다.
  const CHECKS = [['ink', 'card', 4.5], ['ink-2', 'card', 4.5], ['muted', 'card', 4.5], ['good-ink', 'card', 4.5], ['bad-ink', 'card', 4.5],
    ['alert-ink', 'card', 4.5], ['ink-2', 'track', 4.5], ['card', 'ink', 4.5], ['tip-ink', 'tip', 4.5], ['focus', 'card', 3]];
  // 같은 알림은 페이지에서 한 번만: 카드마다 core.js가 따로 들어 있으므로, 공유 기호(Symbol.for)로 document에 한 벌만 둡니다.
  const WARNED = Symbol.for('vizney-blocks.contrast');
  function checkContrast(card) {
    let ctx;
    try { ctx = document.createElement('canvas').getContext('2d', { willReadFrequently: true }); } catch (e) { return; }
    if (!ctx) return;
    const probe = el('i');
    probe.style.cssText = 'position:absolute;visibility:hidden';
    card.append(probe);
    const read = (name) => {
      probe.style.color = name === 'focus' ? 'var(--vzb-focus, var(--vzb-primary))' : `var(--vzb-${name})`;
      return getComputedStyle(probe).color;
    };
    const lum = (bg, fg) => {
      ctx.clearRect(0, 0, 1, 1);
      ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, 1, 1);
      ctx.fillStyle = bg; ctx.fillRect(0, 0, 1, 1);
      if (fg) { ctx.fillStyle = fg; ctx.fillRect(0, 0, 1, 1); }
      const [r, g, b] = ctx.getImageData(0, 0, 1, 1).data;
      const c = (v) => { v /= 255; return v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4; };
      return 0.2126 * c(r) + 0.7152 * c(g) + 0.0722 * c(b);
    };
    const low = [];
    CHECKS.forEach(([fg, bg, need]) => {
      const B = read(bg), a = lum(B), b = lum(B, read(fg));
      const ratio = (Math.max(a, b) + 0.05) / (Math.min(a, b) + 0.05);
      if (ratio < need - 0.005) low.push(`${fg === 'focus' ? '--vzb-focus' : `--vzb-${fg}`} on --vzb-${bg} ${ratio.toFixed(2)}:1 (needs ${need}:1)`);
    });
    probe.remove();
    const msg = low.join('; ');
    const warned = document[WARNED] || (document[WARNED] = new Set());
    if (!low.length || warned.has(msg)) return;
    warned.add(msg);
    console.warn(`Vizney Blocks: low contrast with the page's --vzb-* colors: ${msg}`);
  }
  function mount(type, draw) {
    const watch = (card, src) => {
      if (!src || !window.MutationObserver) return;
      const mo = new MutationObserver(() => {
        mo.disconnect();
        if (!card.isConnected) return;
        const next = card.cloneNode(false);
        next.removeAttribute('data-vzb-ready');
        next.setAttribute('data-vzb-live', '');
        next.append(src);
        const scene = SCENE.get(card);
        if (scene) { try { FROM.set(next, scene()); } catch (e) { /* 장면을 못 넘기면 마지막 장면을 바로 그립니다 */ } }
        card.replaceWith(next);
        run();
      });
      mo.observe(src, { childList: true, characterData: true, subtree: true });
    };
    const run = () => {
      document.querySelectorAll(`.vzb-card[data-vzb="${type}"]:not([data-vzb-ready])`).forEach((card) => {
        card.setAttribute('data-vzb-ready', '');
        const src = card.querySelector('script[type="application/json"]');
        const raw = src ? src.textContent : '';
        watch(card, src);
        let d;
        try { d = JSON.parse(raw); } catch (e) {
          card.append(el('p', 'vzb-error', tx(/"lang"\s*:\s*"en"/.test(raw) ? { lang: 'en' } : null, '카드 데이터(JSON)를 읽지 못했습니다: %s', e.message)));
          return;
        }
        card.setAttribute('role', 'group');
        if (d.title) card.setAttribute('aria-label', d.title);
        // 어두운 바탕: JSON에 "theme": "dark"를 적으면 Vizney 검정 바탕에 노랑을 첫째 색으로 그립니다(tokens.css의 .vzb-dark).
        card.classList.toggle('vzb-dark', d.theme === 'dark');
        tune(card);
        checkContrast(card);
        try { draw(card, d, format(d)); } catch (e) {
          card.append(el('p', 'vzb-error', tx(d, '카드를 그리지 못했습니다: %s', e.message)));
        }
      });
    };
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
    else run();
  }
  // 서식 길이: 페이지가 :root에 적은 --vzb-<name>(0 · 4px · 0.5rem 등)을 px 숫자로 읽습니다. 비워 두었으면(initial) fallback을 돌려줍니다.
  // 스크립트가 그리는 SVG(막대 끝, 말풍선, 칸)가 CSS와 같은 모서리 변수를 읽게 하는 길입니다. 길이는 카드에 잠깐 넣은 요소의 폭으로 재서, rem · calc도 읽힙니다.
  function len(card, name, fallback) {
    if (!getComputedStyle(card).getPropertyValue(`--vzb-${name}`).trim()) return fallback;
    const probe = el('i');
    probe.style.cssText = `position:absolute;visibility:hidden;width:var(--vzb-${name})`;
    card.append(probe);
    const v = parseFloat(getComputedStyle(probe).width);
    probe.remove();
    return isFinite(v) && v >= 0 ? v : fallback;
  }
  // 크기가 바뀔 때마다 그래픽을 다시 그립니다. 차트는 카드에 남은 높이를 채웁니다.
  function onSize(node, fn) {
    let lw = 0, lh = 0;
    new ResizeObserver((entries) => {
      const w = Math.floor(entries[0].contentRect.width), h = Math.floor(entries[0].contentRect.height);
      if (w > 0 && h > 0 && (w !== lw || h !== lh)) { lw = w; lh = h; fn(w, h); }
    }).observe(node);
  }

  /* 범위 슬라이더: 손잡이 둘로 범위를 고릅니다. 카드 머리 없이 조작부만 그립니다.
     고른 값은 손잡이 바로 아래에 붙어 함께 움직이고, 두 값이 가까워 겹치면 '앞 – 뒤' 하나로 합칩니다.
     분포(hist)를 주면 끄는 동안과 키보드로 옮기는 동안만 레일 위에 분포 창이 떠올라, 고른 범위 안의 막대를 글자색으로 채우고
     범위 안 개수를 굴러가는 숫자로 적습니다(가만히 둘 때는 레일과 값만 보여 필터 줄에서 자리를 적게 차지합니다).
     손잡이가 지나가며 막대가 범위에 들어오면 손잡이에서 가까운 차례로 살짝 솟았다 내려앉고, 나가면 살짝 꺼집니다.
     cfg: { n: 칸 수(자리는 0…n), text(i): 그 자리의 글자, value(i): 이벤트로 보내는 값, start: [앞, 뒤] 자리, aria: [앞 손잡이 이름, 뒤 손잡이 이름],
            hist: { counts: min~max를 같은 폭으로 나눈 칸마다의 개수, min, max, noun: 개수 뒤에 붙는 말(비우면 행) },
            edit: { raw(i): 입력칸에 넣을 숫자 글자, read(글자): 그 값의 자리(칸 사이도 됨) 또는 null, pre · suf: 입력칸 앞뒤의 단위 } }
     (hist가 있으면 value(i)는 숫자). 자리는 끌기와 화살표로는 칸마다 옮기고, 직접 친 값은 칸 사이의 자리(소수)도 그대로 씁니다.
     edit가 있으면 손잡이 아래 값을 누르거나 손잡이에서 Enter를 눌러 그 자리에 입력칸을 열고 값을 칩니다(Enter · 칸 밖으로 나가면 적용, Esc는 취소).
     값이 바뀌면 카드에서 vzb-change 이벤트(bubbles, detail: { name, value: [시작, 끝] })를 보내고, 카드의 data-vzb-value에 같은 값을 적습니다.
     조작: 손잡이를 끌거나, 레일을 눌러 가까운 손잡이를 옮기거나, 손잡이에 포커스한 뒤 화살표(한 칸) · Page Up/Down(열 칸쯤) · Home/End로 옮깁니다.
     두 손잡이는 같은 칸까지 만날 수 있고, 겹쳐 있을 때는 끄는 방향의 손잡이가 움직입니다. 손잡이는 용수철로 따라갑니다.
     카드를 감싼 요소에 data-vzb-loop가 있으면(갤러리)
     손을 대기 전까지 몇 가지 범위를 차례로 보여 줍니다. */
  function slider(card, d, f, cfg) {
    const n = cfg.n, R = 10, hist = cfg.hist && cfg.hist.counts.length ? cfg.hist : null;   // R: 손잡이 반지름. 레일과 막대는 양 끝에서 그만큼 들어옵니다.
    let lo = cfg.start[0], hi = cfg.start[1];
    const root = el('div', 'vzb-sl'), area = el('div', 'vzb-sl-area'), say = el('p', 'vzb-sr'), sr = el('p', 'vzb-sr');
    say.setAttribute('aria-live', 'polite');
    area.setAttribute('data-vzb-input', '');
    const gap = f.lang === 'en' ? ' ' : '';
    const nounOf = (c) => gap + (hist.noun || tx(d, c === 1 ? '행|1' : '행'));
    // 분포 창: 끄는 동안과 키보드로 옮기는 동안만 레일 위에 떠오릅니다. 막대는 레일의 칸과 같은 자리에 섭니다.
    let pop = null, count = null, nounEl = null, bars = [];
    if (hist) {
      pop = el('div', 'vzb-sl-pop');
      pop.setAttribute('aria-hidden', 'true');
      const top = el('div', 'vzb-sl-pophd');
      count = el('span', 'vzb-rl');
      nounEl = el('span', 'vzb-sl-noun');
      top.append(count, nounEl);
      const box = el('div', 'vzb-sl-hist'), peak = Math.max(1, ...hist.counts);
      bars = hist.counts.map((c) => { const b = el('i', 'vzb-sl-bar'); b.style.height = `${Math.max(6, (Math.max(0, +c || 0) / peak) * 100)}%`; box.append(b); return b; });
      pop.append(top, box);
      area.append(pop);
    }
    const knob = (name) => {
      const k = el('span', 'vzb-sl-thumb');
      k.tabIndex = 0;
      k.setAttribute('role', 'slider');
      k.setAttribute('aria-label', name);
      k.setAttribute('aria-valuemin', 0);
      k.setAttribute('aria-valuemax', n);
      return k;
    };
    const rail = el('div', 'vzb-sl-rail'), fill = el('span', 'vzb-sl-fill'), A = knob(cfg.aria[0]), B = knob(cfg.aria[1]);
    rail.append(el('span', 'vzb-sl-track'), fill, A, B);
    // 손잡이 아래의 값: 앞 · 뒤 하나씩, 겹치면 합친 하나
    const labs = el('div', 'vzb-sl-labs'), la = el('span', 'vzb-sl-lab'), lb = el('span', 'vzb-sl-lab'), lm = el('span', 'vzb-sl-lab');
    labs.setAttribute('aria-hidden', 'true');
    labs.append(la, lb, lm);
    area.append(rail, labs);
    // 직접 입력: 앞 · 뒤 입력칸을 '–'로 이어 손잡이 사이 가운데 아래에 띄웁니다(레일 아래 값 자리를 덮고, 아래 내용은 밀지 않습니다).
    const ed = cfg.edit ? el('div', 'vzb-sl-ed') : null, inA = cfg.edit ? el('input', 'vzb-sl-in') : null, inB = cfg.edit ? el('input', 'vzb-sl-in') : null;
    if (ed) {
      const field = (inp, k) => {
        const box = el('label', 'vzb-sl-field');
        inp.type = 'text';
        inp.inputMode = 'decimal';
        inp.autocomplete = 'off';
        inp.spellcheck = false;
        inp.setAttribute('aria-label', cfg.aria[k]);
        if (cfg.edit.pre) box.append(el('span', 'vzb-sl-affix', cfg.edit.pre));
        box.append(inp);
        if (cfg.edit.suf) box.append(el('span', 'vzb-sl-affix', cfg.edit.suf));
        return box;
      };
      ed.append(field(inA, 0), el('span', 'vzb-sl-dash', '–'), field(inB, 1));
      ed.hidden = true;
      labs.append(ed);
      labs.classList.add('is-editable');
      [la, lb, lm].forEach((x) => { x.title = tx(d, '눌러서 값 입력'); });
    }
    root.append(area, sr, say);
    card.append(root);

    // 막대가 범위에 드는지와 범위 안의 개수: 칸이 범위에 걸친 만큼 나눠 셉니다(step을 칸 폭에 맞추면 정확히 셉니다).
    const width = hist ? (hist.max - hist.min) / bars.length : 1;
    const inRange = (i, a, b) => { const c = hist.min + width * (i + 0.5); return c >= a && c <= b; };
    const counted = (a, b) => hist.counts.reduce((s, c, i) => {
      const x0 = hist.min + width * i, x1 = x0 + width;
      return s + (Math.max(0, +c || 0) * Math.max(0, Math.min(x1, b) - Math.max(x0, a))) / width;
    }, 0);
    let mover = null;   // 막 움직인 손잡이: 막대가 그 손잡이에서 가까운 차례로 솟습니다
    function paintBars(first) {
      const a = cfg.value(lo), b = cfg.value(hi), changed = [];
      bars.forEach((bar, i) => {
        const on = inRange(i, a, b);
        if (bar.classList.contains('is-in') !== on) { bar.classList.toggle('is-in', on); changed.push([bar, on, i]); }
      });
      if (first || !bars[0] || !bars[0].animate) return;
      const from = mover === A ? lo : hi;
      changed.sort((p, q) => Math.abs(p[2] - (from / n) * bars.length) - Math.abs(q[2] - (from / n) * bars.length))
        .forEach(([bar, on], k) => bar.animate(on ? [{ transform: 'scaleY(1)' }, { transform: 'scaleY(1.22)' }, { transform: 'scaleY(1)' }] : [{ transform: 'scaleY(1)' }, { transform: 'scaleY(0.82)' }, { transform: 'scaleY(1)' }],
          { duration: on ? 340 : 260, delay: Math.min(220, k * 24), easing: 'ease-out' }));
    }
    // 분포 창 보이기: 조작하는 동안 켜고, 손을 떼거나 키를 멈추면 조금 뒤 끕니다.
    let popT = 0;
    function reveal(on, after) {
      if (!pop) return;
      clearTimeout(popT);
      if (on) root.classList.add('is-on');
      if (after != null) popT = setTimeout(() => { if (!drag) root.classList.remove('is-on'); }, after);
    }
    let sayT = 0, wa = 0, wb = 0, wm = 0;
    function write(first) {
      const a = cfg.text(lo), b = cfg.text(hi), words = lo === hi ? a : `${a} – ${b}`;
      la.textContent = a;
      lb.textContent = b;
      lm.textContent = words;
      wa = la.offsetWidth; wb = lb.offsetWidth; wm = lm.offsetWidth;
      A.setAttribute('aria-valuenow', lo);
      A.setAttribute('aria-valuetext', a);
      B.setAttribute('aria-valuenow', hi);
      B.setAttribute('aria-valuetext', b);
      let tail = '';
      if (hist) {
        const c = Math.round(counted(cfg.value(lo), cfg.value(hi)));
        roll(count, num(c, 0));
        nounEl.textContent = nounOf(c);
        tail = `, ${num(c, 0)}${nounOf(c)}`;
        paintBars(first);
      }
      sr.textContent = words + tail;
      if (first) return;
      clearTimeout(sayT);
      sayT = setTimeout(() => { say.textContent = words + tail; }, 500);
    }
    function emit(quiet) {
      const value = [cfg.value(lo), cfg.value(hi)];
      card.setAttribute('data-vzb-value', JSON.stringify(value));
      if (!quiet) card.dispatchEvent(new CustomEvent('vzb-change', { bubbles: true, detail: { name: d.name || '', value } }));
    }
    function set(a, b) {
      if (a === lo && b === hi) return;
      lo = a; hi = b;
      write();
      emit();
    }

    // 손잡이의 가로 자리(px)는 용수철 둘이 따라갑니다. k는 세기, c는 잦아드는 힘입니다.
    const spring = (k, c) => ({ x: 0, v: 0, t: 0, k, c });
    const FOLLOW = motion(card).pair('follow'), SETTLE = motion(card).pair('settle');
    const sa = spring(...SETTLE), sb = spring(...SETTLE), both = [sa, sb];
    const held = (s) => { [s.k, s.c] = FOLLOW; };    // 끄는 동안(Follow): 넘치지 않고 손가락에 붙습니다
    const freed = (s) => { [s.k, s.c] = SETTLE; };   // 놓은 뒤 · 키보드(Settle): 살짝 넘쳤다 자리를 잡습니다
    let W = 0, drag = null, raf = 0, last = 0, idle = 0;
    const pos = (i) => R + (i / n) * Math.max(1, W - 2 * R);
    const at = (x) => Math.round(((x - R) / Math.max(1, W - 2 * R)) * n);
    function draw() {
      A.style.transform = `translateX(${sa.x.toFixed(2)}px)`;
      B.style.transform = `translateX(${sb.x.toFixed(2)}px)`;
      fill.style.transform = `translateX(${sa.x.toFixed(2)}px) scaleX(${Math.max(0, sb.x - sa.x).toFixed(2)})`;
      // 값은 손잡이 가운데 아래에 두되 레일 밖으로 나가지 않게 하고, 두 값이 8px 안으로 붙으면 합친 하나를 가운데에 둡니다.
      const at0 = (x, w) => Math.max(0, Math.min(Math.max(0, W - w), x - w / 2)), xa = at0(sa.x, wa), xb = at0(sb.x, wb);
      const one = lo === hi || xa + wa + 8 > xb;
      la.style.transform = `translateX(${xa.toFixed(1)}px)`;
      lb.style.transform = `translateX(${xb.toFixed(1)}px)`;
      lm.style.transform = `translateX(${at0((sa.x + sb.x) / 2, wm).toFixed(1)}px)`;
      la.classList.toggle('is-off', one);
      lb.classList.toggle('is-off', one);
      lm.classList.toggle('is-off', !one);
      if (editing) placeEd();
    }
    function frame(now) {
      raf = 0;
      if (!card.isConnected) return;
      const dt = Math.min(0.033, last ? (now - last) / 1000 : 0.016) / 4;
      last = now;
      for (let i = 0; i < 4; i++) both.forEach((s) => { s.v += (-s.k * (s.x - s.t) - s.c * s.v) * dt; s.x += s.v * dt; });
      draw();
      if (drag || both.some((s) => Math.abs(s.x - s.t) > 0.05 || Math.abs(s.v) > 1)) raf = requestAnimationFrame(frame);
      else { jump(); last = 0; }
    }
    function jump() { both.forEach((s) => { s.x = s.t; s.v = 0; }); draw(); }
    const wake = () => { if (!raf) raf = requestAnimationFrame(frame); };
    function rest() { freed(sa); freed(sb); sa.t = pos(lo); sb.t = pos(hi); wake(); }

    // 끌기: 조작 영역 어디를 눌러도 가까운 손잡이가 따라옵니다. 두 손잡이가 같은 자리면 끄는 방향으로 정합니다.
    const local = (e) => e.clientX - area.getBoundingClientRect().left;
    function move(x) {
      if (!drag.who) {
        if (Math.abs(x - drag.x) < 2) return;
        drag.who = x > drag.x ? B : A;
      }
      const first = drag.who === A, s = first ? sa : sb;
      mover = drag.who;
      drag.who.classList.add('is-held');
      if (!drag.held) { drag.held = true; held(s); }
      const px = Math.max(R, Math.min(W - R, x - drag.off));
      s.t = first ? Math.min(px, sb.t) : Math.max(px, sa.t);
      if (first) set(Math.min(at(s.t), hi), hi); else set(lo, Math.max(at(s.t), lo));
      wake();
    }
    area.addEventListener('pointerdown', (e) => {
      if (e.button) return;
      if (ed && e.target.closest('.vzb-sl-ed')) return;   // 입력칸 안의 누르기는 끌기가 아닙니다
      if (editing) finish(true);                          // 입력하던 중에 레일을 누르면 친 값을 적용한 뒤 끕니다
      if (ed && e.target.closest('.vzb-sl-lab')) {        // 값을 누르면 입력칸을 엽니다(합친 값은 누른 쪽에 가까운 손잡이)
        e.preventDefault();
        const x = local(e);
        edit(e.target === la ? 0 : e.target === lb ? 1 : Math.abs(x - sa.t) <= Math.abs(x - sb.t) ? 0 : 1);
        return;
      }
      const x = local(e), da = Math.abs(x - sa.t), db = Math.abs(x - sb.t);
      drag = { x, who: da < db ? A : db < da ? B : x > sb.t ? B : x < sa.t ? A : null };
      drag.off = Math.min(da, db) <= R + 2 ? x - (da <= db ? sa.t : sb.t) : 0;   // 손잡이를 잡았으면 잡은 자리를 지키고, 막대 · 트랙을 눌렀으면 손잡이가 그리로 옵니다
      idle = Infinity;
      reveal(true);
      area.classList.add('is-drag');
      area.setPointerCapture(e.pointerId);
      move(x);
    });
    area.addEventListener('pointermove', (e) => { if (drag) move(local(e)); });
    const drop = () => {
      if (!drag) return;
      drag = null;
      idle = performance.now() + 6000;
      area.classList.remove('is-drag');
      A.classList.remove('is-held');
      B.classList.remove('is-held');
      reveal(false, 900);
      rest();
    };
    area.addEventListener('pointerup', drop);
    area.addEventListener('pointercancel', drop);
    // 키보드: 화살표는 한 칸, Page Up · Down은 열 칸쯤, Home · End는 갈 수 있는 끝(다른 손잡이를 넘지 않습니다)
    const keys = (first) => (e) => {
      if (ed && e.key === 'Enter') { e.preventDefault(); edit(first ? 0 : 1); return; }
      const big = Math.max(1, Math.round(n / 10));
      const by = { ArrowRight: 1, ArrowUp: 1, ArrowLeft: -1, ArrowDown: -1, PageUp: big, PageDown: -big }[e.key];
      let v = first ? lo : hi;
      if (by) v = Math.round(v) + by;   // 직접 친 값(칸 사이)에서도 가까운 칸에서 한 칸씩 옮깁니다 else if (e.key === 'Home') v = first ? 0 : lo; else if (e.key === 'End') v = first ? hi : n; else return;
      e.preventDefault();
      idle = performance.now() + 6000;
      reveal(true, 1400);
      mover = first ? A : B;
      if (first) set(Math.max(0, Math.min(hi, v)), hi); else set(lo, Math.max(lo, Math.min(n, v)));
      rest();
    };
    A.addEventListener('keydown', keys(true));
    B.addEventListener('keydown', keys(false));

    // 직접 입력의 흐름: 열면 지금 값을 넣고 누른 쪽을 고릅니다. 치는 동안 손잡이 · 분포가 따라오고(이벤트는 보내지 않음),
    // Enter나 입력칸 밖으로 나가면 적용해 이벤트를 한 번 보냅니다. 앞 값이 뒤 값보다 크면 서로 바꿉니다. Esc는 연 때의 값으로 되돌립니다.
    let editing = null;
    const readOr = (inp, or) => { const i = cfg.edit.read(inp.value); return i == null ? or : Math.max(0, Math.min(n, i)); };
    function placeEd() {
      if (!editing) return;
      const w = ed.offsetWidth, x = Math.max(0, Math.min(Math.max(0, W - w), (sa.t + sb.t) / 2 - w / 2));
      ed.style.transform = `translateX(${x.toFixed(1)}px)`;
    }
    function edit(k) {
      if (!ed) return;
      if (!editing) {
        editing = { lo, hi };
        idle = Infinity;
        inA.value = cfg.edit.raw(lo);
        inB.value = cfg.edit.raw(hi);
        const len = Math.max(5, cfg.edit.raw(n).length + 3, cfg.edit.raw(0).length + 3);   // 소수 두 자리까지 들어갈 폭
        inA.style.width = inB.style.width = `${len}ch`;
        ed.hidden = false;
        labs.classList.add('is-editing');
        reveal(true);
        placeEd();
      }
      const inp = k ? inB : inA;
      inp.focus();
      inp.select();
    }
    function preview() {
      const a = readOr(inA, editing.lo), b = readOr(inB, editing.hi);
      lo = Math.min(a, b); hi = Math.max(a, b);
      write();
      rest();
    }
    function finish(apply, back) {
      if (!editing) return;
      const was = editing;
      if (apply) preview(); else { lo = was.lo; hi = was.hi; write(); rest(); }
      editing = null;
      ed.hidden = true;
      labs.classList.remove('is-editing');
      idle = performance.now() + 6000;
      reveal(false, 900);
      if (lo !== was.lo || hi !== was.hi) emit();
      if (back != null) (back ? B : A).focus();
    }
    if (ed) {
      [inA, inB].forEach((inp, k) => {
        inp.addEventListener('input', () => { if (editing) preview(); });
        inp.addEventListener('keydown', (e) => {
          if (e.key === 'Enter') { e.preventDefault(); finish(true, k); }
          else if (e.key === 'Escape') { e.preventDefault(); e.stopPropagation(); finish(false, k); }
        });
      });
      ed.addEventListener('focusout', (e) => { if (editing && !(e.relatedTarget && ed.contains(e.relatedTarget))) finish(true); });
    }

    write(true);
    emit(true);
    onSize(rail, (w) => { W = w; wa = la.offsetWidth; wb = lb.offsetWidth; wm = lm.offsetWidth; sa.t = pos(lo); sb.t = pos(hi); jump(); });
    if (document.fonts && document.fonts.ready) document.fonts.ready.then(() => { if (W) { wa = la.offsetWidth; wb = lb.offsetWidth; wm = lm.offsetWidth; draw(); } });

    // 갤러리(data-vzb-loop): 화면에 보이는 동안, 손을 대기 전과 손을 뗀 지 6초 뒤부터 몇 가지 범위를 차례로 보여 줍니다.
    if (card.closest('[data-vzb-loop]')) {
      const s0 = [lo / n, hi / n], plan = [[0.08, 0.36], [0.24, 0.72], [0.04, 0.52], [0.44, 0.92], s0];
      let k = 0, seen = true;
      if ('IntersectionObserver' in window) new IntersectionObserver((es) => { seen = es[es.length - 1].isIntersecting; }).observe(card);
      const timer = setInterval(() => {
        if (!card.isConnected) { clearInterval(timer); return; }
        if (drag || !seen || !W || performance.now() < idle) return;
        const p = plan[k++ % plan.length], a = Math.round(p[0] * n), b = Math.max(a, Math.round(p[1] * n));
        mover = Math.abs(a - lo) >= Math.abs(b - hi) ? A : B;
        reveal(true, 1150);
        set(a, b);
        rest();
      }, 1800);
    }
  }

  /* 굴러가는 숫자(컨트롤 공통): 숫자 한 자리마다 0–9를 세로로 쌓은 칸을 두고 새 값의 자리로 굴립니다(구독 수가 오르는 단추처럼).
     roll(box, text): box 안의 글자를 text로 바꿉니다. 자리 수와 쉼표 자리가 그대로면 칸이 제자리에서 구르고, 바뀌면 칸을 새로 만들어
     0에서 굴러 올라옵니다. 움직임은 roll.css의 transition이 그리고, box는 화면 낭독기에 읽히지 않으므로 값은 부르는 쪽이 따로 알립니다. */
  function roll(box, text) {
    const chars = [...String(text)], shape = chars.map((c) => (/\d/.test(c) ? '0' : c)).join('');
    box.setAttribute('aria-hidden', 'true');
    if (box.dataset.shape !== shape) {
      box.dataset.shape = shape;
      box.replaceChildren(...chars.map((c) => {
        if (!/\d/.test(c)) return el('span', 'vzb-rl-ch', c);
        const col = el('span', 'vzb-rl-col'), strip = el('span', 'vzb-rl-strip');
        for (let k = 0; k < 10; k++) strip.append(el('span', null, String(k)));
        col.append(strip);
        return col;
      }));
      box.getBoundingClientRect();   // 새 칸은 0에 선 채로 그려진 뒤 굴러갑니다
    }
    [...box.children].forEach((node, i) => { if (node.classList.contains('vzb-rl-col')) node.firstChild.style.transform = `translateY(${-10 * +chars[i]}%)`; });
  }

  // Range Slider: 숫자 범위를 손잡이 둘로 고르는 컨트롤입니다(주문 금액 · 수량 필터). min과 max 사이를 끌거나 화살표로 step 간격으로 고르고,
  // 손잡이 아래 값을 눌러(손잡이에서 Enter) 칸에 맞지 않는 값도 직접 칠 수 있습니다
  // (step을 비우면 범위의 1/100에 가까운 1 · 2 · 5와 그 열 배들), 고른 값은 format대로 손잡이 아래에 붙어 따라갑니다. value는 처음 범위 [시작, 끝]이고 비우면 전체입니다.
  // counts(min~max를 같은 폭으로 나눈 칸마다의 개수)를 주면 끄는 동안과 키보드로 옮기는 동안 레일 위에 분포 막대가 떠오르고, 범위 안의 개수를 noun(비우면 행)과 함께 적습니다..
  // 값이 바뀌면 vzb-change 이벤트의 detail.value로 [시작, 끝] 숫자를 보냅니다(src/slider.js).
  mount('range-slider', (card, d, f) => {
    const min = +d.min, max = +d.max;
    if (!fin(d.min) || !fin(d.max) || !(max > min)) throw new Error(tx(d, 'min과 max를 확인하세요.'));
    let step = fin(d.step) && +d.step > 0 ? +d.step : 0;
    if (!step) {
      const raw = (max - min) / 100, p = 10 ** Math.floor(Math.log10(raw)), q = raw / p;
      step = (q < 1.5 ? 1 : q < 3.5 ? 2 : q < 7.5 ? 5 : 10) * p;
    }
    const n = Math.max(1, Math.round((max - min) / step));
    const at = (i) => (i >= n ? max : +(min + i * step).toPrecision(12));
    // 자리: 값이 min에서 몇 칸째인지. 칸 사이의 값(직접 친 값, 칸에 맞지 않는 처음 값)은 소수 자리로 그대로 둡니다.
    const seat = (v, or) => (fin(v) ? Math.max(0, Math.min(n, (+v - min) / step)) : or);
    const v = Array.isArray(d.value) ? d.value : [], a = seat(v[0], 0), b = seat(v[1], n);
    const name = d.label || d.title || tx(d, '범위');   // 제목을 그리지 않으므로 화면 낭독기용 이름은 label에서 읽습니다
    const hist = Array.isArray(d.counts) && d.counts.length ? { counts: d.counts.map((c) => (fin(c) ? +c : 0)), min, max, noun: d.noun || '' } : null;
    // 직접 입력: 입력칸에는 단위 없는 숫자(천 단위 쉼표)를 넣고, 단위는 칸 앞뒤에 둡니다($ · 원 · unit 등). 칠 때는 쉼표 · 단위를 빼고 읽고,
    // K · M · B(천 · 백만 · 십억)와 천 · 만 · 억을 붙인 값도 읽습니다(6만 → 60000, 1.2K → 1200).
    const affix = { usd: ['$', ''], krw: ['', '원'], eok: ['', '억'], pct: ['', '%'], days: ['', tx(d, '일')] }[d.format] || ['', d.unit || ''];
    const plain = (v) => num(v, Number.isInteger(v) ? 0 : 2).replace(/(\.\d*?)0+$/, '$1').replace(/\.$/, '');
    // 배수 글자(화면에 쓰지 않으므로 유니코드로 적습니다): \uCC9C 천 · \uB9CC 만 · \uC5B5 억 · \uC870 조. eok은 값이 이미 억 단위입니다.
    const MUL = d.format === 'eok' ? { '\uC5B5': 1, '\uC870': 1e4 } : { k: 1e3, m: 1e6, b: 1e9, '\uCC9C': 1e3, '\uB9CC': 1e4, '\uC5B5': 1e8 };
    const read = (s) => {
      const m = /(-?\d+(?:\.\d+)?)\s*([kmb\uCC9C\uB9CC\uC5B5\uC870])?/i.exec(String(s).replace(/,/g, ''));
      if (!m) return null;
      const v = parseFloat(m[1]) * (MUL[(m[2] || '').toLowerCase()] || 1);
      return isFinite(v) ? seat(Math.max(min, Math.min(max, v)), null) : null;
    };
    slider(card, d, f, { n, text: (i) => show(f, at(i)), value: at, start: [Math.min(a, b), Math.max(a, b)], aria: [tx(d, '%s 최소', name), tx(d, '%s 최대', name)], hist,
      edit: { raw: (i) => plain(at(i)), read, pre: affix[0], suf: affix[1] } });
  });
})();
</script>
